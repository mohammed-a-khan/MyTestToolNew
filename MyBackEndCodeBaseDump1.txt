package com.cstestforge.framework.core;

import com.cstestforge.framework.config.ConfigurationManager;
import com.cstestforge.framework.driver.DriverManager;
import com.cstestforge.framework.reporting.CSReporter;
import com.cstestforge.framework.utils.CSScreenshotManager;
import com.cstestforge.framework.utils.CSWaitStrategy;
import org.openqa.selenium.WebDriver;
import org.testng.ITestResult;
import org.testng.annotations.*;

import java.lang.reflect.Field;
import java.lang.reflect.Method;
import java.util.Arrays;

/**
 * Base test class for all CSTestForge test cases.
 * Handles common test lifecycle operations including:
 * - Driver initialization and cleanup
 * - Page object initialization
 * - Configuration loading
 * - Reporting and logging
 * - Screenshots
 */
public class CSBaseTest {

    protected WebDriver driver;
    protected CSReporter reporter;
    protected ConfigurationManager configManager;
    protected CSWaitStrategy waitStrategy;
    protected CSScreenshotManager screenshotManager;

    /**
     * Initialize common test components before the test suite
     */
    @BeforeSuite(alwaysRun = true)
    public void beforeSuite() {
        // Initialize configuration
        configManager = ConfigurationManager.getInstance();
        configManager.loadConfiguration();
        
        // Initialize reporter
        reporter = CSReporter.getInstance();
        reporter.startReporting(getClass().getSimpleName());
        
        reporter.info("Test suite started: " + getClass().getSimpleName());
    }
    
    /**
     * Initialize driver and other components before each test method
     * 
     * @param method Test method
     * @param parameters Test parameters
     */
    @BeforeMethod(alwaysRun = true)
    public void beforeMethod(Method method, Object[] parameters) {
        // Get test information
        String testName = method.getName();
        String testClass = getClass().getSimpleName();
        String testDescription = "";
        
        // Get test description from TestNG annotation if present
        Test testAnnotation = method.getAnnotation(Test.class);
        if (testAnnotation != null && !testAnnotation.description().isEmpty()) {
            testDescription = testAnnotation.description();
        }
        
        // Start test reporting
        reporter.startTest(testName, testDescription);
        reporter.info("Starting test: " + testName);
        
        // Initialize driver
        driver = DriverManager.getDriver(configManager.getBrowserType());
        
        // Initialize wait strategy
        waitStrategy = new CSWaitStrategy(driver, configManager.getExplicitWaitTimeout());
        
        // Initialize screenshot manager
        screenshotManager = new CSScreenshotManager(driver);
        
        // Initialize page objects
        initializePageObjects();
        
        // Log test parameters if present
        if (parameters != null && parameters.length > 0) {
            reporter.info("Test parameters: " + Arrays.toString(parameters));
        }
    }
    
    /**
     * Clean up after each test method
     * 
     * @param result Test result
     */
    @AfterMethod(alwaysRun = true)
    public void afterMethod(ITestResult result) {
        // Capture test status
        String testName = result.getName();
        
        // Handle test result
        switch (result.getStatus()) {
            case ITestResult.SUCCESS:
                reporter.pass("Test passed: " + testName);
                break;
            case ITestResult.FAILURE:
                // Capture screenshot on failure
                String screenshotPath = screenshotManager.captureScreenshot("failure_" + testName);
                reporter.fail("Test failed: " + testName + "\nError: " + result.getThrowable().getMessage());
                reporter.attachScreenshot(screenshotPath);
                break;
            case ITestResult.SKIP:
                reporter.skip("Test skipped: " + testName);
                if (result.getThrowable() != null) {
                    reporter.info("Skip reason: " + result.getThrowable().getMessage());
                }
                break;
            default:
                reporter.info("Test status unknown: " + testName + " (Status: " + result.getStatus() + ")");
        }
        
        // End test reporting
        reporter.endTest();
        
        // Close driver
        if (driver != null) {
            DriverManager.quitDriver();
            driver = null;
        }
    }
    
    /**
     * Clean up after the test suite
     */
    @AfterSuite(alwaysRun = true)
    public void afterSuite() {
        // End reporting
        reporter.info("Test suite completed: " + getClass().getSimpleName());
        reporter.endReporting();
        
        // Cleanup any remaining resources
        DriverManager.quitAllDrivers();
    }
    
    /**
     * Initialize page objects in the test class
     */
    private void initializePageObjects() {
        try {
            // Find all fields with CSPage type
            Field[] fields = getClass().getDeclaredFields();
            
            for (Field field : fields) {
                // Check if field is a CSBasePage subclass
                if (CSBasePage.class.isAssignableFrom(field.getType())) {
                    // Make field accessible
                    field.setAccessible(true);
                    
                    try {
                        // Create new instance of the page object
                        CSBasePage pageInstance = (CSBasePage) field.getType()
                                .getConstructor(WebDriver.class)
                                .newInstance(driver);
                        
                        // Initialize elements in the page object
                        pageInstance.initElements();
                        
                        // Set field value
                        field.set(this, pageInstance);
                        
                        reporter.debug("Initialized page object: " + field.getName() + " (" + field.getType().getSimpleName() + ")");
                    } catch (Exception e) {
                        reporter.error("Failed to initialize page object: " + field.getName(), e);
                    }
                }
            }
        } catch (Exception e) {
            reporter.error("Error initializing page objects", e);
        }
    }
    
    /**
     * Execute a test step
     * 
     * @param step Test step
     * @param description Step description
     */
    protected void executeStep(CSTestStep step, String description) {
        reporter.info("Step: " + description);
        
        try {
            step.execute();
            reporter.pass("Step completed: " + description);
        } catch (Exception e) {
            // Capture screenshot
            String screenshotPath = screenshotManager.captureScreenshot("step_failure");
            
            reporter.fail("Step failed: " + description + "\nError: " + e.getMessage());
            reporter.attachScreenshot(screenshotPath);
            
            throw e; // Re-throw to fail the test
        }
    }
}


package com.cstestforge.framework.core;

import org.openqa.selenium.WebElement;
import org.openqa.selenium.support.pagefactory.DefaultElementLocatorFactory;
import org.openqa.selenium.support.pagefactory.DefaultFieldDecorator;
import org.openqa.selenium.support.pagefactory.ElementLocator;
import org.openqa.selenium.support.pagefactory.ElementLocatorFactory;

import java.lang.reflect.Constructor;
import java.lang.reflect.Field;
import java.lang.reflect.InvocationTargetException;

/**
 * Base page class for all page objects.
 * Implements Page Object Model pattern with enhanced element handling.
 */
public class CSBasePage {

    protected WebDriver driver;
    protected CSReporter reporter;
    protected CSWaitStrategy waitStrategy;
    protected ConfigurationManager configManager;
    protected String pageName;

    /**
     * Constructor
     * 
     * @param driver WebDriver instance
     */
    public CSBasePage(WebDriver driver) {
        this.driver = driver;
        this.reporter = CSReporter.getInstance();
        this.configManager = ConfigurationManager.getInstance();
        this.waitStrategy = new CSWaitStrategy(driver, configManager.getExplicitWaitTimeout());
        this.pageName = getClass().getSimpleName();
    }

    /**
     * Initialize all elements in the page
     */
    public void initElements() {
        try {
            // Create element locator factory
            ElementLocatorFactory factory = new DefaultElementLocatorFactory(driver);
            
            // Get all fields in the class
            Class<?> clazz = this.getClass();
            Field[] fields = clazz.getDeclaredFields();

            // Process each field
            for (Field field : fields) {
                // Check if field is annotated with @CSFindBy
                if (field.isAnnotationPresent(CSFindBy.class)) {
                    // Make field accessible
                    field.setAccessible(true);
                    
                    // Get the annotation
                    CSFindBy findBy = field.getAnnotation(CSFindBy.class);
                    
                    // Create element locator
                    By by = createByFromFindBy(findBy);
                    
                    // Check if field is of type CSElement
                    if (field.getType().equals(CSElement.class)) {
                        // Create and set CSElement instance
                        CSElement element = createCSElement(by, field.getName());
                        field.set(this, element);
                    } else if (WebElement.class.isAssignableFrom(field.getType())) {
                        // Find element and set field
                        ElementLocator locator = factory.createLocator(field);
                        WebElement element = locator.findElement();
                        field.set(this, element);
                    }
                }
            }
            
            reporter.debug("Initialized page elements for: " + pageName);
        } catch (Exception e) {
            reporter.error("Failed to initialize page elements for: " + pageName, e);
            throw new RuntimeException("Failed to initialize page elements", e);
        }
    }
    
    /**
     * Create By locator from CSFindBy annotation
     * 
     * @param findBy CSFindBy annotation
     * @return By locator
     */
    private By createByFromFindBy(CSFindBy findBy) {
        // Check each locator type in priority order
        if (!findBy.id().isEmpty()) {
            return By.id(findBy.id());
        } else if (!findBy.xpath().isEmpty()) {
            return By.xpath(findBy.xpath());
        } else if (!findBy.css().isEmpty()) {
            return By.cssSelector(findBy.css());
        } else if (!findBy.name().isEmpty()) {
            return By.name(findBy.name());
        } else if (!findBy.className().isEmpty()) {
            return By.className(findBy.className());
        } else if (!findBy.linkText().isEmpty()) {
            return By.linkText(findBy.linkText());
        } else if (!findBy.partialLinkText().isEmpty()) {
            return By.partialLinkText(findBy.partialLinkText());
        } else if (!findBy.tagName().isEmpty()) {
            return By.tagName(findBy.tagName());
        }
        
        throw new IllegalArgumentException("@CSFindBy must specify at least one locator");
    }
    
    /**
     * Create a CSElement instance
     * 
     * @param by By locator
     * @param name Element name
     * @return CSElement instance
     */
    private CSElement createCSElement(By by, String name) {
        try {
            // Create CSElement using reflection
            Constructor<CSElement> constructor = CSElement.class.getConstructor(
                    WebDriver.class, By.class, String.class, CSWaitStrategy.class);
            
            return constructor.newInstance(driver, by, name, waitStrategy);
        } catch (Exception e) {
            reporter.error("Failed to create CSElement: " + name, e);
            throw new RuntimeException("Failed to create CSElement", e);
        }
    }
    
    /**
     * Navigate to a URL
     * 
     * @param url URL to navigate to
     */
    public void navigateTo(String url) {
        reporter.info("Navigating to: " + url);
        driver.get(url);
        waitStrategy.waitForPageLoad();
    }
    
    /**
     * Check if page is loaded
     * 
     * @return true if page is loaded
     */
    public boolean isPageLoaded() {
        // Check page readiness
        return waitStrategy.isPageLoaded();
    }
    
    /**
     * Get page title
     * 
     * @return Page title
     */
    public String getPageTitle() {
        return driver.getTitle();
    }
    
    /**
     * Get current URL
     * 
     * @return Current URL
     */
    public String getCurrentUrl() {
        return driver.getCurrentUrl();
    }
    
    /**
     * Refresh page
     */
    public void refreshPage() {
        reporter.info("Refreshing page: " + pageName);
        driver.navigate().refresh();
        waitStrategy.waitForPageLoad();
    }
    
    /**
     * Go back to previous page
     */
    public void goBack() {
        reporter.info("Navigating back from: " + pageName);
        driver.navigate().back();
        waitStrategy.waitForPageLoad();
    }
    
    /**
     * Go forward to next page
     */
    public void goForward() {
        reporter.info("Navigating forward from: " + pageName);
        driver.navigate().forward();
        waitStrategy.waitForPageLoad();
    }
    
    /**
     * Execute JavaScript
     * 
     * @param script JavaScript to execute
     * @param args Script arguments
     * @return Script result
     */
    protected Object executeJs(String script, Object... args) {
        try {
            reporter.debug("Executing JavaScript: " + script);
            return ((org.openqa.selenium.JavascriptExecutor) driver).executeScript(script, args);
        } catch (Exception e) {
            reporter.error("Failed to execute JavaScript", e);
            throw e;
        }
    }

        ERROR(3, "#dc3545"),
        PASS(4, "#198754"),
        FAIL(5, "#dc3545"),
        SKIP(6, "#6c757d");
        
        private final int level;
        private final String color;
        
        LogLevel(int level, String color) {
            this.level = level;
            this.color = color;
        }
        
        public int getLevel() {
            return level;
        }
        
        public String getColor() {
            return color;
        }
    }
    
    /**
     * Inner class to store test data
     */
    private static class TestData {
        private final String testName;
        private final String testDescription;
        private final long threadId;
        private final long startTime;
        private long endTime;
        private List<LogEntry> logs = Collections.synchronizedList(new ArrayList<>());
        private List<String> screenshots = Collections.synchronizedList(new ArrayList<>());
        private LogLevel status = LogLevel.INFO;
        
        public TestData(String testName, String testDescription, long threadId, long startTime) {
            this.testName = testName;
            this.testDescription = testDescription;
            this.threadId = threadId;
            this.startTime = startTime;
        }
    }
    
    /**
     * Inner class to store log entries
     */
    private static class LogEntry {
        private final long timestamp;
        private final LogLevel level;
        private final String message;
        private final String stackTrace;
        
        public LogEntry(long timestamp, LogLevel level, String message, String stackTrace) {
            this.timestamp = timestamp;
            this.level = level;
            this.message = message;
            this.stackTrace = stackTrace;
        }
    }
    
    /**
     * Private constructor to enforce singleton pattern
     */
    private CSReporter() {
        // Initialize status counts
        Arrays.stream(LogLevel.values())
              .forEach(level -> testStatusCounts.put(level.name(), new AtomicInteger(0)));
        
        // Get log level from configuration
        try {
            ConfigurationManager configManager = ConfigurationManager.getInstance();
            String configLogLevel = configManager.getLogLevel();
            
            if (configLogLevel != null && !configLogLevel.isEmpty()) {
                this.logLevel = LogLevel.valueOf(configLogLevel.toUpperCase());
            }
        } catch (Exception e) {
            System.err.println("Error getting log level from configuration: " + e.getMessage());
        }
        
        // Create report directory
        createReportDirectory();
    }
    
    /**
     * Get singleton instance
     * 
     * @return CSReporter instance
     */
    public static CSReporter getInstance() {
        if (instance == null) {
            synchronized (LOCK) {
                if (instance == null) {
                    instance = new CSReporter();
                }
            }
        }
        
        return instance;
    }
    
    /**
     * Create report directory
     */
    private void createReportDirectory() {
        try {
            // Create base report directory
            String timestamp = fileNameDateFormat.format(new Date());
            reportDir = "reports/cstestforge_" + timestamp;
            
            Path reportPath = Paths.get(reportDir);
            Files.createDirectories(reportPath);
            
            // Create screenshots directory
            Path screenshotsPath = Paths.get(reportDir, "screenshots");
            Files.createDirectories(screenshotsPath);
            
            // Copy assets
            copyReportAssets();
        } catch (Exception e) {
            System.err.println("Error creating report directory: " + e.getMessage());
        }
    }
    
    /**
     * Copy report assets (CSS, JS, etc.)
     */
    private void copyReportAssets() {
        try {
            // Create assets directory
            Path assetsPath = Paths.get(reportDir, "assets");
            Files.createDirectories(assetsPath);
            
            // Create CSS file
            String cssContent = generateCss();
            Files.write(Paths.get(assetsPath.toString(), "report.css"), cssContent.getBytes());
            
            // Create JS file
            String jsContent = generateJs();
            Files.write(Paths.get(assetsPath.toString(), "report.js"), jsContent.getBytes());
        } catch (Exception e) {
            System.err.println("Error copying report assets: " + e.getMessage());
        }
    }

    /**
     * Start a new test
     * 
     * @param testName Test name
     * @param testDescription Test description
     */
    public void startTest(String testName, String testDescription) {
        long threadId = Thread.currentThread().getId();
        TestData testData = new TestData(testName, testDescription, threadId, System.currentTimeMillis());
        testDataMap.put(threadId, testData);
        
        // Log test start
        log(LogLevel.INFO, "Test started: " + testName);
    }
    
    /**
     * End the current test
     */
    public void endTest() {
        long threadId = Thread.currentThread().getId();
        TestData testData = testDataMap.get(threadId);
        
        if (testData != null) {
            testData.endTime = System.currentTimeMillis();
            
            // Update test status count
            testStatusCounts.get(testData.status.name()).incrementAndGet();
            
            // Move to completed tests
            completedTests.add(testData);
            testDataMap.remove(threadId);
            
            // Log test end
            log(LogLevel.INFO, "Test ended: " + testData.testName + " - Duration: " + 
                   formatDuration(testData.endTime - testData.startTime) + 
                   " - Status: " + testData.status);
        }
    }
    
    /**
     * Log a debug message
     * 
     * @param message Message to log
     */
    public void debug(String message) {
        log(LogLevel.DEBUG, message);
    }
    
    /**
     * Log an info message
     * 
     * @param message Message to log
     */
    public void info(String message) {
        log(LogLevel.INFO, message);
    }
    
    /**
     * Log a warning message
     * 
     * @param message Message to log
     */
    public void warn(String message) {
        log(LogLevel.WARN, message);
    }
    
    /**
     * Log an error message
     * 
     * @param message Message to log
     * @param exception Exception (optional)
     */
    public void error(String message, Throwable exception) {
        String stackTrace = exception != null ? getStackTrace(exception) : null;
        log(LogLevel.ERROR, message, stackTrace);
    }
    
    /**
     * Log a pass message
     * 
     * @param message Message to log
     */
    public void pass(String message) {
        log(LogLevel.PASS, message);
        
        // Update test status if higher priority
        long threadId = Thread.currentThread().getId();
        TestData testData = testDataMap.get(threadId);
        
        if (testData != null && LogLevel.PASS.getLevel() > testData.status.getLevel()) {
            testData.status = LogLevel.PASS;
        }
    }
    
    /**
     * Log a fail message
     * 
     * @param message Message to log
     */
    public void fail(String message) {
        log(LogLevel.FAIL, message);
        
        // Update test status
        long threadId = Thread.currentThread().getId();
        TestData testData = testDataMap.get(threadId);
        
        if (testData != null) {
            testData.status = LogLevel.FAIL;
        }
    }
    
    /**
     * Log a skip message
     * 
     * @param message Message to log
     */
    public void skip(String message) {
        log(LogLevel.SKIP, message);
        
        // Update test status
        long threadId = Thread.currentThread().getId();
        TestData testData = testDataMap.get(threadId);
        
        if (testData != null) {
            testData.status = LogLevel.SKIP;
        }
    }
    
    /**
     * Log a message with specified level
     * 
     * @param level Log level
     * @param message Message to log
     */
    private void log(LogLevel level, String message) {
        log(level, message, null);
    }
    
    /**
     * Log a message with specified level and stack trace
     * 
     * @param level Log level
     * @param message Message to log
     * @param stackTrace Stack trace (optional)
     */
    private void log(LogLevel level, String message, String stackTrace) {
        // Only log if level is greater than or equal to configured level
        if (level.getLevel() >= this.logLevel.getLevel()) {
            System.out.println("[" + level + "] " + message);
            
            // Add to test logs
            long threadId = Thread.currentThread().getId();
            TestData testData = testDataMap.get(threadId);
            
            if (testData != null) {
                LogEntry logEntry = new LogEntry(System.currentTimeMillis(), level, message, stackTrace);
                testData.logs.add(logEntry);
            }
        }
    }
    
    /**
     * Get stack trace as string
     * 
     * @param throwable Throwable
     * @return Stack trace string
     */
    private String getStackTrace(Throwable throwable) {
        StringWriter sw = new StringWriter();
        PrintWriter pw = new PrintWriter(sw);
        throwable.printStackTrace(pw);
        return sw.toString();
    }
    
    /**
     * Attach a screenshot to the current test
     * 
     * @param screenshotPath Path to screenshot
     */
    public void attachScreenshot(String screenshotPath) {
        if (screenshotPath != null && !screenshotPath.isEmpty()) {
            long threadId = Thread.currentThread().getId();
            TestData testData = testDataMap.get(threadId);
            
            if (testData != null) {
                testData.screenshots.add(screenshotPath);
                log(LogLevel.INFO, "Screenshot captured: " + screenshotPath);
            }
        }
    }

            .append("        </div>\n")

            .append("            <button class=\"filter-btn active\" data-filter=\"all\">All Tests</button>\n")

            .append("            <button class=\"filter-btn\" data-filter=\"fail\">Failed</button>\n")
            .append("            <button class=\"filter-btn\" data-filter=\"skip\">Skipped</button>\n")

            .append("        <div class=\"test-list\">\n");

                .append("                <div class=\"test-header\" onclick=\"toggleTestLogs(this)\">\n")
                .append("                    <div class=\"test-name\">").append(test.testName).append("</div>\n")

                .append("            </div>\n");
        }
        
        html.append("        </div>\n")
            .append("    </section>\n");
        
        // Footer
        html.append("    <footer>\n")
            .append("        <div>Generated by CSTestForge on ").append(dateFormat.format(new Date())).append("</div>\n")
            .append("    </footer>\n");
        
        // JavaScript
        html.append("    <script src=\"assets/report.js\"></script>\n")
            .append("    <script>\n")
            .append("        // Initialize charts\n")
            .append("        const testData = {\n")
            .append("            passed: ").append(testStatusCounts.get(LogLevel.PASS.name()).get()).append(",\n")
            .append("            failed: ").append(testStatusCounts.get(LogLevel.FAIL.name()).get()).append(",\n")

            .append("        };\n")
            .append("        initializeCharts(testData);\n")
            .append("    </script>\n");
        
        // Close HTML
        html.append("</body>\n")
            .append("</html>");
        
        return html.toString();
    }

    /**
     * Format duration in milliseconds to human-readable string
     * 
     * @param durationMs Duration in milliseconds
     * @return Formatted duration string
     */
    private String formatDuration(long durationMs) {
        long seconds = durationMs / 1000;
        long minutes = seconds / 60;
        long hours = minutes / 60;
        
        seconds %= 60;
        minutes %= 60;
        
        StringBuilder formatted = new StringBuilder();
        
        if (hours > 0) {
            formatted.append(hours).append("h ");
        }
        
        if (minutes > 0 || hours > 0) {
            formatted.append(minutes).append("m ");
        }
        
        formatted.append(seconds).append("s");
        
        return formatted.toString();
    }
    
    /**
     * Escape HTML special characters
     * 
     * @param input Input string
     * @return Escaped string
     */
    private String escapeHtml(String input) {
        if (input == null) {
            return "";
        }
        
        return input.replace("&", "&amp;")
                   .replace("<", "&lt;")
                   .replace(">", "&gt;")
                   .replace("\"", "&quot;")
                   .replace("'", "&#39;");
    }

    /**

     *

     * @param locator Element locator
     * @param name Element name

     */
    public CSElement(WebDriver driver, By locator, String name, CSWaitStrategy waitStrategy) {
        this.driver = driver;
        this.locator = locator;
        this.name = name;
        this.waitStrategy = waitStrategy;
        this.reporter = CSReporter.getInstance();
    }
    
    /**
     * Get the underlying WebElement
     * 
     * @return WebElement
     */
    public WebElement getElement() {
        try {
            return waitStrategy.waitForElement(locator);
        } catch (Exception e) {
            reporter.error("Failed to find element: " + name + " (" + locator + ")", e);
            throw e;
        }
    }

            // Prepend the element as the first argument
            Object[] newArgs = new Object[args.length + 1];

    /**
     * Wait for page to load completely with custom timeout
     * Uses document.readyState to determine if page is loaded
     * 
     * @param timeoutSeconds Timeout in seconds
     * @return true if page is loaded
     */
    public boolean waitForPageLoad(int timeoutSeconds) {
        try {
            reporter.debug("Waiting for page to load");
            
            Wait<WebDriver> wait = new FluentWait<>(driver)
                    .withTimeout(Duration.ofSeconds(timeoutSeconds))
                    .pollingEvery(Duration.ofMillis(pollingIntervalMs))
                    .ignoring(Exception.class);
            
            // Wait for document ready state
            return wait.until(driver -> {
                try {
                    return ((JavascriptExecutor) driver)
                            .executeScript("return document.readyState")
                            .equals("complete");
                } catch (Exception e) {
                    return false;
                }
            });
        } catch (TimeoutException e) {
            reporter.error("Timeout waiting for page to load", e);
            throw e;
        }
    }

    /**
     * Wait for jQuery to be loaded and idle with custom timeout
     * 
     * @param timeoutSeconds Timeout in seconds
     * @return true if jQuery is loaded and idle
     */
    public boolean waitForJQueryLoad(int timeoutSeconds) {
        try {
            reporter.debug("Waiting for jQuery to load and be idle");
            
            // First check if jQuery is available
            boolean jQueryDefined = (Boolean) ((JavascriptExecutor) driver)
                    .executeScript("return typeof jQuery != 'undefined'");
            
            if (!jQueryDefined) {
                reporter.debug("jQuery is not defined on this page");
                return true; // Not an error if jQuery is not used on the page
            }
            
            Wait<WebDriver> wait = new FluentWait<>(driver)
                    .withTimeout(Duration.ofSeconds(timeoutSeconds))
                    .pollingEvery(Duration.ofMillis(pollingIntervalMs))
                    .ignoring(Exception.class);
            
            // Wait for jQuery to be idle
            return wait.until(driver -> {
                try {
                    return (Boolean) ((JavascriptExecutor) driver)
                            .executeScript("return jQuery.active == 0");
                } catch (Exception e) {
                    return false;
                }
            });
        } catch (TimeoutException e) {
            reporter.error("Timeout waiting for jQuery to load and be idle", e);
            throw e;
        }

     * Wait for Angular to be loaded and idle with custom timeout
     * 
     * @param timeoutSeconds Timeout in seconds
     * @return true if Angular is loaded and idle
     */
    public boolean waitForAngularLoad(int timeoutSeconds) {
        try {
            reporter.debug("Waiting for Angular to load and be idle");
            
            // First check if Angular is available (works for both AngularJS and Angular)
            Boolean angularDefined = (Boolean) ((JavascriptExecutor) driver)
                    .executeScript("return window.angular !== undefined || !!window.getAllAngularTestabilities");
            
            if (!angularDefined) {
                reporter.debug("Angular is not defined on this page");
                return true; // Not an error if Angular is not used on the page
            }
            
            // Check for Angular (2+)
            Boolean isAngular2Plus = (Boolean) ((JavascriptExecutor) driver)
                    .executeScript("return !!window.getAllAngularTestabilities");
            
            Wait<WebDriver> wait = new FluentWait<>(driver)
                    .withTimeout(Duration.ofSeconds(timeoutSeconds))
                    .pollingEvery(Duration.ofMillis(pollingIntervalMs))
                    .ignoring(Exception.class);
            
            if (isAngular2Plus) {
                // For Angular 2+
                String angularStabilityScript = 
                        "var callback = arguments[arguments.length - 1];" +
                        "var testabilities = window.getAllAngularTestabilities();" +
                        "var count = testabilities.length;" +
                        "var stabilityCheck = function(testability) {" +
                        "  testability.whenStable(function() {" +
                        "    count--;" +
                        "    if (count === 0) {" +
                        "      callback(true);" +
                        "    }" +
                        "  });" +
                        "};" +
                        "if (count === 0) {" +
                        "  callback(true);" +
                        "} else {" +
                        "  testabilities.forEach(stabilityCheck);" +
                        "}";
                
                try {
                    ((JavascriptExecutor) driver).executeAsyncScript(angularStabilityScript);
                    return true;
                } catch (Exception e) {
                    reporter.error("Error waiting for Angular stability", e);
                    return false;
                }
            } else {
                // For AngularJS
                return wait.until(driver -> {
                    try {
                        return (Boolean) ((JavascriptExecutor) driver)
                                .executeScript("return window.angular.element(document).injector().get('$http').pendingRequests.length === 0");
                    } catch (Exception e) {
                        return false;
                    }
                });
            }
        } catch (TimeoutException e) {
            reporter.error("Timeout waiting for Angular to load and be idle", e);
            throw e;
        }
    }

package com.cstestforge.framework.core;

/**
 * Interface for test steps.
 * Provides a standard way to define test steps for better organization and reporting.
 */
@FunctionalInterface
public interface CSTestStep {
    
    /**
     * Execute the test step
     * 
     * @throws Exception if step execution fails
     */
    void execute() throws Exception;
    
    /**
     * Create a step with a descriptive name
     * 
     * @param name Step name
     * @param step Step implementation
     * @return Named test step
     */
    static CSTestStep named(String name, CSTestStep step) {
        return new NamedTestStep(name, step);
    }
    
    /**
     * Implementation of named test step
     */
    class NamedTestStep implements CSTestStep {
        private final String name;
        private final CSTestStep step;
        
        /**
         * Constructor
         * 
         * @param name Step name
         * @param step Step implementation
         */
        public NamedTestStep(String name, CSTestStep step) {
            this.name = name;
            this.step = step;
        }
        
        @Override
        public void execute() throws Exception {
            step.execute();
        }
        
        /**
         * Get step name
         * 
         * @return Step name
         */
        public String getName() {
            return name;
        }
        
        @Override
        public String toString() {
            return name;
        }
    }
    
    /**
     * Create a conditional step that executes if condition is true
     * 
     * @param condition Condition to check
     * @param step Step to execute if condition is true
     * @return Conditional test step
     */
    static CSTestStep ifThen(java.util.function.BooleanSupplier condition, CSTestStep step) {
        return () -> {
            if (condition.getAsBoolean()) {
                step.execute();
            }
        };
    }
    
    /**
     * Create a conditional step with else branch
     * 
     * @param condition Condition to check
     * @param thenStep Step to execute if condition is true
     * @param elseStep Step to execute if condition is false
     * @return Conditional test step with else branch
     */
    static CSTestStep ifThenElse(java.util.function.BooleanSupplier condition, CSTestStep thenStep, CSTestStep elseStep) {
        return () -> {
            if (condition.getAsBoolean()) {
                thenStep.execute();
            } else {
                elseStep.execute();
            }
        };
    }
    
    /**
     * Create a step that retries on failure
     * 
     * @param step Step to retry
     * @param maxAttempts Maximum number of attempts
     * @return Retry test step
     */
    static CSTestStep retry(CSTestStep step, int maxAttempts) {
        return () -> {
            Exception lastException = null;
            for (int attempt = 1; attempt <= maxAttempts; attempt++) {
                try {
                    step.execute();
                    return; // Success, exit
                } catch (Exception e) {
                    lastException = e;
                    if (attempt < maxAttempts) {
                        // Wait before retry
                        try {
                            Thread.sleep(1000); // 1 second between retries
                        } catch (InterruptedException ie) {
                            Thread.currentThread().interrupt();
                            throw ie;
                        }
                    }
                }
            }
            // All attempts failed
            if (lastException != null) {
                throw lastException;
            }
        };
    }
    
    /**
     * Create a step that executes multiple steps in sequence
     * 
     * @param steps Steps to execute
     * @return Sequence test step
     */
    static CSTestStep sequence(CSTestStep... steps) {
        return () -> {
            for (CSTestStep step : steps) {
                step.execute();
            }
        };
    }
    
    /**
     * Create a step with timeout
     * 
     * @param step Step to execute
     * @param timeoutMillis Timeout in milliseconds
     * @return Timeout test step
     */
    static CSTestStep withTimeout(CSTestStep step, long timeoutMillis) {
        return () -> {
            java.util.concurrent.ExecutorService executor = java.util.concurrent.Executors.newSingleThreadExecutor();
            java.util.concurrent.Future<?> future = executor.submit(() -> {
                try {
                    step.execute();
                } catch (Exception e) {
                    throw new RuntimeException(e);
                }
            });
            
            try {
                future.get(timeoutMillis, java.util.concurrent.TimeUnit.MILLISECONDS);
            } catch (java.util.concurrent.TimeoutException e) {
                future.cancel(true);
                throw new RuntimeException("Step timed out after " + timeoutMillis + "ms", e);
            } catch (Exception e) {
                throw new RuntimeException(e);
            } finally {
                executor.shutdownNow();
            }
        };
    }
    
    /**
     * Create a step that catches exceptions
     * 
     * @param step Step to execute
     * @return Exception-catching test step
     */
    static CSTestStep ignoreExceptions(CSTestStep step) {
        return () -> {
            try {
                step.execute();
            } catch (Exception e) {
                // Ignore exceptions
            }
        };
    }
}

/**
 * Configuration manager for CSTestForge framework.
 * Handles loading and accessing configuration from various sources.
 */
public class ConfigurationManager {

    private static final String DEFAULT_CONFIG_FILE = "cstestforge.properties";
    private static final String ENV_CONFIG_PATH = "CSTESTFORGE_CONFIG";
    
    private static ConfigurationManager instance;
    private static final Object LOCK = new Object();
    
    private final Map<String, String> configuration = new HashMap<>();
    private boolean isInitialized = false;
    
    /**
     * Default values for configuration
     */
    private static final Map<String, String> DEFAULT_VALUES = Map.of(
        "browser.type", "chrome",
        "browser.headless", "false",
        "report.path", "reports",
        "screenshot.path", "screenshots",
        "explicit.wait.timeout", "30",
        "implicit.wait.timeout", "0",
        "page.load.timeout", "60",
        "log.level", "INFO",
        "project.name", "CSTestForge",
        "data.directory", "test-data",
        "test.execution.parallel", "false",
        "test.execution.retry.count", "0",
        "test.execution.retry.delay", "1000",
        "element.highlight", "true"
    );
    
    /**
     * Private constructor to enforce singleton pattern
     */
    private ConfigurationManager() {
        // Initialize with default values
        configuration.putAll(DEFAULT_VALUES);
    }
    
    /**
     * Get singleton instance
     * 
     * @return ConfigurationManager instance
     */
    public static ConfigurationManager getInstance() {
        if (instance == null) {
            synchronized (LOCK) {
                if (instance == null) {
                    instance = new ConfigurationManager();
                }
            }
        }
        
        return instance;
    }
    
    /**
     * Load configuration from all sources
     */
    public void loadConfiguration() {
        if (isInitialized) {
            return;
        }
        
        // Load configuration in order of precedence
        
        // 1. Load from default configuration file
        loadFromFile(DEFAULT_CONFIG_FILE);
        
        // 2. Load from custom configuration file specified by environment variable
        String customConfigPath = System.getenv(ENV_CONFIG_PATH);
        if (customConfigPath != null && !customConfigPath.isEmpty()) {
            loadFromFile(customConfigPath);
        }
        
        // 3. Load from system properties (highest precedence)
        loadFromSystemProperties();
        
        isInitialized = true;
    }
    
    /**
     * Load configuration from file
     * 
     * @param filePath Path to configuration file
     */
    private void loadFromFile(String filePath) {
        Path path = Paths.get(filePath);
        
        if (!Files.exists(path)) {
            System.out.println("Configuration file not found: " + filePath);
            return;
        }
        
        Properties properties = new Properties();
        
        try (InputStream input = new FileInputStream(path.toFile())) {
            properties.load(input);
            
            for (String name : properties.stringPropertyNames()) {
                String value = properties.getProperty(name);
                configuration.put(name, value);
            }
            
            System.out.println("Loaded configuration from: " + filePath);
        } catch (IOException e) {
            System.err.println("Error loading configuration from file: " + filePath);
            e.printStackTrace();
        }
    }
    
    /**
     * Load configuration from system properties
     */
    private void loadFromSystemProperties() {
        for (String key : configuration.keySet()) {
            String systemValue = System.getProperty(key);
            if (systemValue != null && !systemValue.isEmpty()) {
                configuration.put(key, systemValue);
            }
        }
    }
    
    /**
     * Get configuration value
     * 
     * @param key Configuration key
     * @return Configuration value or default value if not found
     */
    public String getValue(String key) {
        if (!isInitialized) {
            loadConfiguration();
        }
        
        return configuration.getOrDefault(key, DEFAULT_VALUES.getOrDefault(key, null));
    }
    
    /**
     * Get configuration value with default
     * 
     * @param key Configuration key
     * @param defaultValue Default value if key not found
     * @return Configuration value or provided default value if not found
     */
    public String getValue(String key, String defaultValue) {
        if (!isInitialized) {
            loadConfiguration();
        }
        
        return configuration.getOrDefault(key, defaultValue);
    }
    
    /**
     * Set configuration value
     * 
     * @param key Configuration key
     * @param value Configuration value
     */
    public void setValue(String key, String value) {
        configuration.put(key, value);
    }
    
    /**
     * Get integer configuration value
     * 
     * @param key Configuration key
     * @param defaultValue Default value if key not found or not an integer
     * @return Configuration value as integer
     */
    public int getIntValue(String key, int defaultValue) {
        String value = getValue(key);
        
        if (value == null) {
            return defaultValue;
        }
        
        try {
            return Integer.parseInt(value);
        } catch (NumberFormatException e) {
            System.err.println("Invalid integer value for key: " + key + ", using default: " + defaultValue);
            return defaultValue;
        }
    }
    
    /**
     * Get boolean configuration value
     * 
     * @param key Configuration key
     * @param defaultValue Default value if key not found
     * @return Configuration value as boolean
     */
    public boolean getBooleanValue(String key, boolean defaultValue) {
        String value = getValue(key);
        
        if (value == null) {
            return defaultValue;
        }
        
        return Boolean.parseBoolean(value);
    }

     */
    public String getBrowserType() {
        return getValue("browser.type");
    }

import org.openqa.selenium.firefox.FirefoxOptions;
import org.openqa.selenium.safari.SafariDriver;
import org.openqa.selenium.safari.SafariOptions;

import java.time.Duration;
import java.util.HashMap;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

/**
 * Driver manager for WebDriver instances.
 * Manages creation, retrieval, and cleanup of WebDriver instances.
 */
public class DriverManager {

    private static final ThreadLocal<WebDriver> driverThreadLocal = new ThreadLocal<>();
    private static final Map<Long, WebDriver> allDrivers = new ConcurrentHashMap<>();
    
    /**
     * Get WebDriver instance for current thread
     * 
     * @param browserType Browser type (chrome, firefox, edge, safari)
     * @return WebDriver instance
     */
    public static WebDriver getDriver(String browserType) {
        // Check if driver already exists for current thread
        WebDriver driver = driverThreadLocal.get();
        
        if (driver != null) {
            return driver;
        }
        
        // Create new driver
        driver = createDriver(browserType);
        
        // Configure driver
        configureDriver(driver);
        
        // Store driver
        driverThreadLocal.set(driver);
        allDrivers.put(Thread.currentThread().getId(), driver);
        
        return driver;
    }
    
    /**
     * Create new WebDriver instance
     * 
     * @param browserType Browser type (chrome, firefox, edge, safari)
     * @return WebDriver instance
     */
    private static WebDriver createDriver(String browserType) {
        ConfigurationManager configManager = ConfigurationManager.getInstance();
        boolean headless = configManager.isHeadless();
        
        switch (browserType.toLowerCase()) {
            case "chrome":
                return createChromeDriver(headless);
            case "firefox":
                return createFirefoxDriver(headless);
            case "edge":
                return createEdgeDriver(headless);
            case "safari":
                return createSafariDriver();
            default:
                System.out.println("Unsupported browser type: " + browserType + ", defaulting to Chrome");
                return createChromeDriver(headless);
        }
    }
    
    /**
     * Create Chrome WebDriver instance
     * 
     * @param headless true for headless mode
     * @return ChromeDriver instance
     */
    private static WebDriver createChromeDriver(boolean headless) {
        ChromeOptions options = new ChromeOptions();
        
        // Common options
        options.addArguments("--start-maximized");
        options.addArguments("--disable-notifications");
        options.addArguments("--disable-popup-blocking");
        options.addArguments("--disable-infobars");
        options.addArguments("--disable-dev-shm-usage");
        options.addArguments("--no-sandbox");
        
        // Experimental options
        Map<String, Object> prefs = new HashMap<>();
        prefs.put("credentials_enable_service", false);
        prefs.put("profile.password_manager_enabled", false);
        options.setExperimentalOption("prefs", prefs);
        options.setExperimentalOption("excludeSwitches", new String[]{"enable-automation"});
        
        // Headless mode
        if (headless) {
            options.addArguments("--headless=new");
            options.addArguments("--window-size=1920,1080");
        }
        
        return new ChromeDriver(options);
    }
    
    /**
     * Create Firefox WebDriver instance
     * 
     * @param headless true for headless mode
     * @return FirefoxDriver instance
     */
    private static WebDriver createFirefoxDriver(boolean headless) {
        FirefoxOptions options = new FirefoxOptions();
        
        // Common options
        options.addArguments("-width=1920");
        options.addArguments("-height=1080");
        
        // Headless mode
        if (headless) {
            options.addArguments("--headless");
        }
        
        return new FirefoxDriver(options);
    }
    
    /**
     * Create Edge WebDriver instance
     * 
     * @param headless true for headless mode
     * @return EdgeDriver instance
     */
    private static WebDriver createEdgeDriver(boolean headless) {
        EdgeOptions options = new EdgeOptions();
        
        // Common options
        options.addArguments("--start-maximized");
        options.addArguments("--disable-notifications");
        options.addArguments("--disable-popup-blocking");
        options.addArguments("--disable-infobars");
        
        // Experimental options
        Map<String, Object> prefs = new HashMap<>();
        prefs.put("credentials_enable_service", false);
        prefs.put("profile.password_manager_enabled", false);
        options.setExperimentalOption("prefs", prefs);
        options.setExperimentalOption("excludeSwitches", new String[]{"enable-automation"});
        
        // Headless mode
        if (headless) {
            options.addArguments("--headless=new");
            options.addArguments("--window-size=1920,1080");
        }
        
        return new EdgeDriver(options);
    }
    
    /**
     * Create Safari WebDriver instance
     * 
     * @return SafariDriver instance
     */
    private static WebDriver createSafariDriver() {
        SafariOptions options = new SafariOptions();
        
        // Safari doesn't support headless mode
        // Configure other options
        options.setAutomaticInspection(false);
        
        return new SafariDriver(options);
    }
    
    /**
     * Configure WebDriver instance
     * 
     * @param driver WebDriver instance
     */
    private static void configureDriver(WebDriver driver) {
        ConfigurationManager configManager = ConfigurationManager.getInstance();
        
        // Set timeouts
        driver.manage().timeouts().implicitlyWait(Duration.ofSeconds(configManager.getImplicitWaitTimeout()));
        driver.manage().timeouts().pageLoadTimeout(Duration.ofSeconds(configManager.getPageLoadTimeout()));
        driver.manage().timeouts().scriptTimeout(Duration.ofSeconds(configManager.getPageLoadTimeout()));
        
        // Maximize window
        driver.manage().window().maximize();
    }
    
    /**
     * Quit WebDriver instance for current thread
     */
    public static void quitDriver() {
        WebDriver driver = driverThreadLocal.get();
        
        if (driver != null) {
            try {
                driver.quit();
            } catch (Exception e) {
                System.err.println("Error quitting WebDriver: " + e.getMessage());
            } finally {
                driverThreadLocal.remove();
                allDrivers.remove(Thread.currentThread().getId());
            }
        }
    }
    
    /**
     * Quit all WebDriver instances
     */
    public static void quitAllDrivers() {
        for (Map.Entry<Long, WebDriver> entry : allDrivers.entrySet()) {
            try {
                WebDriver driver = entry.getValue();
                if (driver != null) {
                    driver.quit();
                }
            } catch (Exception e) {
                System.err.println("Error quitting WebDriver for thread " + entry.getKey() + ": " + e.getMessage());
            }
        }
        
        allDrivers.clear();
        driverThreadLocal.remove();
    }
    
    /**
     * Check if WebDriver instance exists for current thread
     * 
     * @return true if WebDriver instance exists
     */
    public static boolean hasDriver() {
        return driverThreadLocal.get() != null;
    }
    
    /**
     * Get current browser name
     * 
     * @return Current browser name
     */
    public static String getCurrentBrowserName() {
        WebDriver driver = driverThreadLocal.get();
        
        if (driver == null) {
            return "unknown";
        }
        
        String browserName = driver.getClass().getSimpleName();
        return browserName.replace("Driver", "").toLowerCase();
    }
}


package com.cstestforge.framework.utils;

import com.cstestforge.framework.config.ConfigurationManager;
import org.openqa.selenium.OutputType;
import org.openqa.selenium.TakesScreenshot;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;

import java.io.File;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.nio.file.StandardCopyOption;
import java.text.SimpleDateFormat;
import java.util.Date;

/**
 * Manager for taking and saving screenshots.
 */
public class CSScreenshotManager {

    private final WebDriver driver;
    private final String screenshotDir;
    private final ConfigurationManager configManager;
    
    /**
     * Constructor
     * 
     * @param driver WebDriver instance
     */
    public CSScreenshotManager(WebDriver driver) {
        this.driver = driver;
        this.configManager = ConfigurationManager.getInstance();
        
        // Get screenshot directory from configuration
        String baseDir = configManager.getScreenshotPath();
        
        // Create timestamp subdirectory
        SimpleDateFormat dateFormat = new SimpleDateFormat("yyyyMMdd_HHmmss");
        String timestamp = dateFormat.format(new Date());
        
        this.screenshotDir = baseDir + File.separator + timestamp;
        
        // Create directory
        createScreenshotDirectory();
    }
    
    /**
     * Create screenshot directory
     */
    private void createScreenshotDirectory() {
        try {
            Files.createDirectories(Paths.get(screenshotDir));
        } catch (IOException e) {
            System.err.println("Error creating screenshot directory: " + e.getMessage());
        }
    }
    
    /**
     * Capture screenshot of current page
     * 
     * @param name Screenshot name
     * @return Path to saved screenshot
     */
    public String captureScreenshot(String name) {
        if (driver == null) {
            System.err.println("WebDriver is null, cannot capture screenshot");
            return null;
        }
        
        if (!(driver instanceof TakesScreenshot)) {
            System.err.println("WebDriver does not support taking screenshots");
            return null;
        }
        
        try {
            // Take screenshot
            TakesScreenshot screenshotDriver = (TakesScreenshot) driver;
            File screenshot = screenshotDriver.getScreenshotAs(OutputType.FILE);
            
            // Generate filename
            SimpleDateFormat dateFormat = new SimpleDateFormat("yyyyMMdd_HHmmss_SSS");
            String timestamp = dateFormat.format(new Date());
            String fileName = sanitizeFileName(name) + "_" + timestamp + ".png";
            String filePath = screenshotDir + File.separator + fileName;
            
            // Save screenshot
            Files.copy(screenshot.toPath(), Paths.get(filePath), StandardCopyOption.REPLACE_EXISTING);
            
            return filePath;
        } catch (Exception e) {
            System.err.println("Error capturing screenshot: " + e.getMessage());
            return null;
        }
    }
    
    /**
     * Capture screenshot of specific element
     * 
     * @param element WebElement to capture
     * @param name Screenshot name
     * @return Path to saved screenshot
     */
    public String captureElementScreenshot(WebElement element, String name) {
        if (element == null) {
            System.err.println("WebElement is null, cannot capture screenshot");
            return null;
        }
        
        try {
            // Check if element supports taking screenshots
            if (element instanceof TakesScreenshot) {
                TakesScreenshot screenshotElement = (TakesScreenshot) element;
                File screenshot = screenshotElement.getScreenshotAs(OutputType.FILE);
                
                // Generate filename
                SimpleDateFormat dateFormat = new SimpleDateFormat("yyyyMMdd_HHmmss_SSS");
                String timestamp = dateFormat.format(new Date());
                String fileName = "element_" + sanitizeFileName(name) + "_" + timestamp + ".png";
                String filePath = screenshotDir + File.separator + fileName;
                
                // Save screenshot
                Files.copy(screenshot.toPath(), Paths.get(filePath), StandardCopyOption.REPLACE_EXISTING);
                
                return filePath;
            } else {
                // Fall back to full page screenshot with element highlighted
                highlightElement(element);
                String filePath = captureScreenshot("highlighted_" + name);
                unhighlightElement(element);
                
                return filePath;
            }
        } catch (Exception e) {
            System.err.println("Error capturing element screenshot: " + e.getMessage());
            return null;
        }
    }
    
    /**
     * Highlight an element on the page
     * 
     * @param element WebElement to highlight
     */
    private void highlightElement(WebElement element) {
        if (driver != null && element != null && configManager.isElementHighlightEnabled()) {
            try {
                org.openqa.selenium.JavascriptExecutor js = (org.openqa.selenium.JavascriptExecutor) driver;
                js.executeScript(
                    "arguments[0].setAttribute('style', 'border: 2px solid red; background-color: yellow;');", 
                    element
                );
            } catch (Exception e) {
                // Ignore highlight errors
            }
        }
    }
    
    /**
     * Remove highlight from element
     * 
     * @param element WebElement to unhighlight
     */
    private void unhighlightElement(WebElement element) {
        if (driver != null && element != null) {
            try {
                org.openqa.selenium.JavascriptExecutor js = (org.openqa.selenium.JavascriptExecutor) driver;
                js.executeScript(
                    "arguments[0].setAttribute('style', '');", 
                    element
                );
            } catch (Exception e) {
                // Ignore unhighlight errors
            }
        }
    }
    
    /**
     * Sanitize filename to remove invalid characters
     * 
     * @param name Original filename
     * @return Sanitized filename
     */
    private String sanitizeFileName(String name) {
        if (name == null || name.isEmpty()) {
            return "screenshot";
        }
        
        // Remove invalid file characters
        return name.replaceAll("[^a-zA-Z0-9._-]", "_");
    }
    
    /**
     * Get screenshot as Base64 encoded string
     * 
     * @return Base64 encoded screenshot
     */
    public String getScreenshotAsBase64() {
        if (driver == null || !(driver instanceof TakesScreenshot)) {
            return null;
        }
        
        try {
            TakesScreenshot screenshotDriver = (TakesScreenshot) driver;
            return screenshotDriver.getScreenshotAs(OutputType.BASE64);
        } catch (Exception e) {
            System.err.println("Error getting screenshot as Base64: " + e.getMessage());
            return null;
        }
    }
    
    /**
     * Get screenshot as bytes
     * 
     * @return Screenshot bytes
     */
    public byte[] getScreenshotAsBytes() {
        if (driver == null || !(driver instanceof TakesScreenshot)) {
            return null;
        }
        
        try {
            TakesScreenshot screenshotDriver = (TakesScreenshot) driver;
            return screenshotDriver.getScreenshotAs(OutputType.BYTES);
        } catch (Exception e) {
            System.err.println("Error getting screenshot as bytes: " + e.getMessage());
            return null;
        }
    }
    
    /**
     * Capture screenshot with timestamp
     * 
     * @return Path to saved screenshot
     */
    public String captureTimestampedScreenshot() {
        SimpleDateFormat dateFormat = new SimpleDateFormat("yyyyMMdd_HHmmss");
        String timestamp = dateFormat.format(new Date());
        return captureScreenshot("screenshot_" + timestamp);
    }
    
    /**
     * Get screenshot directory
     * 
     * @return Screenshot directory
     */
    public String getScreenshotDirectory() {
        return screenshotDir;
    }
}

    /**
     * Save a generated test to the file system
     * 
     * @param test Generated test
     * @return true if successful
     */
    public boolean saveTestToFileSystem(GeneratedTest test) {
        try {
            // Create project name from test name
            String projectName = test.getName().replaceAll("[^a-zA-Z0-9-_]", "_");
            
            // Generate project key based on technology stack
            String projectKey = getProjectKey(test.getLanguage(), test.getFramework(), test.getTestType());
            
            // Get project structure for the technology stack
            ProjectStructure structure = PROJECT_STRUCTURES.get(projectKey);
            if (structure == null) {
                throw new IllegalArgumentException("Unsupported technology stack: " + projectKey);
            }
            
            // Create temp project directory
            String projectDir = TEMP_DIR + File.separator + PROJECT_BASE_DIR + File.separator + projectName;
            createDirectory(projectDir);
            
            // Create project directories
            for (String dir : structure.getDirectories()) {
                createDirectory(projectDir + File.separator + dir);
            }
            
            // Copy framework files
            for (Map.Entry<String, String> entry : structure.getFrameworkFiles().entrySet()) {
                String destPath = projectDir + File.separator + entry.getKey();
                String sourcePath = entry.getValue();
                
                // Ensure parent directory exists
                File destFile = new File(destPath);
                destFile.getParentFile().mkdirs();
                
                // Copy file from classpath resource
                copyResource(sourcePath, destPath);
            }
            
            // Save test code based on test type
            String testFilePath = getTestFilePath(projectDir, test, structure);
            saveToFile(testFilePath, test.getCode());
            
            // Update file path in test object
            test.setFilePath(testFilePath);
            
            return true;
        } catch (Exception e) {
            e.printStackTrace();
            return false;
        }
    }
    
    /**
     * Get project key based on technology stack
     * 
     * @param language Programming language
     * @param framework Testing framework
     * @param testType Test type
     * @return Project key
     */
    private String getProjectKey(Object language, Object framework, Object testType) {
        return language + "_" + framework + "_" + testType;
    }
    
    /**
     * Get file path for saving the test
     * 
     * @param projectDir Project directory
     * @param test Generated test
     * @param structure Project structure
     * @return Test file path
     */
    private String getTestFilePath(String projectDir, GeneratedTest test, ProjectStructure structure) {
        // Package directories
        String packageDirs = "";
        if (structure.getBasePackage() != null && !structure.getBasePackage().isEmpty()) {
            packageDirs = structure.getBasePackage().replace('.', File.separatorChar) + File.separator;
        }
        
        // Class name
        String className = test.getClassName();
        if (className == null || className.isEmpty()) {
            // Generate class name from test name
            className = test.getName().replaceAll("[^a-zA-Z0-9]", "");
            // Ensure first letter is capitalized
            className = className.substring(0, 1).toUpperCase() + className.substring(1);
            
            // Add Test or Tests suffix if not present
            if (!className.endsWith("Test") && !className.endsWith("Tests")) {
                className += "Test";
            }
        }
        
        // File path based on test type and language
        String filePath;
        if (test.getLanguage().equals(ProgrammingLanguage.JAVA)) {
            if (test.getTestType().equals(TestType.BDD)) {
                // BDD Feature file
                String featureName = className.replace("Test", "").replace("Tests", "");
                filePath = projectDir + File.separator + "src/test/resources/features" + File.separator + featureName + ".feature";
            } else {
                // TestNG test class
                filePath = projectDir + File.separator + "src/test/java" + File.separator + packageDirs + "tests" + File.separator + className + ".java";
            }
        } else {
            // TypeScript
            if (test.getTestType().equals(TestType.BDD)) {
                // BDD Feature file
                String featureName = className.replace("Test", "").replace("Tests", "");
                filePath = projectDir + File.separator + "features" + File.separator + featureName + ".feature";
            } else {
                // TestNG-style test
                filePath = projectDir + File.separator + "tests" + File.separator + className + ".spec.ts";
            }
        }
        
        return filePath;
    }
    
    /**
     * Create directory
     * 
     * @param dirPath Directory path
     */
    private void createDirectory(String dirPath) {
        File dir = new File(dirPath);
        if (!dir.exists()) {
            dir.mkdirs();
        }
    }
    
    /**
     * Save content to file
     * 
     * @param filePath File path
     * @param content File content
     * @throws IOException if file cannot be written
     */
    private void saveToFile(String filePath, String content) throws IOException {
        // Ensure parent directory exists
        File file = new File(filePath);
        file.getParentFile().mkdirs();
        
        // Write content to file
        try (BufferedWriter writer = new BufferedWriter(new FileWriter(file, StandardCharsets.UTF_8))) {
            writer.write(content);
        }
    }
    
    /**
     * Copy resource from classpath to file
     * 
     * @param resourcePath Resource path
     * @param destPath Destination path
     * @throws IOException if resource cannot be copied
     */
    private void copyResource(String resourcePath, String destPath) throws IOException {
        // Get resource from classpath
        ClassLoader classLoader = getClass().getClassLoader();
        try (var inputStream = classLoader.getResourceAsStream(resourcePath)) {
            if (inputStream == null) {
                throw new IOException("Resource not found: " + resourcePath);
            }
            
            // Ensure parent directory exists
            File destFile = new File(destPath);
            destFile.getParentFile().mkdirs();
            
            // Copy resource to file
            Files.copy(inputStream, Paths.get(destPath), StandardCopyOption.REPLACE_EXISTING);
        }
    }
    
    /**
     * Export project to zip file
     * 
     * @param projectDir Project directory
     * @param outputPath Output zip file path
     * @return true if successful
     */
    public boolean exportProjectToZip(String projectDir, String outputPath) {
        try {
            File projectDirFile = new File(projectDir);
            if (!projectDirFile.exists() || !projectDirFile.isDirectory()) {
                throw new IllegalArgumentException("Project directory does not exist: " + projectDir);
            }
            
            // Create output directory if it doesn't exist
            File outputFile = new File(outputPath);
            outputFile.getParentFile().mkdirs();
            
            // Create zip file
            try (ZipOutputStream zipOut = new ZipOutputStream(new FileOutputStream(outputPath))) {
                zipDirectory(projectDirFile, projectDirFile.getName(), zipOut);
            }
            
            return true;
        } catch (Exception e) {
            e.printStackTrace();
            return false;
        }
    }
    
    /**
     * Zip directory
     * 
     * @param folder Folder to zip
     * @param parentFolder Parent folder name
     * @param zipOut Zip output stream
     * @throws IOException if folder cannot be zipped
     */
    private void zipDirectory(File folder, String parentFolder, ZipOutputStream zipOut) throws IOException {
        File[] files = folder.listFiles();
        if (files == null) {
            return;
        }
        
        byte[] buffer = new byte[BUFFER_SIZE];
        
        for (File file : files) {
            if (file.isDirectory()) {
                zipDirectory(file, parentFolder + "/" + file.getName(), zipOut);
                continue;
            }
            
            try (FileInputStream fis = new FileInputStream(file)) {
                String entryPath = parentFolder + "/" + file.getName();
                ZipEntry zipEntry = new ZipEntry(entryPath);
                zipOut.putNextEntry(zipEntry);
                
                int length;
                while ((length = fis.read(buffer)) > 0) {
                    zipOut.write(buffer, 0, length);
                }
                
                zipOut.closeEntry();
            }
        }
    }
    
    /**
     * Import project from zip file
     * 
     * @param zipFilePath Zip file path
     * @param targetDir Target directory
     * @return Project directory path
     * @throws IOException if project cannot be imported
     */
    public String importProjectFromZip(String zipFilePath, String targetDir) throws IOException {
        File zipFile = new File(zipFilePath);
        if (!zipFile.exists() || !zipFile.isFile()) {
            throw new IllegalArgumentException("Zip file does not exist: " + zipFilePath);
        }
        
        // Create target directory if it doesn't exist
        File targetDirFile = new File(targetDir);
        if (!targetDirFile.exists()) {
            targetDirFile.mkdirs();
        }
        
        // Extract zip file
        try (ZipInputStream zipIn = new ZipInputStream(new FileInputStream(zipFile))) {
            ZipEntry entry = zipIn.getNextEntry();
            
            String projectName = null;
            
            // Find root directory (first entry)
            while (entry != null && projectName == null) {
                String entryName = entry.getName();
                if (entryName.indexOf('/') > 0) {
                    projectName = entryName.substring(0, entryName.indexOf('/'));
                }
                entry = zipIn.getNextEntry();
            }
            
            // Reset zip stream
            zipIn.close();
            
            // Extract files
            try (ZipInputStream zipIn2 = new ZipInputStream(new FileInputStream(zipFile))) {
                byte[] buffer = new byte[BUFFER_SIZE];
                
                ZipEntry entry2 = zipIn2.getNextEntry();
                while (entry2 != null) {
                    String filePath = targetDir + File.separator + entry2.getName();
                    
                    if (!entry2.isDirectory()) {
                        // Create parent directories
                        new File(filePath).getParentFile().mkdirs();
                        
                        // Extract file
                        try (FileOutputStream fos = new FileOutputStream(filePath)) {
                            int length;
                            while ((length = zipIn2.read(buffer)) > 0) {
                                fos.write(buffer, 0, length);
                            }
                        }
                    } else {
                        // Create directory
                        new File(filePath).mkdirs();
                    }
                    
                    zipIn2.closeEntry();
                    entry2 = zipIn2.getNextEntry();
                }
            }
            
            // Return project directory path
            return targetDir + File.separator + projectName;
        }
    }
    
    /**
     * Delete project directory
     * 
     * @param projectDir Project directory
     * @return true if successful
     */
    public boolean deleteProject(String projectDir) {
        try {
            File projectDirFile = new File(projectDir);
            if (!projectDirFile.exists()) {
                return true; // Already deleted
            }
            
            // Delete all files and subdirectories
            deleteDirectory(projectDirFile);
            
            return true;
        } catch (Exception e) {
            e.printStackTrace();
            return false;
        }
    }
    
    /**
     * Delete directory recursively
     * 
     * @param directory Directory to delete
     * @throws IOException if directory cannot be deleted
     */
    private void deleteDirectory(File directory) throws IOException {
        if (!directory.exists()) {
            return;
        }
        
        File[] files = directory.listFiles();
        if (files != null) {
            for (File file : files) {
                if (file.isDirectory()) {
                    deleteDirectory(file);
                } else {
                    file.delete();
                }
            }
        }
        
        directory.delete();
    }
    
    /**
     * Find test files in project
     * 
     * @param projectDir Project directory
     * @return List of test file paths
     */
    public List<String> findTestFiles(String projectDir) {
        List<String> testFiles = new ArrayList<>();
        File projectDirFile = new File(projectDir);
        
        if (!projectDirFile.exists() || !projectDirFile.isDirectory()) {
            return testFiles;
        }
        
        // Walk through project directory and find test files
        walk(projectDirFile, testFiles);
        
        return testFiles;
    }
    
    /**
     * Walk through directory and find test files
     * 
     * @param directory Directory to walk through
     * @param testFiles List to add test files to
     */
    private void walk(File directory, List<String> testFiles) {
        File[] files = directory.listFiles();
        if (files == null) {
            return;
        }
        
        for (File file : files) {
            if (file.isDirectory()) {
                walk(file, testFiles);
            } else {
                String fileName = file.getName();
                // Java test files
                if (fileName.endsWith("Test.java") || fileName.endsWith("Tests.java")) {
                    testFiles.add(file.getAbsolutePath());
                }
                // TypeScript test files
                else if (fileName.endsWith(".spec.ts") || fileName.endsWith(".test.ts")) {
                    testFiles.add(file.getAbsolutePath());
                }
                // Feature files
                else if (fileName.endsWith(".feature")) {
                    testFiles.add(file.getAbsolutePath());
                }
            }
        }
    }

    public String readFile(String filePath) throws IOException {

    }

package com.cstestforge.model.project;

import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

/**
 * Represents the structure of a test automation project.
 * Used when generating framework code and exporting projects.
 */
public class ProjectStructure {

    /**
     * Directories to create in the project
     */
    private List<String> directories = new ArrayList<>();
    
    /**
     * Base package for Java projects
     */
    private String basePackage;
    
    /**
     * Framework files to include in the project
     * Key: Destination path in project
     * Value: Source path in classpath resources
     */
    private Map<String, String> frameworkFiles = new HashMap<>();
    
    /**
     * Default constructor
     */
    public ProjectStructure() {
    }
    
    /**
     * Constructor with parameters
     * 
     * @param directories Directories to create
     * @param basePackage Base package
     * @param frameworkFiles Framework files
     */
    public ProjectStructure(List<String> directories, String basePackage, Map<String, String> frameworkFiles) {
        this.directories = directories;
        this.basePackage = basePackage;
        this.frameworkFiles = frameworkFiles;
    }
    
    /**
     * Get directories
     * 
     * @return Directories
     */
    public List<String> getDirectories() {
        return directories;
    }
    
    /**
     * Set directories
     * 
     * @param directories Directories
     */
    public void setDirectories(List<String> directories) {
        this.directories = directories;
    }
    
    /**
     * Get base package
     * 
     * @return Base package
     */
    public String getBasePackage() {
        return basePackage;
    }
    
    /**
     * Set base package
     * 
     * @param basePackage Base package
     */
    public void setBasePackage(String basePackage) {
        this.basePackage = basePackage;
    }
    
    /**
     * Get framework files
     * 
     * @return Framework files
     */
    public Map<String, String> getFrameworkFiles() {
        return frameworkFiles;
    }
    
    /**
     * Set framework files
     * 
     * @param frameworkFiles Framework files
     */
    public void setFrameworkFiles(Map<String, String> frameworkFiles) {
        this.frameworkFiles = frameworkFiles;
    }
    
    /**
     * Add directory
     * 
     * @param directory Directory
     */
    public void addDirectory(String directory) {
        this.directories.add(directory);
    }
    
    /**
     * Add framework file
     * 
     * @param destPath Destination path
     * @param sourcePath Source path
     */
    public void addFrameworkFile(String destPath, String sourcePath) {
        this.frameworkFiles.put(destPath, sourcePath);
    }
    
    /**
     * Create a copy of this structure
     * 
     * @return Copy of this structure
     */
    public ProjectStructure copy() {
        ProjectStructure copy = new ProjectStructure();
        copy.setDirectories(new ArrayList<>(this.directories));
        copy.setBasePackage(this.basePackage);
        copy.setFrameworkFiles(new HashMap<>(this.frameworkFiles));
        return copy;
    }
    
    /**
     * Merge with another structure
     * 
     * @param other Other structure
     * @return Merged structure
     */
    public ProjectStructure merge(ProjectStructure other) {
        ProjectStructure merged = this.copy();
        
        // Add directories from other structure
        for (String directory : other.getDirectories()) {
            if (!merged.getDirectories().contains(directory)) {
                merged.addDirectory(directory);
            }
        }
        
        // Add framework files from other structure
        merged.getFrameworkFiles().putAll(other.getFrameworkFiles());
        
        return merged;
    }
    
    /**
     * Check if structure has Java sources
     * 
     * @return true if structure has Java sources
     */
    public boolean hasJavaSources() {
        return basePackage != null && !basePackage.isEmpty();
    }
    
    /**
     * Check if structure has TypeScript sources
     * 
     * @return true if structure has TypeScript sources
     */
    public boolean hasTypeScriptSources() {
        return frameworkFiles.containsKey("tsconfig.json");
    }
    
    /**
     * Check if structure uses Maven
     * 
     * @return true if structure uses Maven
     */
    public boolean usesMaven() {
        return frameworkFiles.containsKey("pom.xml");
    }
    
    /**
     * Check if structure uses npm
     * 
     * @return true if structure uses npm
     */
    public boolean usesNpm() {
        return frameworkFiles.containsKey("package.json");
    }
    
    /**
     * Get source directories
     * 
     * @return Source directories
     */
    public List<String> getSourceDirectories() {
        List<String> sourceDirs = new ArrayList<>();
        
        for (String directory : directories) {
            if (directory.contains("src") || directory.equals("src")) {
                sourceDirs.add(directory);
            }
        }
        
        return sourceDirs;
    }
    
    /**
     * Get test directories
     * 
     * @return Test directories
     */
    public List<String> getTestDirectories() {
        List<String> testDirs = new ArrayList<>();
        
        for (String directory : directories) {
            if (directory.contains("test") || directory.equals("tests") || 
                directory.contains("features") || directory.contains("step-definitions")) {
                testDirs.add(directory);
            }
        }
        
        return testDirs;
    }
    
    /**
     * Check if structure uses TestNG
     * 
     * @return true if structure uses TestNG
     */
    public boolean usesTestNG() {
        return frameworkFiles.containsValue("framework/java/selenium/testng/testng.xml") ||
               frameworkFiles.containsValue("framework/java/playwright/testng/testng.xml") ||
               frameworkFiles.containsKey("playwright.config.ts");
    }
    
    /**
     * Check if structure uses BDD
     * 
     * @return true if structure uses BDD
     */
    public boolean usesBDD() {
        return frameworkFiles.containsValue("framework/java/selenium/bdd/cucumber.properties") ||
               frameworkFiles.containsValue("framework/java/playwright/bdd/cucumber.properties") ||
               frameworkFiles.containsKey("cucumber.js");
    }
}
