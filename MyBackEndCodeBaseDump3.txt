    /**
     * Check connection to Azure DevOps
     */
    public boolean checkConnection() {
        try {
            String organization = configManager.getConfigProperty("azure.organization", "");
            String project = configManager.getConfigProperty("azure.project", "");
            String token = configManager.getConfigProperty("azure.token", "");
            
            if (organization.isEmpty() || project.isEmpty() || token.isEmpty()) {
                return false;
            }
            
            String url = String.format("https://dev.azure.com/%s/%s/_apis/projects?%s", 
                organization, project, API_VERSION);
            
            HttpHeaders headers = createAuthHeaders(token);
            HttpEntity<String> entity = new HttpEntity<>(headers);
            
            ResponseEntity<String> response = restTemplate.exchange(
                url, HttpMethod.GET, entity, String.class);
            
            return response.getStatusCode() == HttpStatus.OK;
        } catch (Exception e) {
            return false;
        }
    }
    
    /**
     * Create a test plan in Azure DevOps
     */
    public AzureTestPlan createTestPlan(String name, String description) {
        try {
            String organization = configManager.getConfigProperty("azure.organization", "");
            String project = configManager.getConfigProperty("azure.project", "");
            String token = configManager.getConfigProperty("azure.token", "");
            
            if (organization.isEmpty() || project.isEmpty() || token.isEmpty()) {
                throw new RuntimeException("Azure DevOps configuration is missing");
            }
            
            String url = String.format("https://dev.azure.com/%s/%s/_apis/testplan/plans?%s", 
                organization, project, API_VERSION);
            
            HttpHeaders headers = createAuthHeaders(token);
            headers.setContentType(MediaType.APPLICATION_JSON);
            
            Map<String, Object> requestBody = new HashMap<>();
            requestBody.put("name", name);
            
            if (description != null && !description.isEmpty()) {
                requestBody.put("description", description);
            }
            
            String startDate = LocalDateTime.now().format(DateTimeFormatter.ISO_DATE);
            String endDate = LocalDateTime.now().plusMonths(3).format(DateTimeFormatter.ISO_DATE);
            
            requestBody.put("startDate", startDate);
            requestBody.put("endDate", endDate);
            requestBody.put("area", new HashMap<String, Object>());
            
            HttpEntity<Map<String, Object>> entity = new HttpEntity<>(requestBody, headers);
            
            ResponseEntity<String> response = restTemplate.exchange(
                url, HttpMethod.POST, entity, String.class);
            
            if (response.getStatusCode() == HttpStatus.OK || response.getStatusCode() == HttpStatus.CREATED) {
                JsonNode root = objectMapper.readTree(response.getBody());
                
                AzureTestPlan testPlan = new AzureTestPlan();
                testPlan.setId(root.path("id").asInt());
                testPlan.setName(root.path("name").asText());
                testPlan.setDescription(root.path("description").asText());
                testPlan.setUrl(root.path("url").asText());
                
                return testPlan;
            } else {
                throw new RuntimeException("Failed to create test plan: " + response.getStatusCode());
            }
        } catch (HttpClientErrorException e) {
            throw new RuntimeException("Failed to create test plan: " + e.getResponseBodyAsString(), e);
        } catch (Exception e) {
            throw new RuntimeException("Failed to create test plan: " + e.getMessage(), e);
        }
    }

    /**
     * Create a test case in Azure DevOps
     */
    public AzureTestCase createTestCase(int testSuiteId, int testPlanId, GeneratedTest generatedTest) {
        try {
            String organization = configManager.getConfigProperty("azure.organization", "");
            String project = configManager.getConfigProperty("azure.project", "");
            String token = configManager.getConfigProperty("azure.token", "");
            
            if (organization.isEmpty() || project.isEmpty() || token.isEmpty()) {
                throw new RuntimeException("Azure DevOps configuration is missing");
            }
            
            // First, create the work item (test case)
            String workItemUrl = String.format("https://dev.azure.com/%s/%s/_apis/wit/workitems/$Test%%20Case?%s", 
                organization, project, API_VERSION);
            
            HttpHeaders headers = createAuthHeaders(token);
            headers.setContentType(MediaType.valueOf("application/json-patch+json"));
            
            List<Map<String, Object>> operations = new ArrayList<>();
            
            // Add title
            operations.add(createJsonPatchOperation("add", "/fields/System.Title", generatedTest.getName()));
            
            // Add description
            String description = generatedTest.getDescription();
            if (description == null || description.isEmpty()) {
                description = "Automated test created by CSTestForge";
            }
            operations.add(createJsonPatchOperation("add", "/fields/System.Description", description));
            
            // Add additional details
            operations.add(createJsonPatchOperation("add", "/fields/Microsoft.VSTS.TCM.AutomationStatus", "Automated"));
            operations.add(createJsonPatchOperation("add", "/fields/Microsoft.VSTS.TCM.AutomatedTestName", 
                generatedTest.getPackageName() + "." + generatedTest.getClassName()));
            operations.add(createJsonPatchOperation("add", "/fields/Microsoft.VSTS.TCM.AutomatedTestType", "Unit Test"));
            
            HttpEntity<List<Map<String, Object>>> entity = new HttpEntity<>(operations, headers);
            
            ResponseEntity<String> response = restTemplate.exchange(
                workItemUrl, HttpMethod.POST, entity, String.class);
            
            if (response.getStatusCode() == HttpStatus.OK || response.getStatusCode() == HttpStatus.CREATED) {
                JsonNode root = objectMapper.readTree(response.getBody());
                int workItemId = root.path("id").asInt();
                
                // Now add test steps
                if (generatedTest.getSteps() != null && !generatedTest.getSteps().isEmpty()) {
                    addTestSteps(workItemId, generatedTest.getSteps());
                }
                
                // Now add the test case to the test suite
                String testCaseUrl = String.format("https://dev.azure.com/%s/%s/_apis/testplan/plans/%d/suites/%d/testcases/%d?%s", 
                    organization, project, testPlanId, testSuiteId, workItemId, API_VERSION);
                
                HttpHeaders suiteHeaders = createAuthHeaders(token);
                suiteHeaders.setContentType(MediaType.APPLICATION_JSON);
                
                HttpEntity<String> suiteEntity = new HttpEntity<>("{}", suiteHeaders);
                
                ResponseEntity<String> suiteResponse = restTemplate.exchange(
                    testCaseUrl, HttpMethod.POST, suiteEntity, String.class);
                
                if (suiteResponse.getStatusCode() == HttpStatus.OK || suiteResponse.getStatusCode() == HttpStatus.CREATED) {
                    JsonNode suiteRoot = objectMapper.readTree(suiteResponse.getBody());
                    
                    AzureTestCase testCase = new AzureTestCase();
                    testCase.setId(workItemId);
                    testCase.setName(generatedTest.getName());
                    testCase.setDescription(description);
                    testCase.setSuiteId(testSuiteId);
                    testCase.setPlanId(testPlanId);
                    
                    return testCase;
                } else {
                    throw new RuntimeException("Failed to add test case to suite: " + suiteResponse.getStatusCode());
                }
            } else {
                throw new RuntimeException("Failed to create test case: " + response.getStatusCode());
            }
        } catch (HttpClientErrorException e) {
            throw new RuntimeException("Failed to create test case: " + e.getResponseBodyAsString(), e);
        } catch (Exception e) {
            throw new RuntimeException("Failed to create test case: " + e.getMessage(), e);
        }
    }
    
    /**
     * Add test steps to a test case
     */
    private void addTestSteps(int testCaseId, List<TestStep> steps) {
        try {
            String organization = configManager.getConfigProperty("azure.organization", "");
            String project = configManager.getConfigProperty("azure.project", "");
            String token = configManager.getConfigProperty("azure.token", "");
            
            String url = String.format("https://dev.azure.com/%s/%s/_apis/wit/workitems/%d?%s", 
                organization, project, testCaseId, API_VERSION);
            
            HttpHeaders headers = createAuthHeaders(token);
            headers.setContentType(MediaType.valueOf("application/json-patch+json"));
            
            // Format test steps for Azure DevOps
            StringBuilder stepsHtml = new StringBuilder();
            stepsHtml.append("<steps id=\"0\">");
            
            for (int i = 0; i < steps.size(); i++) {
                TestStep step = steps.get(i);
                
                stepsHtml.append("<step id=\"").append(i + 1).append("\" type=\"ActionStep\">");
                stepsHtml.append("<parameterizedString isformatted=\"true\">");
                stepsHtml.append("<p>").append(escapeHtml(step.getAction())).append("</p>");
                stepsHtml.append("</parameterizedString>");
                stepsHtml.append("<parameterizedString isformatted=\"true\">");
                
                if (step.getExpectedResult() != null && !step.getExpectedResult().isEmpty()) {
                    stepsHtml.append("<p>").append(escapeHtml(step.getExpectedResult())).append("</p>");
                } else {
                    stepsHtml.append("<p>Action is performed successfully</p>");
                }
                
                stepsHtml.append("</parameterizedString>");
                stepsHtml.append("</step>");
            }
            
            stepsHtml.append("</steps>");
            
            List<Map<String, Object>> operations = new ArrayList<>();
            operations.add(createJsonPatchOperation("add", "/fields/Microsoft.VSTS.TCM.Steps", stepsHtml.toString()));
            
            HttpEntity<List<Map<String, Object>>> entity = new HttpEntity<>(operations, headers);
            
            ResponseEntity<String> response = restTemplate.exchange(
                url, HttpMethod.PATCH, entity, String.class);
            
            if (response.getStatusCode() != HttpStatus.OK) {
                throw new RuntimeException("Failed to add test steps: " + response.getStatusCode());
            }
        } catch (HttpClientErrorException e) {
            throw new RuntimeException("Failed to add test steps: " + e.getResponseBodyAsString(), e);
        } catch (Exception e) {
            throw new RuntimeException("Failed to add test steps: " + e.getMessage(), e);
        }
    }
    
    /**
     * Update test execution results in Azure DevOps
     */
    public void updateTestResults(AzureTestCase testCase, TestExecutionResult executionResult) {
        try {
            String organization = configManager.getConfigProperty("azure.organization", "");
            String project = configManager.getConfigProperty("azure.project", "");
            String token = configManager.getConfigProperty("azure.token", "");
            
            if (organization.isEmpty() || project.isEmpty() || token.isEmpty()) {
                throw new RuntimeException("Azure DevOps configuration is missing");
            }
            
            // Create a test run
            String runUrl = String.format("https://dev.azure.com/%s/%s/_apis/test/runs?%s", 
                organization, project, API_VERSION);
            
            HttpHeaders headers = createAuthHeaders(token);
            headers.setContentType(MediaType.APPLICATION_JSON);
            
            // Create test run data
            Map<String, Object> runData = new HashMap<>();
            runData.put("name", "CSTestForge Execution - " + testCase.getName());
            runData.put("plan", new HashMap<String, Object>() {{
                put("id", testCase.getPlanId());
            }});
            
            List<Map<String, Object>> testCases = new ArrayList<>();
            testCases.add(new HashMap<String, Object>() {{
                put("id", testCase.getId());
            }});
            
            runData.put("pointIds", new ArrayList<>());
            
            HttpEntity<Map<String, Object>> runEntity = new HttpEntity<>(runData, headers);
            
            ResponseEntity<String> runResponse = restTemplate.exchange(
                runUrl, HttpMethod.POST, runEntity, String.class);
            
            if (runResponse.getStatusCode() != HttpStatus.OK && runResponse.getStatusCode() != HttpStatus.CREATED) {
                throw new RuntimeException("Failed to create test run: " + runResponse.getStatusCode());
            }
            
            JsonNode runRoot = objectMapper.readTree(runResponse.getBody());
            int runId = runRoot.path("id").asInt();
            
            // Update test results
            String resultsUrl = String.format("https://dev.azure.com/%s/%s/_apis/test/runs/%d/results?%s", 
                organization, project, runId, API_VERSION);
            
            List<Map<String, Object>> results = new ArrayList<>();
            Map<String, Object> result = new HashMap<>();
            
            result.put("testCaseTitle", testCase.getName());
            result.put("testCase", new HashMap<String, Object>() {{
                put("id", testCase.getId());
            }});
            
            // Map the test status
            String outcome;
            if ("PASSED".equals(executionResult.getStatus())) {
                outcome = "Passed";
            } else if ("FAILED".equals(executionResult.getStatus())) {
                outcome = "Failed";
            } else {
                outcome = "NotExecuted";
            }
            
            result.put("outcome", outcome);
            result.put("state", "Completed");
            
            if (executionResult.getErrorMessage() != null && !executionResult.getErrorMessage().isEmpty()) {
                result.put("errorMessage", executionResult.getErrorMessage());
            }
            
            // Add stack trace if available
            if (executionResult.getStackTrace() != null && !executionResult.getStackTrace().isEmpty()) {
                result.put("stackTrace", executionResult.getStackTrace());
            }
            
            // Add execution time
            long duration = executionResult.getEndTime() - executionResult.getStartTime();
            result.put("durationInMs", duration);
            
            // Add start and complete time
            DateTimeFormatter formatter = DateTimeFormatter.ISO_DATE_TIME;
            LocalDateTime startTime = LocalDateTime.now().minusMillis(duration);
            LocalDateTime endTime = LocalDateTime.now();
            
            result.put("startedDate", startTime.format(formatter));
            result.put("completedDate", endTime.format(formatter));
            
            // Add any additional details
            if (!executionResult.getLogs().isEmpty()) {
                StringBuilder details = new StringBuilder();
                for (String log : executionResult.getLogs()) {
                    details.append(log).append("\n");
                }
                result.put("comment", details.toString());
            }
            
            results.add(result);
            
            HttpEntity<List<Map<String, Object>>> resultsEntity = new HttpEntity<>(results, headers);
            
            ResponseEntity<String> resultsResponse = restTemplate.exchange(
                resultsUrl, HttpMethod.POST, resultsEntity, String.class);
            
            if (resultsResponse.getStatusCode() != HttpStatus.OK && resultsResponse.getStatusCode() != HttpStatus.CREATED) {
                throw new RuntimeException("Failed to update test results: " + resultsResponse.getStatusCode());
            }
            
            // Complete the test run
            String completeUrl = String.format("https://dev.azure.com/%s/%s/_apis/test/runs/%d?%s", 
                organization, project, runId, API_VERSION);
            
            Map<String, Object> completeData = new HashMap<>();
            completeData.put("state", "Completed");
            
            HttpEntity<Map<String, Object>> completeEntity = new HttpEntity<>(completeData, headers);
            
            ResponseEntity<String> completeResponse = restTemplate.exchange(
                completeUrl, HttpMethod.PATCH, completeEntity, String.class);
            
            if (completeResponse.getStatusCode() != HttpStatus.OK) {
                throw new RuntimeException("Failed to complete test run: " + completeResponse.getStatusCode());
            }
        } catch (HttpClientErrorException e) {
            throw new RuntimeException("Failed to update test results: " + e.getResponseBodyAsString(), e);
        } catch (Exception e) {
            throw new RuntimeException("Failed to update test results: " + e.getMessage(), e);
        }
    }
    
    /**
     * Upload attachments to a test case or test result
     */
    public void uploadAttachments(int testCaseId, List<String> attachmentPaths) {
        try {
            String organization = configManager.getConfigProperty("azure.organization", "");
            String project = configManager.getConfigProperty("azure.project", "");
            String token = configManager.getConfigProperty("azure.token", "");
            
            if (attachmentPaths == null || attachmentPaths.isEmpty()) {
                return;
            }
            
            for (String attachmentPath : attachmentPaths) {
                try {
                    // Create a file name from the path
                    String fileName = attachmentPath.substring(attachmentPath.lastIndexOf('/') + 1);
                    
                    // Read the file as bytes
                    byte[] fileContent = java.nio.file.Files.readAllBytes(java.nio.file.Paths.get(attachmentPath));
                    
                    // Create attachment URL
                    String url = String.format("https://dev.azure.com/%s/%s/_apis/wit/attachments?fileName=%s&%s", 
                        organization, project, fileName, API_VERSION);
                    
                    HttpHeaders headers = createAuthHeaders(token);
                    headers.setContentType(MediaType.APPLICATION_OCTET_STREAM);
                    
                    HttpEntity<byte[]> entity = new HttpEntity<>(fileContent, headers);
                    
                    ResponseEntity<String> response = restTemplate.exchange(
                        url, HttpMethod.POST, entity, String.class);
                    
                    if (response.getStatusCode() == HttpStatus.OK || response.getStatusCode() == HttpStatus.CREATED) {
                        JsonNode root = objectMapper.readTree(response.getBody());
                        String attachmentUrl = root.path("url").asText();
                        
                        // Add attachment to work item
                        addAttachmentToWorkItem(testCaseId, attachmentUrl, fileName);
                    } else {
                        throw new RuntimeException("Failed to upload attachment: " + response.getStatusCode());
                    }
                } catch (Exception e) {
                    // Log error but continue with other attachments
                    System.err.println("Failed to upload attachment: " + attachmentPath + " - " + e.getMessage());
                }
            }
        } catch (Exception e) {
            throw new RuntimeException("Failed to upload attachments: " + e.getMessage(), e);
        }
    }

                JsonNode root = objectMapper.readTree(response.getBody());
                JsonNode values = root.path("value");
                
                List<AzureTestCase> testCases = new ArrayList<>();
                
                for (JsonNode testCase : values) {
                    int testCaseId = testCase.path("testCase").path("id").asInt();
                    
                    // Get full test case details
                    AzureTestCase azureTestCase = getTestCaseDetails(testCaseId);
                    azureTestCase.setSuiteId(testSuiteId);
                    azureTestCase.setPlanId(testPlanId);
                    
                    testCases.add(azureTestCase);
                }
                
                return testCases;
            } else {
                throw new RuntimeException("Failed to get test cases: " + response.getStatusCode());
            }
        } catch (HttpClientErrorException e) {
            throw new RuntimeException("Failed to get test cases: " + e.getResponseBodyAsString(), e);
        } catch (Exception e) {
            throw new RuntimeException("Failed to get test cases: " + e.getMessage(), e);
        }
    }
    
    /**
     * Get details of a specific test case
     */
    public AzureTestCase getTestCaseDetails(int testCaseId) {
        try {
            String organization = configManager.getConfigProperty("azure.organization", "");
            String project = configManager.getConfigProperty("azure.project", "");
            String token = configManager.getConfigProperty("azure.token", "");
            
            if (organization.isEmpty() || project.isEmpty() || token.isEmpty()) {
                throw new RuntimeException("Azure DevOps configuration is missing");
            }
            
            String url = String.format("https://dev.azure.com/%s/%s/_apis/wit/workitems/%d?%s", 
                organization, project, testCaseId, API_VERSION);
            
            HttpHeaders headers = createAuthHeaders(token);
            HttpEntity<String> entity = new HttpEntity<>(headers);
            
            ResponseEntity<String> response = restTemplate.exchange(
                url, HttpMethod.GET, entity, String.class);
            
            if (response.getStatusCode() == HttpStatus.OK) {
                JsonNode root = objectMapper.readTree(response.getBody());
                JsonNode fields = root.path("fields");
                
                AzureTestCase testCase = new AzureTestCase();
                testCase.setId(testCaseId);
                testCase.setName(fields.path("System.Title").asText());
                testCase.setDescription(fields.path("System.Description").asText());
                
                if (fields.has("Microsoft.VSTS.TCM.Steps")) {
                    // Parse test steps from HTML format
                    String stepsHtml = fields.path("Microsoft.VSTS.TCM.Steps").asText();
                    testCase.setSteps(parseTestStepsFromHtml(stepsHtml));
                }
                
                return testCase;
            } else {
                throw new RuntimeException("Failed to get test case details: " + response.getStatusCode());
            }
        } catch (HttpClientErrorException e) {
            throw new RuntimeException("Failed to get test case details: " + e.getResponseBodyAsString(), e);
        } catch (Exception e) {
            throw new RuntimeException("Failed to get test case details: " + e.getMessage(), e);
        }
    }
    
    /**
     * Parse test steps from HTML format
     */
    private List<AzureTestStep> parseTestStepsFromHtml(String stepsHtml) {
        List<AzureTestStep> steps = new ArrayList<>();
        
        try {
            // This is a simplified parser for the HTML-based test steps
            // In a production environment, use a proper HTML parser
            
            int stepIndex = 0;
            while (true) {
                int stepStartIndex = stepsHtml.indexOf("<step id=\"", stepIndex);
                if (stepStartIndex == -1) {
                    break;
                }
                
                int stepEndIndex = stepsHtml.indexOf("</step>", stepStartIndex);
                if (stepEndIndex == -1) {
                    break;
                }
                
                String stepHtml = stepsHtml.substring(stepStartIndex, stepEndIndex + 7);
                
                // Extract action
                int actionStartIndex = stepHtml.indexOf("<parameterizedString isformatted=\"true\">");
                if (actionStartIndex == -1) {
                    stepIndex = stepEndIndex + 7;
                    continue;
                }
                
                int actionEndIndex = stepHtml.indexOf("</parameterizedString>", actionStartIndex);
                if (actionEndIndex == -1) {
                    stepIndex = stepEndIndex + 7;
                    continue;
                }
                
                String actionHtml = stepHtml.substring(
                    actionStartIndex + "<parameterizedString isformatted=\"true\">".length(), 
                    actionEndIndex
                );
                
                // Extract action text from HTML (simple approach)
                String action = extractTextFromHtml(actionHtml);
                
                // Extract expected result
                String expectedResult = "";
                int expectedStartIndex = stepHtml.indexOf("<parameterizedString isformatted=\"true\">", actionEndIndex);
                if (expectedStartIndex != -1) {
                    int expectedEndIndex = stepHtml.indexOf("</parameterizedString>", expectedStartIndex);
                    if (expectedEndIndex != -1) {
                        String expectedHtml = stepHtml.substring(
                            expectedStartIndex + "<parameterizedString isformatted=\"true\">".length(), 
                            expectedEndIndex
                        );
                        expectedResult = extractTextFromHtml(expectedHtml);
                    }
                }
                
                // Create step
                AzureTestStep step = new AzureTestStep();
                step.setAction(action);
                step.setExpectedResult(expectedResult);
                
                steps.add(step);
                
                stepIndex = stepEndIndex + 7;
            }
        } catch (Exception e) {
            System.err.println("Error parsing test steps: " + e.getMessage());
        }
        
        return steps;
    }
    
    /**
     * Extract text from simple HTML
     */
    private String extractTextFromHtml(String html) {
        // Simple approach to extract text from basic HTML
        return html.replaceAll("<[^>]*>", "").trim();
    }
    
    /**
     * Create authorization headers with Basic Auth
     */
    private HttpHeaders createAuthHeaders(String token) {
        HttpHeaders headers = new HttpHeaders();
        
        String auth = ":" + token;
        byte[] encodedAuth = Base64.getEncoder().encode(auth.getBytes(StandardCharsets.UTF_8));
        String authHeader = "Basic " + new String(encodedAuth);
        
        headers.set("Authorization", authHeader);
        
        return headers;
    }
    
    /**
     * Create a JSON Patch operation
     */
    private Map<String, Object> createJsonPatchOperation(String op, String path, Object value) {
        Map<String, Object> operation = new HashMap<>();
        operation.put("op", op);
        operation.put("path", path);
        operation.put("value", value);
        
        return operation;
    }
    
    /**
     * Escape HTML for safe XML usage
     */
    private String escapeHtml(String text) {
        if (text == null) {
            return "";
        }
        
        return text.replace("&", "&amp;")
                  .replace("<", "&lt;")
                  .replace(">", "&gt;")
                  .replace("\"", "&quot;")
                  .replace("'", "&apos;");
    }
}

     * Start a new recording session
     */
    @PostMapping("/start")
    public ResponseEntity<ApiResponse<RecordingSession>> startRecording(@RequestBody RecordingRequest request) {
        try {
            // Validate request
            if (request.getUrl() == null || request.getUrl().isEmpty()) {
                return ResponseEntity.badRequest().body(
                    new ApiResponse<>(false, "URL is required", null));
            }
            
            // Generate a new recording ID
            String recordingId = "rec_" + UUID.randomUUID().toString().replace("-", "");
            
            // Create a new recording session
            RecordingSession session = new RecordingSession();
            session.setId(recordingId);
            session.setUrl(request.getUrl());
            session.setBrowser(request.getBrowser());
            session.setViewportWidth(request.getViewportWidth());
            session.setViewportHeight(request.getViewportHeight());
            session.setTool(request.getTool());
            session.setLanguage(request.getLanguage());
            session.setTestType(request.getTestType());
            
            // Set defaults if not provided
            if (session.getBrowser() == null || session.getBrowser().isEmpty()) {
                session.setBrowser("chrome");
            }
            
            if (session.getViewportWidth() <= 0) {
                session.setViewportWidth(1280);
            }
            
            if (session.getViewportHeight() <= 0) {
                session.setViewportHeight(720);
            }
            
            if (session.getTool() == null || session.getTool().isEmpty()) {
                session.setTool("selenium");
            }
            
            if (session.getLanguage() == null || session.getLanguage().isEmpty()) {
                session.setLanguage("java");
            }
            
            if (session.getTestType() == null || session.getTestType().isEmpty()) {
                session.setTestType("testng");
            }
            
            // Initialize the session
            session.setStartTime(System.currentTimeMillis());
            session.setStatus("initializing");
            
            // Add to active sessions
            activeSessions.put(recordingId, session);
            
            // Initialize the recorder service for this session
            recorderService.initializeSession(session);
            
            return ResponseEntity.ok(new ApiResponse<>(true, "Recording session started", session));
        } catch (Exception e) {
            return ResponseEntity.badRequest().body(
                new ApiResponse<>(false, "Failed to start recording: " + e.getMessage(), null));
        }

                return ResponseEntity.badRequest().body(
                    new ApiResponse<>(false, "Recording ID is required", null));

            RecordingSession session = activeSessions.get(recordingId);

                return ResponseEntity.notFound().build();

            // Stop the recording
            session.setStatus("stopping");
            
            // Generate the test code
            RecordingResult result = recorderService.finalizeSession(session);
            
            // Remove from active sessions
            activeSessions.remove(recordingId);
            
            return ResponseEntity.ok(new ApiResponse<>(true, "Recording stopped", result));
        } catch (Exception e) {
            return ResponseEntity.badRequest().body(
                new ApiResponse<>(false, "Failed to stop recording: " + e.getMessage(), null));
        }

                return ResponseEntity.badRequest().body(
                    new ApiResponse<>(false, "Recording session not found", null));
            }
            
            // Generate the code
            GeneratedTest generatedTest = codeGenerationService.generateTest(
                session, 
                request.getTestName(),
                request.getPackageName(),
                request.getClassName(),
                request.getDescription(),
                request.isGeneratePageObjects(),
                request.getCustomizations()
            );
            
            return ResponseEntity.ok(new ApiResponse<>(true, "Code generated", generatedTest));
        } catch (Exception e) {
            return ResponseEntity.badRequest().body(
                new ApiResponse<>(false, "Failed to generate code: " + e.getMessage(), null));
        }
    }
    
    /**
     * Execute a generated test
     */
    @PostMapping("/execute")
    public ResponseEntity<ApiResponse<TestExecutionResult>> executeTest(@RequestBody ExecuteTestRequest request) {
        try {
            // Validate request
            if (request.getGeneratedTest() == null) {
                return ResponseEntity.badRequest().body(
                    new ApiResponse<>(false, "Generated test is required", null));
            }
            
            // Execute the test
            TestExecutionResult result = testExecutionService.executeTest(
                request.getGeneratedTest(), 
                request.getProjectPath()
            );
            
            return ResponseEntity.ok(new ApiResponse<>(true, "Test executed", result));
        } catch (Exception e) {
            return ResponseEntity.badRequest().body(
                new ApiResponse<>(false, "Failed to execute test: " + e.getMessage(), null));
        }
    }
    
    /**
     * Stop test execution
     */
    @PostMapping("/execute/stop")
    public ResponseEntity<ApiResponse<Boolean>> stopExecution() {
        try {

        } catch (Exception e) {
            return ResponseEntity.badRequest().body(

    /**
     * Get recorder status
     */
    @GetMapping("/status")
    public ResponseEntity<ApiResponse<Map<String, Object>>> getRecorderStatus(
            @RequestParam("recordingId") String recordingId) {
        
        try {
            // Get the session
            RecordingSession session = activeSessions.get(recordingId);
            
            if (session == null) {
                return ResponseEntity.notFound().build();
            }
            
            // Prepare status information
            Map<String, Object> status = new HashMap<>();
            status.put("id", session.getId());
            status.put("url", session.getUrl());
            status.put("status", session.getStatus());
            status.put("eventCount", session.getEvents().size());
            status.put("startTime", session.getStartTime());
            status.put("lastEventTime", session.getLastEventTime());
            
            // Include screenshot if available
            if (session.getLastScreenshotPath() != null) {
                status.put("lastScreenshot", session.getLastScreenshotPath());
            }
            
            return ResponseEntity.ok(new ApiResponse<>(true, "Recorder status", status));
        } catch (Exception e) {
            return ResponseEntity.badRequest().body(
                new ApiResponse<>(false, "Failed to get recorder status: " + e.getMessage(), null));
        }

    /**
     * Get active recording sessions
     */
    @GetMapping("/sessions")
    public ResponseEntity<ApiResponse<List<Map<String, Object>>>> getActiveSessions() {
        try {
            // Prepare session information
            List<Map<String, Object>> sessions = activeSessions.values().stream()
                .map(session -> {
                    Map<String, Object> sessionInfo = new HashMap<>();
                    sessionInfo.put("id", session.getId());
                    sessionInfo.put("url", session.getUrl());
                    sessionInfo.put("status", session.getStatus());
                    sessionInfo.put("startTime", session.getStartTime());
                    sessionInfo.put("eventCount", session.getEvents().size());
                    return sessionInfo;
                })
                .collect(Collectors.toList());
            
            return ResponseEntity.ok(new ApiResponse<>(true, "Active sessions", sessions));
        } catch (Exception e) {
            return ResponseEntity.badRequest().body(
                new ApiResponse<>(false, "Failed to get active sessions: " + e.getMessage(), null));
        }
    }
    
    /**
     * Add a custom action to the recording
     */
    @PostMapping("/action")
    public ResponseEntity<ApiResponse<Boolean>> addCustomAction(@RequestBody CustomActionRequest request) {
        try {
            // Validate request
            if (request.getRecordingId() == null || request.getRecordingId().isEmpty()) {
                return ResponseEntity.badRequest().body(
                    new ApiResponse<>(false, "Recording ID is required", null));
            }
            
            if (request.getActionType() == null || request.getActionType().isEmpty()) {
                return ResponseEntity.badRequest().body(
                    new ApiResponse<>(false, "Action type is required", null));
            }
            
            // Get the session
            RecordingSession session = activeSessions.get(request.getRecordingId());
            
            if (session == null) {
                return ResponseEntity.notFound().build();
            }
            
            // Add the custom action
            recorderService.addCustomAction(session, request);
            
            return ResponseEntity.ok(new ApiResponse<>(true, "Custom action added", true));
        } catch (Exception e) {
            return ResponseEntity.badRequest().body(
                new ApiResponse<>(false, "Failed to add custom action: " + e.getMessage(), null));
        }
    }
    
    /**
     * Download recorder script
     */
    @GetMapping("/download/script")
    public ResponseEntity<Resource> downloadRecorderScript(@RequestParam("recordingId") String recordingId) {
        try {
            // Get the session
            RecordingSession session = activeSessions.get(recordingId);
            
            if (session == null) {
                return ResponseEntity.notFound().build();
            }
            
            // Get the injector script
            String script = recorderService.getInjectorScript(session, null);
            
            ByteArrayResource resource = new ByteArrayResource(script.getBytes(StandardCharsets.UTF_8));
            
            return ResponseEntity.ok()
                .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename=recorder-script.js")
                .contentType(MediaType.parseMediaType("application/javascript"))
                .body(resource);
        } catch (Exception e) {
            return ResponseEntity.badRequest().build();
        }
    }
    
    /**
     * Download generated code
     */
    @GetMapping("/download/code")
    public ResponseEntity<Resource> downloadGeneratedCode(@RequestParam("id") String testId) {
        try {
            // Get the generated test from the code generation service
            GeneratedTest generatedTest = codeGenerationService.getGeneratedTest(testId);
            
            if (generatedTest == null) {
                return ResponseEntity.notFound().build();
            }
            
            // Get the code
            String code = generatedTest.getCode();
            
            ByteArrayResource resource = new ByteArrayResource(code.getBytes(StandardCharsets.UTF_8));
            
            // Determine file extension based on language
            String fileExtension = "java";
            if ("typescript".equalsIgnoreCase(generatedTest.getLanguage())) {
                fileExtension = "ts";
            }
            
            return ResponseEntity.ok()
                .header(HttpHeaders.CONTENT_DISPOSITION, 
                    "attachment; filename=" + generatedTest.getClassName() + "." + fileExtension)
                .contentType(MediaType.parseMediaType("text/plain"))
                .body(resource);
        } catch (Exception e) {
            return ResponseEntity.badRequest().build();
        }
    }
}

    /**
     * Get the best locator for a specific tool
     */
    public String getBestLocator(String tool) {
        if (this.locators == null || this.locators.isEmpty()) {
            return null;
        }
        
        // For Selenium, prefer ID, name, CSS
        if ("selenium".equalsIgnoreCase(tool)) {
            if (this.locators.containsKey("id")) {
                return "By.id(\"" + this.locators.get("id") + "\")";
            } else if (this.locators.containsKey("name")) {
                return "By.name(\"" + this.locators.get("name") + "\")";
            } else if (this.locators.containsKey("cssSelector")) {
                return "By.cssSelector(\"" + this.locators.get("cssSelector") + "\")";
            } else if (this.locators.containsKey("xpath")) {
                return "By.xpath(\"" + this.locators.get("xpath") + "\")";
            }
        }
        
        // For Playwright, prefer CSS selectors, but also handle text selection
        if ("playwright".equalsIgnoreCase(tool)) {
            if (this.locators.containsKey("cssSelector")) {
                return "\"" + this.locators.get("cssSelector") + "\"";
            } else if (this.locators.containsKey("text") && !this.locators.get("text").isEmpty()) {
                return "page.getByText(\"" + this.locators.get("text") + "\")";
            } else if (this.locators.containsKey("xpath")) {
                return "page.locator(\"xpath=" + this.locators.get("xpath") + "\")";
            }
        }
        
        // Default case, return the first available locator
        for (Map.Entry<String, String> entry : this.locators.entrySet()) {
            return entry.getValue();
        }
        
        return null;
    }

            clone.setAdditionalParams(new HashMap<>(additionalParams));

        if (codeLinesSelenium != null) {
            clone.setCodeLinesSelenium(new ArrayList<>(codeLinesSelenium));
        }
        
        if (codeLinesPlaywright != null) {
            clone.setCodeLinesPlaywright(new ArrayList<>(codeLinesPlaywright));

            return "When user clicks on " + element;
        } else if ("input".equals(type) || "change".equals(type)) {
            return "When user enters \"" + value + "\" into " + element;
        } else if ("select".equals(type)) {
            return "When user selects \"" + value + "\" from " + element;
        } else if ("assertion".equals(type)) {
            return "Then user verifies that " + expectedResult;
        } else if ("wait".equals(type)) {
            return "And user waits for " + waitTimeMs + " milliseconds";
        } else {
            return "And user performs " + action;
        }
    }

        return (String) getExecutionOption("screenshotCapture", "onFailure");
    }

    private String screenshotPath;

    private String injectorScriptPath;

    private String cachedInjectorScript;
    
    // Keep track of active recording sessions
    private final Map<String, RecordingSession> activeSessions = new ConcurrentHashMap<>();
    
    /**
     * Initialize a recording session
     */
    public void initializeSession(RecordingSession session) {
        // Create a directory for screenshots if it doesn't exist
        try {
            Path sessionScreenshotPath = Paths.get(screenshotPath, session.getId());
            Files.createDirectories(sessionScreenshotPath);
        } catch (IOException e) {
            logger.error("Failed to create screenshot directory", e);
        }
        
        // Initialize AI services
        elementRecognitionService.initialize();
        smartLocatorService.initialize();
        
        // Add to active sessions
        activeSessions.put(session.getId(), session);
        
        logger.info("Recording session initialized: {}", session.getId());
    }
    
    /**
     * Finalize a recording session and generate test code
     */
    public RecordingResult finalizeSession(RecordingSession session) {
        logger.info("Finalizing recording session: {}", session.getId());
        
        // Create result object
        RecordingResult result = new RecordingResult();
        result.setRecordingId(session.getId());
        result.setUrl(session.getUrl());
        result.setBrowser(session.getBrowser());
        result.setStartTime(session.getStartTime());
        result.setEndTime(System.currentTimeMillis());
        result.setEventCount(session.getEvents().size());
        
        // Process events and generate steps
        processEvents(session);
        
        // Generate steps for the result
        List<String> steps = generateSteps(session);
        result.setSteps(steps);
        
        // Generate BDD steps if using BDD
        if (session.getTestType().toLowerCase().contains("bdd") || 
            session.getTestType().toLowerCase().contains("cucumber")) {
            List<String> bddSteps = generateBddSteps(session);
            result.setBddSteps(bddSteps);
        }
        
        // Get screenshots
        List<String> screenshots = findSessionScreenshots(session.getId());
        result.setScreenshotPaths(screenshots);
        
        // Generate test code
        try {
            GeneratedTest generatedTest = codeGenerationService.generateTest(
                session,
                "Recorded Test",
                "com.cstestforge.tests",
                "RecordedTest",
                "Test generated from recording session " + session.getId(),
                true,
                new HashMap<>()
            );
            
            result.setGeneratedTest(generatedTest);
        } catch (Exception e) {
            logger.error("Failed to generate test code", e);
            result.addMetadata("codeGenerationError", e.getMessage());
        }
        
        // Remove from active sessions
        activeSessions.remove(session.getId());
        
        return result;
    }
    
    /**
     * Process events from the recorder
     */
    public void processEvents(RecordingSession session, RecordingEventBatch eventBatch) {
        logger.debug("Processing {} events for session {}", 
            eventBatch.getEvents().size(), session.getId());
        
        // Register the frame if not already registered
        if (eventBatch.getFrameId() != null && !eventBatch.getFrameId().isEmpty() && 
            !session.getFrames().containsKey(eventBatch.getFrameId())) {
            
            RecordingSession.FrameInfo frameInfo = new RecordingSession.FrameInfo();
            frameInfo.setId(eventBatch.getFrameId());
            frameInfo.setUrl(eventBatch.getUrl());
            frameInfo.setTopFrame(false);
            
            session.addFrame(eventBatch.getFrameId(), frameInfo);
        }
        
        // Process the screenshot if provided
        if (eventBatch.getScreenshot() != null && !eventBatch.getScreenshot().isEmpty()) {
            String screenshotPath = saveScreenshot(session, eventBatch.getScreenshot());
            session.setLastScreenshotPath(screenshotPath);
        }
        
        // Process each event
        for (RecordingEvent event : eventBatch.getEvents()) {
            // Skip if recording is paused
            if ("paused".equals(session.getStatus())) {
                continue;
            }
            
            // Process the event
            processEvent(session, event);
            
            // Add the event to the session
            session.addEvent(event);
        }
        
        // Update session status
        if ("initializing".equals(session.getStatus())) {
            session.setStatus("recording");
        }
    }
    
    /**
     * Process a single event
     */
    private void processEvent(RecordingSession session, RecordingEvent event) {
        // Skip if recording is paused
        if ("paused".equals(session.getStatus())) {
            return;
        }
        
        // Set frame ID if not set
        if (event.getFrameId() == null || event.getFrameId().isEmpty()) {
            for (Map.Entry<String, RecordingSession.FrameInfo> entry : session.getFrames().entrySet()) {
                if (entry.getValue().isTopFrame()) {
                    event.setFrameId(entry.getKey());
                    break;
                }
            }
        }
        
        // Process element locators using AI
        if (event.getTarget() != null) {
            // Generate smart locators if not already generated
            if (event.getLocators() == null || event.getLocators().isEmpty()) {
                List<RecordingEvent.LocatorInfo> locators = smartLocatorService.generateLocators(
                    event.getTarget(), session);
                event.setLocators(locators);
            }
            
            // Enhance locators with AI
            List<RecordingEvent.LocatorInfo> enhancedLocators = 
                elementRecognitionService.enhanceLocators(event.getTarget(), event.getLocators(), session);
            
            if (enhancedLocators != null && !enhancedLocators.isEmpty()) {
                event.setLocators(enhancedLocators);
            }
        }
        
        // Log the event
        logger.debug("Processed event: {} for session {}", event.getType(), session.getId());
    }
    
    /**
     * Process all events in a session for test generation
     */
    public void processEvents(RecordingSession session) {
        // Skip if already processed
        if (session.getCustomData().containsKey("eventsProcessed")) {
            return;
        }
        
        // Process each event
        for (RecordingEvent event : session.getEvents()) {
            processEvent(session, event);
        }
        
        // Mark as processed
        session.addCustomData("eventsProcessed", true);
    }
    
    /**
     * Generate steps from a recording session
     */
    private List<String> generateSteps(RecordingSession session) {
        List<String> steps = new ArrayList<>();
        
        // Sort events by timestamp
        List<RecordingEvent> sortedEvents = new ArrayList<>(session.getEvents());
        sortedEvents.sort(Comparator.comparing(RecordingEvent::getTimestamp));
        
        for (RecordingEvent event : sortedEvents) {
            String step = generateStep(event);
            if (step != null && !step.isEmpty()) {
                steps.add(step);
            }
        }
        
        return steps;
    }
    
    /**
     * Generate a step description from an event
     */
    private String generateStep(RecordingEvent event) {
        String type = event.getType();
        
        if ("navigation".equals(type)) {
            if (event.getPageState() != null) {
                return "Navigate to " + event.getPageState().getUrl();
            }
            return "Navigate to page";
        } else if ("click".equals(type)) {
            if (event.getTarget() != null) {
                return "Click on " + getElementDescription(event.getTarget());
            }
            return "Click event";
        } else if ("input".equals(type) || "change".equals(type)) {
            if (event.getTarget() != null) {
                String value = event.getTargetValue();
                if (value != null) {
                    return "Enter \"" + value + "\" into " + getElementDescription(event.getTarget());
                }
                return "Input into " + getElementDescription(event.getTarget());
            }
            return "Input event";
        } else if ("select".equals(type)) {
            if (event.getTarget() != null) {
                String value = event.getTargetValue();
                if (value != null) {
                    return "Select \"" + value + "\" from " + getElementDescription(event.getTarget());
                }
                return "Select from " + getElementDescription(event.getTarget());
            }
            return "Select event";
        } else if ("assertion".equals(type)) {
            // Handle custom assertions
            String assertionType = (String) event.getAdditionalData("assertionType");
            String assertionValue = (String) event.getAdditionalData("assertionValue");
            
            if (assertionType != null) {
                if ("isVisible".equals(assertionType)) {
                    if (event.getTarget() != null) {
                        return "Assert that " + getElementDescription(event.getTarget()) + " is visible";
                    }
                } else if ("hasText".equals(assertionType)) {
                    if (event.getTarget() != null) {
                        return "Assert that " + getElementDescription(event.getTarget()) + 
                               " has text \"" + assertionValue + "\"";
                    }
                } else if ("hasValue".equals(assertionType)) {
                    if (event.getTarget() != null) {
                        return "Assert that " + getElementDescription(event.getTarget()) + 
                               " has value \"" + assertionValue + "\"";
                    }
                }
            }
            
            return "Assertion";
        } else if ("wait".equals(type)) {
            // Handle custom waits
            String waitType = (String) event.getAdditionalData("waitType");
            String waitValue = (String) event.getAdditionalData("waitValue");
            Object waitTimeout = event.getAdditionalData("waitTimeout");
            
            if (waitType != null) {
                if ("waitForElement".equals(waitType)) {
                    if (event.getTarget() != null) {
                        return "Wait for " + getElementDescription(event.getTarget()) + " to be visible";
                    }
                } else if ("waitForText".equals(waitType)) {
                    return "Wait for text \"" + waitValue + "\"";
                } else if ("waitForNavigation".equals(waitType)) {
                    return "Wait for navigation to complete";
                } else if ("waitForTimeout".equals(waitType)) {
                    if (waitTimeout != null) {
                        return "Wait for " + waitTimeout + " milliseconds";
                    }
                    return "Wait for timeout";
                }
            }
            
            return "Wait";
        } else if ("custom_code".equals(type)) {
            // Handle custom code
            String code = (String) event.getAdditionalData("code");
            if (code != null && !code.isEmpty()) {
                return "Execute custom code: " + (code.length() > 30 ? 
                       code.substring(0, 30) + "..." : code);
            }
            return "Execute custom code";
        } else if ("contextmenu".equals(type)) {
            if (event.getTarget() != null) {
                return "Right-click on " + getElementDescription(event.getTarget());
            }
            return "Context menu event";
        } else if ("dragstart".equals(type) || "drop".equals(type)) {
            if (event.getTarget() != null) {
                return (type.equals("dragstart") ? "Start drag from " : "Drop onto ") + 
                       getElementDescription(event.getTarget());
            }
            return type.equals("dragstart") ? "Drag start event" : "Drop event";
        } else if ("keydown".equals(type)) {
            String key = (String) event.getAdditionalData("key");
            if (key != null) {
                if (event.getTarget() != null) {
                    return "Press " + key + " key in " + getElementDescription(event.getTarget());
                }
                return "Press " + key + " key";
            }
            return "Key press event";
        }
        
        // Default case
        return "Event: " + type;
    }
    
    /**
     * Generate BDD steps from a recording session
     */
    private List<String> generateBddSteps(RecordingSession session) {
        List<String> bddSteps = new ArrayList<>();
        
        // Sort events by timestamp
        List<RecordingEvent> sortedEvents = new ArrayList<>(session.getEvents());
        sortedEvents.sort(Comparator.comparing(RecordingEvent::getTimestamp));
        
        // Track if we have added Given/When/Then
        boolean hasGiven = false;
        boolean hasWhen = false;
        boolean hasThen = false;
        
        for (RecordingEvent event : sortedEvents) {
            String type = event.getType();
            
            // First step is always Given
            if (!hasGiven) {
                if ("navigation".equals(type)) {
                    if (event.getPageState() != null) {
                        bddSteps.add("Given user navigates to \"" + event.getPageState().getUrl() + "\"");
                    } else {
                        bddSteps.add("Given user navigates to the application");
                    }
                    hasGiven = true;
                    continue;
                } else {
                    // If first event is not navigation, add a default Given
                    bddSteps.add("Given user navigates to the application");
                    hasGiven = true;
                }
            }
            
            // Use When for action steps
            if (("click".equals(type) || "input".equals(type) || "change".equals(type) || 
                "select".equals(type) || "contextmenu".equals(type) || "dragstart".equals(type) || 
                "drop".equals(type) || "keydown".equals(type)) && !hasWhen) {
                
                // Add a When prefix
                String step = generateBddStep(event);
                if (step != null && !step.isEmpty()) {
                    if (step.startsWith("When ") || step.startsWith("Given ") || 
                        step.startsWith("Then ") || step.startsWith("And ")) {
                        bddSteps.add(step);
                    } else {
                        bddSteps.add("When " + step);
                    }
                    hasWhen = true;
                    continue;
                }
            }
            
            // Use Then for assertions
            if ("assertion".equals(type) && !hasThen) {
                // Add a Then prefix
                String step = generateBddStep(event);
                if (step != null && !step.isEmpty()) {
                    if (step.startsWith("When ") || step.startsWith("Given ") || 
                        step.startsWith("Then ") || step.startsWith("And ")) {
                        bddSteps.add(step);
                    } else {
                        bddSteps.add("Then " + step);
                    }
                    hasThen = true;
                    continue;
                }
            }
            
            // Use And for remaining steps
            String step = generateBddStep(event);
            if (step != null && !step.isEmpty()) {
                if (step.startsWith("When ") || step.startsWith("Given ") || 
                    step.startsWith("Then ") || step.startsWith("And ")) {
                    bddSteps.add(step);
                } else {
                    bddSteps.add("And " + step);
                }
            }
        }
        
        return bddSteps;
    }
    
    /**
     * Generate a BDD style step from an event
     */
    private String generateBddStep(RecordingEvent event) {
        String type = event.getType();
        
        if ("navigation".equals(type)) {
            if (event.getPageState() != null) {
                return "Given user navigates to \"" + event.getPageState().getUrl() + "\"";
            }
            return "Given user navigates to the application";
        } else if ("click".equals(type)) {
            if (event.getTarget() != null) {
                return "When user clicks on " + getElementDescription(event.getTarget());
            }
            return "When user clicks";
        } else if ("input".equals(type) || "change".equals(type)) {
            if (event.getTarget() != null) {
                String value = event.getTargetValue();
                if (value != null) {
                    return "When user enters \"" + value + "\" into " + getElementDescription(event.getTarget());
                }
                return "When user enters data into " + getElementDescription(event.getTarget());
            }
            return "When user enters data";
        } else if ("select".equals(type)) {
            if (event.getTarget() != null) {
                String value = event.getTargetValue();
                if (value != null) {
                    return "When user selects \"" + value + "\" from " + getElementDescription(event.getTarget());
                }
                return "When user selects an option from " + getElementDescription(event.getTarget());
            }
            return "When user selects an option";
        } else if ("assertion".equals(type)) {
            // Handle custom assertions
            String assertionType = (String) event.getAdditionalData("assertionType");
            String assertionValue = (String) event.getAdditionalData("assertionValue");
            
            if (assertionType != null) {
                if ("isVisible".equals(assertionType)) {
                    if (event.getTarget() != null) {
                        return "Then user verifies that " + getElementDescription(event.getTarget()) + " is visible";
                    }
                } else if ("hasText".equals(assertionType)) {
                    if (event.getTarget() != null) {
                        return "Then user verifies that " + getElementDescription(event.getTarget()) + 
                               " has text \"" + assertionValue + "\"";
                    }
                } else if ("hasValue".equals(assertionType)) {
                    if (event.getTarget() != null) {
                        return "Then user verifies that " + getElementDescription(event.getTarget()) + 
                               " has value \"" + assertionValue + "\"";
                    }
                }
            }
            
            return "Then user verifies the assertion";
        } else if ("wait".equals(type)) {
            // Handle custom waits
            String waitType = (String) event.getAdditionalData("waitType");
            String waitValue = (String) event.getAdditionalData("waitValue");
            Object waitTimeout = event.getAdditionalData("waitTimeout");
            
            if (waitType != null) {
                if ("waitForElement".equals(waitType)) {
                    if (event.getTarget() != null) {
                        return "And user waits for " + getElementDescription(event.getTarget()) + " to be visible";
                    }
                } else if ("waitForText".equals(waitType)) {
                    return "And user waits for text \"" + waitValue + "\"";
                } else if ("waitForNavigation".equals(waitType)) {
                    return "And user waits for navigation to complete";
                } else if ("waitForTimeout".equals(waitType)) {
                    if (waitTimeout != null) {
                        return "And user waits for " + waitTimeout + " milliseconds";
                    }
                    return "And user waits";
                }
            }
            
            return "And user waits";
        } else if ("custom_code".equals(type)) {
            // Handle custom code
            String code = (String) event.getAdditionalData("code");
            if (code != null && !code.isEmpty()) {
                return "And user executes custom code";
            }
            return "And user executes custom code";
        } else if ("contextmenu".equals(type)) {
            if (event.getTarget() != null) {
                return "When user right-clicks on " + getElementDescription(event.getTarget());
            }
            return "When user right-clicks";
        } else if ("dragstart".equals(type) || "drop".equals(type)) {
            if (event.getTarget() != null) {
                if (type.equals("dragstart")) {
                    return "When user starts dragging " + getElementDescription(event.getTarget());
                } else {
                    return "And user drops onto " + getElementDescription(event.getTarget());
                }
            }
            return type.equals("dragstart") ? 
                   "When user starts dragging an element" : "And user drops onto an element";
        } else if ("keydown".equals(type)) {
            String key = (String) event.getAdditionalData("key");
            if (key != null) {
                if (event.getTarget() != null) {
                    return "When user presses " + key + " key in " + getElementDescription(event.getTarget());
                }
                return "When user presses " + key + " key";
            }
            return "When user presses a key";
        }
        
        // Default case
        return "And user performs an action";
    }
    
    /**
     * Get a description of an element
     */
    private String getElementDescription(RecordingEvent.ElementInfo element) {
        if (element == null) {
            return "element";
        }
        
        String tagName = element.getTagName();
        if (tagName == null || tagName.isEmpty()) {
            return "element";
        }
        
        String id = element.getId();
        String name = element.getName();
        String type = element.getType();
        String text = element.getText();
        
        // Handle common element types
        if ("A".equalsIgnoreCase(tagName)) {
            if (text != null && !text.isEmpty()) {
                if (text.length() > 20) {
                    return "link \"" + text.substring(0, 17) + "...\"";
                }
                return "link \"" + text + "\"";
            } else if (id != null && !id.isEmpty()) {
                return "link with ID \"" + id + "\"";
            }
            return "link";
        } else if ("BUTTON".equalsIgnoreCase(tagName)) {
            if (text != null && !text.isEmpty()) {
                if (text.length() > 20) {
                    return "button \"" + text.substring(0, 17) + "...\"";
                }
                return "button \"" + text + "\"";
            } else if (id != null && !id.isEmpty()) {
                return "button with ID \"" + id + "\"";
            }
            return "button";
        } else if ("INPUT".equalsIgnoreCase(tagName)) {
            if (type != null && !type.isEmpty()) {
                if ("text".equalsIgnoreCase(type)) {
                    if (id != null && !id.isEmpty()) {
                        return "text field with ID \"" + id + "\"";
                    } else if (name != null && !name.isEmpty()) {
                        return "text field with name \"" + name + "\"";
                    }
                    return "text field";
                } else if ("password".equalsIgnoreCase(type)) {
                    return "password field";
                } else if ("checkbox".equalsIgnoreCase(type)) {
                    return "checkbox";
                } else if ("radio".equalsIgnoreCase(type)) {
                    return "radio button";
                } else if ("submit".equalsIgnoreCase(type)) {
                    if (value != null && !value.isEmpty()) {
                        return "submit button \"" + value + "\"";
                    }
                    return "submit button";
                }
            }
            
            if (id != null && !id.isEmpty()) {
                return "input field with ID \"" + id + "\"";
            } else if (name != null && !name.isEmpty()) {
                return "input field with name \"" + name + "\"";
            }
            return "input field";
        } else if ("SELECT".equalsIgnoreCase(tagName)) {
            if (id != null && !id.isEmpty()) {
                return "dropdown with ID \"" + id + "\"";
            } else if (name != null && !name.isEmpty()) {
                return "dropdown with name \"" + name + "\"";
            }
            return "dropdown";
        } else if ("TEXTAREA".equalsIgnoreCase(tagName)) {
            if (id != null && !id.isEmpty()) {
                return "text area with ID \"" + id + "\"";
            } else if (name != null && !name.isEmpty()) {
                return "text area with name \"" + name + "\"";
            }
            return "text area";
        } else if ("LABEL".equalsIgnoreCase(tagName)) {
            if (text != null && !text.isEmpty()) {
                if (text.length() > 20) {
                    return "label \"" + text.substring(0, 17) + "...\"";
                }
                return "label \"" + text + "\"";
            }
            return "label";
        } else if ("DIV".equalsIgnoreCase(tagName) || "SPAN".equalsIgnoreCase(tagName)) {
            if (text != null && !text.isEmpty()) {
                if (text.length() > 20) {
                    return tagName.toLowerCase() + " \"" + text.substring(0, 17) + "...\"";
                }
                return tagName.toLowerCase() + " \"" + text + "\"";
            } else if (id != null && !id.isEmpty()) {
                return tagName.toLowerCase() + " with ID \"" + id + "\"";
            } else if (element.getClassName() != null && !element.getClassName().isEmpty()) {
                return tagName.toLowerCase() + " with class \"" + element.getClassName() + "\"";
            }
            return tagName.toLowerCase() + " element";
        } else if ("IMG".equalsIgnoreCase(tagName)) {
            String alt = element.getAttribute("alt");
            if (alt != null && !alt.isEmpty()) {
                return "image \"" + alt + "\"";
            } else if (id != null && !id.isEmpty()) {
                return "image with ID \"" + id + "\"";
            }
            return "image";
        } else if ("TABLE".equalsIgnoreCase(tagName)) {
            if (id != null && !id.isEmpty()) {
                return "table with ID \"" + id + "\"";
            }
            return "table";
        } else if ("TR".equalsIgnoreCase(tagName)) {
            return "table row";
        } else if ("TD".equalsIgnoreCase(tagName) || "TH".equalsIgnoreCase(tagName)) {
            if (text != null && !text.isEmpty()) {
                if (text.length() > 20) {
                    return "table cell \"" + text.substring(0, 17) + "...\"";
                }
                return "table cell \"" + text + "\"";
            }
            return "table cell";
        }
        
        // Generic case
        if (text != null && !text.isEmpty()) {
            if (text.length() > 20) {
                return tagName.toLowerCase() + " \"" + text.substring(0, 17) + "...\"";
            }
            return tagName.toLowerCase() + " \"" + text + "\"";
        } else if (id != null && !id.isEmpty()) {
            return tagName.toLowerCase() + " with ID \"" + id + "\"";
        }
        
        return tagName.toLowerCase() + " element";
    }
    
    /**
     * Save a screenshot
     */
    public String saveScreenshot(RecordingSession session, String screenshotData) {
        try {
            // Extract the Base64 data
            String base64Image = screenshotData;
            if (base64Image.contains(",")) {
                base64Image = base64Image.split(",")[1];
            }
            
            // Decode the image
            byte[] imageBytes = Base64.getDecoder().decode(base64Image);
            
            // Create the file name
            String fileName = "screenshot_" + System.currentTimeMillis() + ".png";
            Path filePath = Paths.get(screenshotPath, session.getId(), fileName);
            
            // Write the file
            Files.write(filePath, imageBytes);
            
            return filePath.toString();
        } catch (Exception e) {
            logger.error("Failed to save screenshot", e);
            return null;
        }
    }
    
    /**
     * Find all screenshots for a session
     */
    public List<String> findSessionScreenshots(String sessionId) {
        try {
            Path sessionPath = Paths.get(screenshotPath, sessionId);
            if (!Files.exists(sessionPath)) {
                return new ArrayList<>();
            }
            
            return Files.list(sessionPath)
                .filter(file -> file.toString().endsWith(".png"))
                .map(Path::toString)
                .collect(Collectors.toList());
        } catch (Exception e) {
            logger.error("Failed to find session screenshots", e);
            return new ArrayList<>();
        }
    }
    
    /**
     * Get the injector script for a session
     */
    public String getInjectorScript(RecordingSession session, String frameId) {
        try {
            // Read the script if not cached
            if (cachedInjectorScript == null) {
                File scriptFile = ResourceUtils.getFile(injectorScriptPath);
                cachedInjectorScript = new String(Files.readAllBytes(scriptFile.toPath()));
            }
            
            // Get configuration for the session
            Map<String, Object> config = getRecorderConfig(session, frameId);
            
            // Convert the configuration to JSON
            StringBuilder configJson = new StringBuilder();
            configJson.append("{\n");
            for (Map.Entry<String, Object> entry : config.entrySet()) {
                configJson.append("    ").append(entry.getKey()).append(": ");
                
                Object value = entry.getValue();
                if (value instanceof String) {
                    configJson.append("'").append(value).append("'");
                } else if (value instanceof Boolean || value instanceof Number) {
                    configJson.append(value);
                } else if (value instanceof List) {
                    configJson.append("[");
                    List<?> list = (List<?>) value;
                    boolean first = true;
                    for (Object item : list) {
                        if (!first) {
                            configJson.append(", ");
                        }
                        first = false;
                        
                        if (item instanceof String) {
                            configJson.append("'").append(item).append("'");
                        } else {
                            configJson.append(item);
                        }
                    }
                    configJson.append("]");
                } else {
                    configJson.append("null");
                }
                
                configJson.append(",\n");
            }
            // Remove the last comma and new line
            if (config.size() > 0) {
                configJson.delete(configJson.length() - 2, configJson.length());
                configJson.append("\n");
            }
            configJson.append("}");
            
            // Replace the configuration placeholder
            String script = cachedInjectorScript.replace(
                "const config = {", 
                "const config = " + configJson.toString()
            );
            
            return script;
        } catch (Exception e) {
            logger.error("Failed to get injector script", e);
            return "console.error('Failed to get injector script: " + e.getMessage() + "');";
        }
    }
    
    /**
     * Get recorder configuration for a session
     */
    public Map<String, Object> getRecorderConfig(RecordingSession session, String frameId) {
        Map<String, Object> config = new HashMap<>();
        
        // Server URL
        String serverUrl = configManager.getConfigProperty(
            "cstestforge.server.url", 
            "http://localhost:8080"
        );
        config.put("serverUrl", serverUrl);
        
        // Recording ID
        config.put("recordingId", session.getId());
        
        // Frame ID
        if (frameId != null && !frameId.isEmpty()) {
            config.put("frameId", frameId);
            
            // Check if this is the top frame
            RecordingSession.FrameInfo frameInfo = session.getFrame(frameId);
            if (frameInfo != null) {
                config.put("isTopFrame", frameInfo.isTopFrame());
            } else {
                config.put("isTopFrame", false);
            }
        } else {
            config.put("frameId", "main");
            config.put("isTopFrame", true);
        }
        
        // Capture configuration
        config.put("captureScreenshots", Boolean.parseBoolean(
            configManager.getConfigProperty("cstestforge.recorder.capture.screenshots", "true")));
        
        config.put("captureNetwork", Boolean.parseBoolean(
            configManager.getConfigProperty("cstestforge.recorder.capture.network", "true")));
        
        config.put("captureConsole", Boolean.parseBoolean(
            configManager.getConfigProperty("cstestforge.recorder.capture.console", "true")));
        
        // AI configuration
        config.put("smartWaitEnabled", Boolean.parseBoolean(
            configManager.getConfigProperty("cstestforge.recorder.smart.wait", "true")));
        
        config.put("aiLocatorsEnabled", Boolean.parseBoolean(
            configManager.getConfigProperty("cstestforge.recorder.ai.locators", "true")));
        
        // Excluded elements
        List<String> excludedElements = Arrays.asList(
            configManager.getConfigProperty(
                "cstestforge.recorder.excluded.elements", 
                ".cs-testforge-highlight,.cs-testforge-toolbar"
            ).split(",")
        );
        config.put("excludedElements", excludedElements);
        
        // Recording state
        config.put("isPaused", "paused".equals(session.getStatus()));
        
        return config;
    }
    
    /**
     * Register a frame in a recording session
     */
    public void registerFrame(RecordingSession session, String frameId, String url, String title, Boolean isTopFrame) {
        if (frameId == null || frameId.isEmpty()) {
            return;
        }
        
        // Check if the frame is already registered
        if (session.getFrames().containsKey(frameId)) {
            RecordingSession.FrameInfo frameInfo = session.getFrame(frameId);
            
            // Update the frame information
            if (url != null && !url.isEmpty()) {
                frameInfo.setUrl(url);
            }
            
            if (title != null) {
                frameInfo.setTitle(title);
            }
            
            if (isTopFrame != null) {
                frameInfo.setTopFrame(isTopFrame);
            }
        } else {
            // Create a new frame info
            RecordingSession.FrameInfo frameInfo = new RecordingSession.FrameInfo();
            frameInfo.setId(frameId);
            frameInfo.setUrl(url);
            frameInfo.setTitle(title);
            
            if (isTopFrame != null) {
                frameInfo.setTopFrame(isTopFrame);
            } else {
                // If this is the first frame, assume it's the top frame
                frameInfo.setTopFrame(session.getFrames().isEmpty());
            }
            
            session.addFrame(frameId, frameInfo);
        }
        
        logger.debug("Registered frame: {} for session {}", frameId, session.getId());
    }
    
    /**
     * Add a custom action to a recording session
     */
    public void addCustomAction(RecordingSession session, CustomActionRequest request) {
        String actionType = request.getActionType();
        if (actionType == null || actionType.isEmpty()) {
            return;
        }
        
        // Create a new event
        RecordingEvent event = new RecordingEvent();
        event.setId("custom_" + System.currentTimeMillis());
        event.setType(actionType);
        event.setTimestamp(System.currentTimeMillis());
        
        // Find the top frame
        for (Map.Entry<String, RecordingSession.FrameInfo> entry : session.getFrames().entrySet()) {
            if (entry.getValue().isTopFrame()) {
                event.setFrameId(entry.getKey());
                break;
            }
        }
        
        // Add additional data
        if (request.getActionTarget() != null && !request.getActionTarget().isEmpty()) {
            // Find the latest screenshot
            String screenshotPath = session.getLastScreenshotPath();
            
            // Create element info from the target selector
            RecordingEvent.ElementInfo target = new RecordingEvent.ElementInfo();
            
            // Parse the target selector to extract element information
            String targetSelector = request.getActionTarget();
            
            if (targetSelector.startsWith("#")) {
                // ID selector
                target.setTagName("*");
                target.setId(targetSelector.substring(1));
                target.setCssSelector(targetSelector);
                target.setXpath("//*[@id='" + targetSelector.substring(1) + "']");
            } else if (targetSelector.startsWith(".")) {
                // Class selector
                target.setTagName("*");
                target.setClassName(targetSelector.substring(1));
                target.setCssSelector(targetSelector);
                target.setXpath("//*[contains(@class, '" + targetSelector.substring(1) + "')]");
            } else if (targetSelector.contains("[") && targetSelector.contains("]")) {
                // Attribute selector
                target.setTagName("*");
                
                // Extract attribute name and value
                int startBracket = targetSelector.indexOf("[");
                int endBracket = targetSelector.indexOf("]");
                String attr = targetSelector.substring(startBracket + 1, endBracket);
                
                if (targetSelector.startsWith("*")) {
                    target.setTagName("*");
                } else {
                    String tagName = targetSelector.substring(0, startBracket);
                    target.setTagName(tagName.toUpperCase());
                }
                
                // Extract attribute name and value
                if (attr.contains("=")) {
                    String[] parts = attr.split("=");
                    String attrName = parts[0];
                    String attrValue = parts[1].replace("'", "").replace("\"", "");
                    
                    if (attrName.equals("id")) {
                        target.setId(attrValue);
                    } else if (attrName.equals("name")) {
                        target.setName(attrValue);
                    } else if (attrName.equals("class")) {
                        target.setClassName(attrValue);
                    } else {
                        Map<String, String> attributes = new HashMap<>();
                        attributes.put(attrName, attrValue);
                        target.setAttributes(attributes);
                    }
                    
                    target.setCssSelector(targetSelector);
                    target.setXpath("//" + target.getTagName().toLowerCase() + 
                                    "[@" + attrName + "='" + attrValue + "']");
                }
            } else {
                // Element selector or XPath
                if (targetSelector.startsWith("//") || targetSelector.startsWith("(//")) {
                    // XPath
                    target.setXpath(targetSelector);
                    
                    // Extract element tag from XPath
                    Pattern pattern = Pattern.compile("//([a-zA-Z0-9_-]+)");
                    Matcher matcher = pattern.matcher(targetSelector);
                    if (matcher.find()) {
                        target.setTagName(matcher.group(1).toUpperCase());
                    } else {
                        target.setTagName("*");
                    }
                } else {
                    // Simple element selector
                    target.setTagName(targetSelector.toUpperCase());
                    target.setCssSelector(targetSelector);
                    target.setXpath("//" + targetSelector.toLowerCase());
                }
            }
            
            // Set visible status
            target.setVisible(true);
            
            // Set text if provided
            if (request.getActionValue() != null && 
                (actionType.equals("text") || actionType.equals("assertion"))) {
                target.setText(request.getActionValue());
            }
            
            // Set position if provided in additional params
            if (request.getAdditionalParams() != null) {
                Object x = request.getAdditionalParams().get("x");
                Object y = request.getAdditionalParams().get("y");
                
                if (x != null && y != null) {
                    Map<String, Object> bounds = new HashMap<>();
                    bounds.put("x", x);
                    bounds.put("y", y);
                    bounds.put("width", request.getAdditionalParams().getOrDefault("width", 0));
                    bounds.put("height", request.getAdditionalParams().getOrDefault("height", 0));
                    target.setBounds(bounds);
                }
            }
            
            event.setTarget(target);
            
            // Generate AI-based locators
            List<RecordingEvent.LocatorInfo> locators = smartLocatorService.generateLocators(target, session);
            event.setLocators(locators);
        }
        
        if (request.getActionValue() != null) {
            event.addAdditionalData("value", request.getActionValue());
        }
        
        if (request.getAdditionalParams() != null) {
            for (Map.Entry<String, Object> entry : request.getAdditionalParams().entrySet()) {
                event.addAdditionalData(entry.getKey(), entry.getValue());
            }
        }
        
        // Set page state
        RecordingEvent.PageState pageState = new RecordingEvent.PageState();
        
        // Find the current URL from the top frame
        for (Map.Entry<String, RecordingSession.FrameInfo> entry : session.getFrames().entrySet()) {
            if (entry.getValue().isTopFrame()) {
                pageState.setUrl(entry.getValue().getUrl());
                pageState.setTitle(entry.getValue().getTitle());
                break;
            }
        }
        
        pageState.setTimestamp(new Date().toString());
        event.setPageState(pageState);
        
        // Process the event
        processEvent(session, event);
        
        // Add the event to the session
        session.addEvent(event);
        
        logger.debug("Added custom action: {} to session {}", actionType, session.getId());
    }

    /**
     * Generate locators for an element
     */
    public List<RecordingEvent.LocatorInfo> generateLocators(
            RecordingEvent.ElementInfo element, RecordingSession session) {
        if (element == null) {
            return new ArrayList<>();
        }
        
        // Create a unique key for the element
        String elementKey = createElementKey(element);
        
        // Check cache for this element
        if (locatorCache.containsKey(elementKey)) {
            return locatorCache.get(elementKey);
        }
        
        List<RecordingEvent.LocatorInfo> locators = new ArrayList<>();
        
        // Generate locators based on element attributes
        generateIdBasedLocators(element, locators);
        generateAttributeBasedLocators(element, locators);
        generateTextBasedLocators(element, locators);
        generatePositionBasedLocators(element, locators);
        generateXPathLocators(element, locators);
        generateCssSelectorLocators(element, locators);
        
        // Sort locators by priority
        locators.sort((a, b) -> Integer.compare(b.getPriority(), a.getPriority()));
        
        // Cache the locators
        locatorCache.put(elementKey, locators);
        
        return locators;
    }
    
    /**
     * Create a unique key for an element
     */
    private String createElementKey(RecordingEvent.ElementInfo element) {
        StringBuilder key = new StringBuilder();
        
        key.append(element.getTagName()).append("_");
        
        if (element.getId() != null && !element.getId().isEmpty()) {
            key.append("id_").append(element.getId()).append("_");
        }
        
        if (element.getClassName() != null && !element.getClassName().isEmpty()) {
            key.append("class_").append(element.getClassName()).append("_");
        }
        
        if (element.getName() != null && !element.getName().isEmpty()) {
            key.append("name_").append(element.getName()).append("_");
        }
        
        if (element.getText() != null && !element.getText().isEmpty()) {
            // Use only first 20 chars of text to avoid giant keys
            String text = element.getText();
            if (text.length() > 20) {
                text = text.substring(0, 20);
            }
            key.append("text_").append(text).append("_");
        }
        
        // Add position info if available
        if (element.getBounds() != null && !element.getBounds().isEmpty()) {
            Object x = element.getBounds().get("x");
            Object y = element.getBounds().get("y");
            
            if (x != null && y != null) {
                key.append("pos_").append(x).append("_").append(y).append("_");
            }
        }
        
        return key.toString();
    }
    
    /**
     * Generate ID-based locators
     */
    private void generateIdBasedLocators(RecordingEvent.ElementInfo element, 
                                         List<RecordingEvent.LocatorInfo> locators) {
        // ID attribute
        if (element.getId() != null && !element.getId().isEmpty()) {
            RecordingEvent.LocatorInfo locator = new RecordingEvent.LocatorInfo();
            locator.setType("id");
            locator.setValue(element.getId());
            locator.setPriority(LOCATOR_PRIORITIES.get("id"));
            locator.setRobustness("high");
            locators.add(locator);
        }
        
        // Name attribute
        if (element.getName() != null && !element.getName().isEmpty()) {
            RecordingEvent.LocatorInfo locator = new RecordingEvent.LocatorInfo();
            locator.setType("name");
            locator.setValue(element.getName());
            locator.setPriority(LOCATOR_PRIORITIES.get("name"));
            locator.setRobustness("high");
            locators.add(locator);
        }
        
        // Test ID attributes
        if (element.getAttributes() != null) {
            for (String testIdAttr : TEST_ID_ATTRIBUTES) {
                String testId = element.getAttributes().get(testIdAttr);
                if (testId != null && !testId.isEmpty()) {
                    RecordingEvent.LocatorInfo locator = new RecordingEvent.LocatorInfo();
                    locator.setType("testId");
                    locator.setValue(testIdAttr + "=\"" + testId + "\"");
                    locator.setPriority(LOCATOR_PRIORITIES.get("testId"));
                    locator.setRobustness("high");
                    locators.add(locator);
                }
            }
        }
    }
    
    /**
     * Generate attribute-based locators
     */
    private void generateAttributeBasedLocators(RecordingEvent.ElementInfo element, 
                                                List<RecordingEvent.LocatorInfo> locators) {
        if (element.getAttributes() == null) {
            return;
        }
        
        // Role attribute
        String role = element.getAttributes().get("role");
        if (role != null && !role.isEmpty()) {
            RecordingEvent.LocatorInfo locator = new RecordingEvent.LocatorInfo();
            locator.setType("role");
            locator.setValue(role);
            locator.setPriority(LOCATOR_PRIORITIES.get("role"));
            locator.setRobustness("medium");
            locators.add(locator);
        }
        
        // ARIA label
        String ariaLabel = element.getAttributes().get("aria-label");
        if (ariaLabel != null && !ariaLabel.isEmpty()) {
            RecordingEvent.LocatorInfo locator = new RecordingEvent.LocatorInfo();
            locator.setType("label");
            locator.setValue(ariaLabel);
            locator.setPriority(LOCATOR_PRIORITIES.get("label"));
            locator.setRobustness("medium");
            locators.add(locator);
        }
        
        // Check for other useful attributes
        for (Map.Entry<String, String> entry : element.getAttributes().entrySet()) {
            String attrName = entry.getKey();
            String attrValue = entry.getValue();
            
            // Skip already processed and ignored attributes
            if (attrName.equals("id") || attrName.equals("name") || attrName.equals("class") || 
                attrName.equals("role") || attrName.equals("aria-label") || 
                TEST_ID_ATTRIBUTES.contains(attrName) || IGNORE_ATTRIBUTES.contains(attrName) || 
                attrValue == null || attrValue.isEmpty()) {
                continue;
            }
            
            // Add attribute-based locator
            RecordingEvent.LocatorInfo locator = new RecordingEvent.LocatorInfo();
            locator.setType("attribute");
            locator.setValue(attrName + "=\"" + attrValue + "\"");
            locator.setPriority(65); // Lower priority than standard identifiers
            locator.setRobustness("medium");
            locators.add(locator);
        }
    }
    
    /**
     * Generate text-based locators
     */
    private void generateTextBasedLocators(RecordingEvent.ElementInfo element, 
                                           List<RecordingEvent.LocatorInfo> locators) {
        String text = element.getText();
        if (text == null || text.isEmpty()) {
            return;
        }
        
        // Use full text if it's short
        if (text.length() <= 50) {
            RecordingEvent.LocatorInfo locator = new RecordingEvent.LocatorInfo();
            locator.setType("text");
            locator.setValue(text);
            locator.setPriority(LOCATOR_PRIORITIES.get("text"));
            locator.setRobustness("medium");
            locators.add(locator);
        } else {
            // Use first few words for long text
            String shortText = text.substring(0, Math.min(text.length(), 50));
            int lastSpace = shortText.lastIndexOf(' ');
            if (lastSpace > 10) {
                shortText = shortText.substring(0, lastSpace);
            }
            
            RecordingEvent.LocatorInfo locator = new RecordingEvent.LocatorInfo();
            locator.setType("partialText");
            locator.setValue(shortText + "...");
            locator.setPriority(LOCATOR_PRIORITIES.get("text") - 5);
            locator.setRobustness("low");
            locators.add(locator);
        }
        
        // Check for patterns in the text
        Map<String, String> patterns = identifyPatterns(text);
        if (!patterns.isEmpty()) {
            for (Map.Entry<String, String> entry : patterns.entrySet()) {
                RecordingEvent.LocatorInfo locator = new RecordingEvent.LocatorInfo();
                locator.setType("pattern");
                locator.setValue(entry.getKey() + ": " + entry.getValue());
                locator.setPriority(LOCATOR_PRIORITIES.get("pattern"));
                locator.setRobustness("medium");
                locators.add(locator);
            }
        }
    }
    
    /**
     * Generate position-based locators
     */
    private void generatePositionBasedLocators(RecordingEvent.ElementInfo element, 
                                               List<RecordingEvent.LocatorInfo> locators) {
        if (element.getBounds() == null || element.getBounds().isEmpty()) {
            return;
        }
        
        Object x = element.getBounds().get("x");
        Object y = element.getBounds().get("y");
        Object width = element.getBounds().get("width");
        Object height = element.getBounds().get("height");
        
        if (x != null && y != null) {
            RecordingEvent.LocatorInfo locator = new RecordingEvent.LocatorInfo();
            locator.setType("position");
            locator.setValue("position: (" + x + ", " + y + ")");
            locator.setPriority(LOCATOR_PRIORITIES.get("position"));
            locator.setRobustness("low");
            locators.add(locator);
        }
        
        // Check for elements with distinctive size
        if (width != null && height != null) {
            // Square elements
            if (width.equals(height) && Double.parseDouble(width.toString()) > 10) {
                RecordingEvent.LocatorInfo locator = new RecordingEvent.LocatorInfo();
                locator.setType("size");
                locator.setValue("square: " + width + "x" + height);
                locator.setPriority(LOCATOR_PRIORITIES.get("position") - 5);
                locator.setRobustness("low");
                locators.add(locator);
            }
            
            // Very small elements (likely icons)
            if (Double.parseDouble(width.toString()) < 20 && Double.parseDouble(height.toString()) < 20 && 
                Double.parseDouble(width.toString()) > 0 && Double.parseDouble(height.toString()) > 0) {
                
                RecordingEvent.LocatorInfo locator = new RecordingEvent.LocatorInfo();
                locator.setType("size");
                locator.setValue("small: " + width + "x" + height);
                locator.setPriority(LOCATOR_PRIORITIES.get("position") - 5);
                locator.setRobustness("low");
                locators.add(locator);
            }
        }
    }
    
    /**
     * Generate XPath locators
     */
    private void generateXPathLocators(RecordingEvent.ElementInfo element, 
                                       List<RecordingEvent.LocatorInfo> locators) {
        String xpath = element.getXpath();
        if (xpath == null || xpath.isEmpty()) {
            return;
        }
        
        // Add the full XPath
        RecordingEvent.LocatorInfo fullXpath = new RecordingEvent.LocatorInfo();
        fullXpath.setType("xpath");
        fullXpath.setValue(xpath);
        fullXpath.setPriority(LOCATOR_PRIORITIES.get("xpath"));
        fullXpath.setRobustness("low");
        locators.add(fullXpath);
        
        // Try to create a more reliable XPath if possible
        String tagName = element.getTagName();
        if (tagName == null || tagName.isEmpty()) {
            return;
        }
        
        // XPath with ID
        if (element.getId() != null && !element.getId().isEmpty()) {
            RecordingEvent.LocatorInfo idXpath = new RecordingEvent.LocatorInfo();
            idXpath.setType("xpath");
            idXpath.setValue("//" + tagName.toLowerCase() + "[@id='" + element.getId() + "']");
            idXpath.setPriority(LOCATOR_PRIORITIES.get("xpath") + 5);
            idXpath.setRobustness("high");
            locators.add(idXpath);
        }
        
        // XPath with name
        if (element.getName() != null && !element.getName().isEmpty()) {
            RecordingEvent.LocatorInfo nameXpath = new RecordingEvent.LocatorInfo();
            nameXpath.setType("xpath");
            nameXpath.setValue("//" + tagName.toLowerCase() + "[@name='" + element.getName() + "']");
            nameXpath.setPriority(LOCATOR_PRIORITIES.get("xpath") + 4);
            nameXpath.setRobustness("high");
            locators.add(nameXpath);
        }
        
        // XPath with text
        if (element.getText() != null && !element.getText().isEmpty()) {
            String text = element.getText();
            if (text.length() <= 50) {
                RecordingEvent.LocatorInfo textXpath = new RecordingEvent.LocatorInfo();
                textXpath.setType("xpath");
                textXpath.setValue("//" + tagName.toLowerCase() + "[text()='" + text + "']");
                textXpath.setPriority(LOCATOR_PRIORITIES.get("xpath") + 3);
                textXpath.setRobustness("medium");
                locators.add(textXpath);
                
                // XPath with contains for partial text
                if (text.length() > 10) {
                    String partialText = text.substring(0, 10);
                    RecordingEvent.LocatorInfo containsXpath = new RecordingEvent.LocatorInfo();
                    containsXpath.setType("xpath");
                    containsXpath.setValue("//" + tagName.toLowerCase() + 
                                          "[contains(text(),'" + partialText + "')]");
                    containsXpath.setPriority(LOCATOR_PRIORITIES.get("xpath") + 2);
                    containsXpath.setRobustness("low");
                    locators.add(containsXpath);
                }
            }
        }
        
        // XPath with test attributes
        if (element.getAttributes() != null) {
            for (String testIdAttr : TEST_ID_ATTRIBUTES) {
                String testId = element.getAttributes().get(testIdAttr);
                if (testId != null && !testId.isEmpty()) {
                    RecordingEvent.LocatorInfo testIdXpath = new RecordingEvent.LocatorInfo();
                    testIdXpath.setType("xpath");
                    testIdXpath.setValue("//" + tagName.toLowerCase() + "[@" + testIdAttr + "='" + testId + "']");
                    testIdXpath.setPriority(LOCATOR_PRIORITIES.get("xpath") + 5);
                    testIdXpath.setRobustness("high");
                    locators.add(testIdXpath);
                }
            }
        }
    }
    
    /**
     * Generate CSS selector locators
     */
    private void generateCssSelectorLocators(RecordingEvent.ElementInfo element, 
                                            List<RecordingEvent.LocatorInfo> locators) {
        String cssSelector = element.getCssSelector();
        if (cssSelector == null || cssSelector.isEmpty()) {
            return;
        }
        
        // Add the full CSS selector
        RecordingEvent.LocatorInfo fullCss = new RecordingEvent.LocatorInfo();
        fullCss.setType("cssSelector");
        fullCss.setValue(cssSelector);
        fullCss.setPriority(LOCATOR_PRIORITIES.get("cssSelector"));
        fullCss.setRobustness("low");
        locators.add(fullCss);
        
        // Try to create more reliable CSS selectors
        String tagName = element.getTagName();
        if (tagName == null || tagName.isEmpty()) {
            return;
        }
        
        // CSS with ID
        if (element.getId() != null && !element.getId().isEmpty()) {
            RecordingEvent.LocatorInfo idCss = new RecordingEvent.LocatorInfo();
            idCss.setType("cssSelector");
            idCss.setValue("#" + element.getId());
            idCss.setPriority(LOCATOR_PRIORITIES.get("cssSelector") + 5);
            idCss.setRobustness("high");
            locators.add(idCss);
        }
        
        // CSS with class
        if (element.getClassName() != null && !element.getClassName().isEmpty()) {
            // Handle multiple classes
            String[] classes = element.getClassName().split("\\s+");
            
            if (classes.length == 1) {
                RecordingEvent.LocatorInfo classCss = new RecordingEvent.LocatorInfo();
                classCss.setType("cssSelector");
                classCss.setValue("." + classes[0]);
                classCss.setPriority(LOCATOR_PRIORITIES.get("cssSelector") + 2);
                classCss.setRobustness("medium");
                locators.add(classCss);
            } else if (classes.length > 1) {
                // Use the most specific looking class (avoid common utility classes)
                List<String> filteredClasses = Arrays.stream(classes)
                    .filter(c -> !c.startsWith("ng-") && !c.startsWith("ui-") && 
                           !c.equals("active") && !c.equals("disabled") && 
                           !c.equals("selected") && !c.equals("hidden") && 
                           !c.equals("visible") && !c.equals("focused"))
                    .collect(Collectors.toList());
                
                if (!filteredClasses.isEmpty()) {
                    // Use the longest class as it's likely more specific
                    String bestClass = filteredClasses.stream()
                        .max(Comparator.comparingInt(String::length))
                        .orElse(filteredClasses.get(0));
                    
                    RecordingEvent.LocatorInfo classCss = new RecordingEvent.LocatorInfo();
                    classCss.setType("cssSelector");
                    classCss.setValue("." + bestClass);
                    classCss.setPriority(LOCATOR_PRIORITIES.get("cssSelector") + 2);
                    classCss.setRobustness("medium");
                    locators.add(classCss);
                    
                    // Also add tag with class for more specificity
                    RecordingEvent.LocatorInfo tagClassCss = new RecordingEvent.LocatorInfo();
                    tagClassCss.setType("cssSelector");
                    tagClassCss.setValue(tagName.toLowerCase() + "." + bestClass);
                    tagClassCss.setPriority(LOCATOR_PRIORITIES.get("cssSelector") + 3);
                    tagClassCss.setRobustness("medium");
                    locators.add(tagClassCss);
                }
            }
        }
        
        // CSS with attribute selectors for test IDs
        if (element.getAttributes() != null) {
            for (String testIdAttr : TEST_ID_ATTRIBUTES) {
                String testId = element.getAttributes().get(testIdAttr);
                if (testId != null && !testId.isEmpty()) {
                    RecordingEvent.LocatorInfo testIdCss = new RecordingEvent.LocatorInfo();
                    testIdCss.setType("cssSelector");
                    testIdCss.setValue("[" + testIdAttr + "='" + testId + "']");
                    testIdCss.setPriority(LOCATOR_PRIORITIES.get("cssSelector") + 4);
                    testIdCss.setRobustness("high");
                    locators.add(testIdCss);
                    
                    // Also add tag with attribute for more specificity
                    RecordingEvent.LocatorInfo tagAttrCss = new RecordingEvent.LocatorInfo();
                    tagAttrCss.setType("cssSelector");
                    tagAttrCss.setValue(tagName.toLowerCase() + "[" + testIdAttr + "='" + testId + "']");
                    tagAttrCss.setPriority(LOCATOR_PRIORITIES.get("cssSelector") + 5);
                    tagAttrCss.setRobustness("high");
                    locators.add(tagAttrCss);
                }
            }
        }
        
        // CSS with name attribute
        if (element.getName() != null && !element.getName().isEmpty()) {
            RecordingEvent.LocatorInfo nameCss = new RecordingEvent.LocatorInfo();
            nameCss.setType("cssSelector");
            nameCss.setValue("[name='" + element.getName() + "']");
            nameCss.setPriority(LOCATOR_PRIORITIES.get("cssSelector") + 3);
            nameCss.setRobustness("high");
            locators.add(nameCss);
            
            // Also add tag with name for more specificity
            RecordingEvent.LocatorInfo tagNameCss = new RecordingEvent.LocatorInfo();
            tagNameCss.setType("cssSelector");
            tagNameCss.setValue(tagName.toLowerCase() + "[name='" + element.getName() + "']");
            tagNameCss.setPriority(LOCATOR_PRIORITIES.get("cssSelector") + 4);
            tagNameCss.setRobustness("high");
            locators.add(tagNameCss);
        }
    }
    
    /**
     * Identify patterns in text
     */
    private Map<String, String> identifyPatterns(String text) {
        Map<String, String> foundPatterns = new HashMap<>();
        
        for (Map.Entry<String, Pattern> entry : PATTERNS.entrySet()) {
            String patternName = entry.getKey();
            Pattern pattern = entry.getValue();
            
            Matcher matcher = pattern.matcher(text);
            if (matcher.find()) {
                String match = matcher.group();
                foundPatterns.put(patternName, match);
            }
        }
        
        return foundPatterns;
    }
    
    /**
     * Analyze the structure of a page to enhance locator reliability
     */
    public void analyzePageStructure(RecordingSession session) {
        // Create a map of elements by their position
        Map<String, List<RecordingEvent.ElementInfo>> elementsByPosition = new HashMap<>();
        
        // Find all elements from previous events
        for (RecordingEvent event : session.getEvents()) {
            RecordingEvent.ElementInfo element = event.getTarget();
            if (element != null && element.getBounds() != null && !element.getBounds().isEmpty()) {
                Object x = element.getBounds().get("x");
                Object y = element.getBounds().get("y");
                
                if (x != null && y != null) {
                    // Round the position to create zones
                    int xZone = ((Number) x).intValue() / 100;
                    int yZone = ((Number) y).intValue() / 100;
                    String zone = xZone + "_" + yZone;
                    
                    if (!elementsByPosition.containsKey(zone)) {
                        elementsByPosition.put(zone, new ArrayList<>());
                    }
                    elementsByPosition.get(zone).add(element);
                }
            }
        }
        
        // Analyze patterns in each zone
        for (Map.Entry<String, List<RecordingEvent.ElementInfo>> entry : elementsByPosition.entrySet()) {
            String zone = entry.getKey();
            List<RecordingEvent.ElementInfo> elements = entry.getValue();
            
            if (elements.size() > 3) {
                // This zone has multiple elements, analyze patterns
                Map<String, Integer> tagCounts = new HashMap<>();
                
                for (RecordingEvent.ElementInfo element : elements) {
                    String tag = element.getTagName();
                    if (tag != null && !tag.isEmpty()) {
                        tagCounts.put(tag, tagCounts.getOrDefault(tag, 0) + 1);
                    }
                }
                
                // Find the most common tag
                String mostCommonTag = null;
                int maxCount = 0;
                
                for (Map.Entry<String, Integer> tagEntry : tagCounts.entrySet()) {
                    if (tagEntry.getValue() > maxCount) {
                        maxCount = tagEntry.getValue();
                        mostCommonTag = tagEntry.getKey();
                    }
                }
                
                // Store this information for later use
                if (mostCommonTag != null && maxCount > 2) {
                    session.addCustomData("zone_" + zone + "_common_tag", mostCommonTag);
                    session.addCustomData("zone_" + zone + "_elements", elements.size());
                }
            }
        }
    }
