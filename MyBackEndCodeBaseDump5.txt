import React, { useState, useEffect, useRef } from 'react';
import { useRecording } from '../../hooks/useRecording';
import { RecorderContext } from '../../contexts/RecorderContext';
import BrowserFrame from './BrowserFrame';
import ElementHighlighter from './ElementHighlighter';
import ActionToolbar from './ActionToolbar';
import RecorderControls from './RecorderControls';
import RecorderSettings from './RecorderSettings';
import StepsList from './StepsList';
import { RecordingSession, RecordingEvent, RecordingStep } from '../../types/recorder';
import { Api } from '../../services/api';
import { toast } from 'react-toastify';
import './RecorderComponent.css';

interface RecorderComponentProps {
  initialUrl?: string;
  onTestGenerated?: (testId: string) => void;
}

const RecorderComponent: React.FC<RecorderComponentProps> = ({ initialUrl = '', onTestGenerated }) => {
  // State for recorder
  const [url, setUrl] = useState<string>(initialUrl);
  const [isRecording, setIsRecording] = useState<boolean>(false);
  const [isPaused, setIsPaused] = useState<boolean>(false);
  const [recordingSession, setRecordingSession] = useState<RecordingSession | null>(null);
  const [recordingSteps, setRecordingSteps] = useState<RecordingStep[]>([]);
  const [currentHighlight, setCurrentHighlight] = useState<any>(null);
  const [settings, setSettings] = useState({
    browser: 'chrome',
    viewportWidth: 1280,
    viewportHeight: 720,
    tool: 'selenium',
    language: 'java',
    testType: 'testng',
    customOptions: {}
  });
  const [showSettings, setShowSettings] = useState<boolean>(!initialUrl);
  const [isGeneratingTest, setIsGeneratingTest] = useState<boolean>(false);
  const [statusRefreshInterval, setStatusRefreshInterval] = useState<NodeJS.Timeout | null>(null);
  
  // Refs
  const browserFrameRef = useRef<HTMLIFrameElement>(null);
  
  // Use the recording hook
  const { 
    startRecording, 
    stopRecording, 
    togglePause, 
    addCustomAction, 
    getRecordingStatus 
  } = useRecording();
  
  // Initialize the recorder
  useEffect(() => {
    // Cleanup function
    return () => {
      if (recordingSession) {
        // Stop recording when component unmounts
        handleStopRecording();
      }
      
      if (statusRefreshInterval) {
        clearInterval(statusRefreshInterval);
      }
    };
  }, [recordingSession, statusRefreshInterval]);
  
  // Handle starting the recording
  const handleStartRecording = async () => {
    if (!url) {
      toast.error('Please enter a URL to record');
      return;
    }
    
    try {
      // Start recording
      const session = await startRecording({
        url,
        browser: settings.browser,
        viewportWidth: settings.viewportWidth,
        viewportHeight: settings.viewportHeight,
        tool: settings.tool,
        language: settings.language,
        testType: settings.testType
      });
      
      setRecordingSession(session);
      setIsRecording(true);
      setIsPaused(false);
      setRecordingSteps([]);
      
      // Start polling for status updates
      const interval = setInterval(refreshRecordingStatus, 2000);
      setStatusRefreshInterval(interval);
      
      toast.success('Recording started');
    } catch (error) {
      console.error('Failed to start recording:', error);
      toast.error('Failed to start recording. Please try again.');
    }
  };
  
  // Handle stopping the recording
  const handleStopRecording = async () => {
    if (!recordingSession) {
      return;
    }
    
    try {
      // Stop status refresh interval
      if (statusRefreshInterval) {
        clearInterval(statusRefreshInterval);
        setStatusRefreshInterval(null);
      }
      
      // Stop recording
      const result = await stopRecording(recordingSession.id);
      
      setIsRecording(false);
      setIsPaused(false);
      
      if (result.generatedTest && onTestGenerated) {
        onTestGenerated(result.generatedTest.id);
      }
      
      toast.success('Recording completed');
    } catch (error) {
      console.error('Failed to stop recording:', error);
      toast.error('Failed to stop recording. Please try again.');
    }
  };
  
  // Handle pausing/resuming the recording
  const handleTogglePause = async () => {
    if (!recordingSession) {
      return;
    }
    
    try {
      await togglePause(recordingSession.id, !isPaused);
      setIsPaused(!isPaused);
      
      toast.info(isPaused ? 'Recording resumed' : 'Recording paused');
    } catch (error) {
      console.error('Failed to toggle pause:', error);
      toast.error('Failed to toggle pause. Please try again.');
    }
  };
  
  // Handle refreshing recording status
  const refreshRecordingStatus = async () => {
    if (!recordingSession) {
      return;
    }
    
    try {
      const status = await getRecordingStatus(recordingSession.id);
      
      // Update steps based on events
      if (status.eventCount > recordingSteps.length) {
        // Fetch the latest events
        const response = await Api.get(`/api/recorder/events?recordingId=${recordingSession.id}&offset=${recordingSteps.length}`);
        
        if (response.data && response.data.success && response.data.data) {
          const newEvents: RecordingEvent[] = response.data.data;
          
          // Convert events to steps
          const newSteps = newEvents.map(convertEventToStep);
          
          setRecordingSteps(prevSteps => [...prevSteps, ...newSteps]);
        }
      }
      
      // Update pause state
      if (status.isPaused !== undefined) {
        setIsPaused(status.isPaused);
      }
      
      // Update highlights
      if (status.currentElement) {
        setCurrentHighlight(status.currentElement);
      }
    } catch (error) {
      console.error('Failed to refresh recording status:', error);
    }
  };
  
  // Handle changing URL
  const handleUrlChange = (newUrl: string) => {
    setUrl(newUrl);
  };
  
  // Handle changing settings
  const handleSettingsChange = (newSettings: any) => {
    setSettings(prev => ({ ...prev, ...newSettings }));
  };
  
  // Handle adding a custom action
  const handleAddCustomAction = async (actionType: string, actionTarget: string, actionValue: string, additionalParams: any = {}) => {
    if (!recordingSession) {
      return;
    }
    
    try {
      await addCustomAction(recordingSession.id, {
        actionType,
        actionTarget,
        actionValue,
        additionalParams
      });
      
      // Refresh status to get the new action
      await refreshRecordingStatus();
      
      toast.success(`Added ${actionType} action`);
    } catch (error) {
      console.error('Failed to add custom action:', error);
      toast.error('Failed to add custom action. Please try again.');
    }
  };
  
  // Handle generating test code
  const handleGenerateTest = async () => {
    if (!recordingSession) {
      return;
    }
    
    try {
      setIsGeneratingTest(true);
      
      const response = await Api.post('/api/recorder/generate', {
        recordingId: recordingSession.id,
        testName: 'Recorded Test',
        packageName: 'com.cstestforge.tests',
        className: 'RecordedTest',
        description: 'Test generated from recording session',
        generatePageObjects: true
      });
      
      if (response.data && response.data.success && response.data.data) {
        const generatedTest = response.data.data;
        
        if (onTestGenerated) {
          onTestGenerated(generatedTest.id);
        }
        
        toast.success('Test code generated successfully');
      } else {
        toast.error('Failed to generate test code');
      }
    } catch (error) {
      console.error('Failed to generate test code:', error);
      toast.error('Failed to generate test code. Please try again.');
    } finally {
      setIsGeneratingTest(false);
    }
  };
  
  // Handle deleting a step
  const handleDeleteStep = (index: number) => {
    setRecordingSteps(prevSteps => prevSteps.filter((_, i) => i !== index));
  };
  
  // Handle editing a step
  const handleEditStep = (index: number, updatedStep: RecordingStep) => {
    setRecordingSteps(prevSteps => {
      const newSteps = [...prevSteps];
      newSteps[index] = updatedStep;
      return newSteps;
    });
  };
  
  // Convert a recording event to a step
  const convertEventToStep = (event: RecordingEvent): RecordingStep => {
    return {
      id: event.id,
      type: event.type,
      timestamp: event.timestamp,
      description: getEventDescription(event),
      elementInfo: event.target,
      additionalData: event.additionalData || {},
      locators: event.locators || []
    };
  };
  
  // Get description for an event
  const getEventDescription = (event: RecordingEvent): string => {
    const type = event.type;
    const target = event.target;
    
    if (type === 'navigation') {
      if (event.pageState?.url) {
        return `Navigate to ${event.pageState.url}`;
      }
      return 'Navigate to page';
    } else if (type === 'click') {
      if (target) {
        const elementDesc = getElementDescription(target);
        return `Click on ${elementDesc}`;
      }
      return 'Click event';
    } else if (type === 'input' || type === 'change') {
      if (target) {
        const elementDesc = getElementDescription(target);
        const value = event.additionalData?.value || '';
        return `Enter "${value}" into ${elementDesc}`;
      }
      return 'Input event';
    } else if (type === 'select') {
      if (target) {
        const elementDesc = getElementDescription(target);
        const value = event.additionalData?.value || '';
        return `Select "${value}" from ${elementDesc}`;
      }
      return 'Select event';
    } else if (type === 'assertion') {
      // Handle custom assertions
      const assertionType = event.additionalData?.assertionType;
      const assertionValue = event.additionalData?.assertionValue;
      
      if (assertionType) {
        if (assertionType === 'isVisible') {
          if (target) {
            return `Assert that ${getElementDescription(target)} is visible`;
          }
        } else if (assertionType === 'hasText') {
          if (target) {
            return `Assert that ${getElementDescription(target)} has text "${assertionValue}"`;
          }
        } else if (assertionType === 'hasValue') {
          if (target) {
            return `Assert that ${getElementDescription(target)} has value "${assertionValue}"`;
          }
        }
      }
      
      return 'Assertion';
    } else if (type === 'wait') {
      // Handle custom waits
      const waitType = event.additionalData?.waitType;
      const waitValue = event.additionalData?.waitValue;
      const waitTimeout = event.additionalData?.waitTimeout;
      
      if (waitType) {
        if (waitType === 'waitForElement') {
          if (target) {
            return `Wait for ${getElementDescription(target)} to be visible`;
          }
        } else if (waitType === 'waitForText') {
          return `Wait for text "${waitValue}"`;
        } else if (waitType === 'waitForNavigation') {
          return 'Wait for navigation to complete';
        } else if (waitType === 'waitForTimeout') {
          if (waitTimeout) {
            return `Wait for ${waitTimeout} milliseconds`;
          }
          return 'Wait for timeout';
        }
      }
      
      return 'Wait';
    } else if (type === 'custom_code') {
      // Handle custom code
      const code = event.additionalData?.code;
      if (code) {
        return `Execute custom code: ${code.length > 30 ? code.substring(0, 30) + '...' : code}`;
      }
      return 'Execute custom code';
    } else if (type === 'contextmenu') {
      if (target) {
        return `Right-click on ${getElementDescription(target)}`;
      }
      return 'Context menu event';
    } else if (type === 'dragstart' || type === 'drop') {
      if (target) {
        return (type === 'dragstart' ? 'Start drag from ' : 'Drop onto ') + 
               getElementDescription(target);
      }
      return type === 'dragstart' ? 'Drag start event' : 'Drop event';
    } else if (type === 'keydown') {
      const key = event.additionalData?.key;
      if (key) {
        if (target) {
          return `Press ${key} key in ${getElementDescription(target)}`;
        }
        return `Press ${key} key`;
      }
      return 'Key press event';
    }
    
    // Default case
    return `Event: ${type}`;
  };
  
  // Get description for an element
  const getElementDescription = (element: any): string => {
    if (!element) {
      return 'element';
    }
    
    const tagName = element.tagName;
    if (!tagName) {
      return 'element';
    }
    
    const id = element.id;
    const name = element.name;
    const type = element.type;
    const text = element.text;
    
    // Handle common element types
    if (tagName.toUpperCase() === 'A') {
      if (text) {
        if (text.length > 20) {
          return `link "${text.substring(0, 17)}..."`;
        }
        return `link "${text}"`;
      } else if (id) {
        return `link with ID "${id}"`;
      }
      return 'link';
    } else if (tagName.toUpperCase() === 'BUTTON') {
      if (text) {
        if (text.length > 20) {
          return `button "${text.substring(0, 17)}..."`;
        }
        return `button "${text}"`;
      } else if (id) {
        return `button with ID "${id}"`;
      }
      return 'button';
    } else if (tagName.toUpperCase() === 'INPUT') {
      if (type) {
        if (type.toLowerCase() === 'text') {
          if (id) {
            return `text field with ID "${id}"`;
          } else if (name) {
            return `text field with name "${name}"`;
          }
          return 'text field';
        } else if (type.toLowerCase() === 'password') {
          return 'password field';
        } else if (type.toLowerCase() === 'checkbox') {
          return 'checkbox';
        } else if (type.toLowerCase() === 'radio') {
          return 'radio button';
        } else if (type.toLowerCase() === 'submit') {
          if (element.value) {
            return `submit button "${element.value}"`;
          }
          return 'submit button';
        }
      }
      
      if (id) {
        return `input field with ID "${id}"`;
      } else if (name) {
        return `input field with name "${name}"`;
      }
      return 'input field';
    } else if (tagName.toUpperCase() === 'SELECT') {
      if (id) {
        return `dropdown with ID "${id}"`;
      } else if (name) {
        return `dropdown with name "${name}"`;
      }
      return 'dropdown';
    } else if (tagName.toUpperCase() === 'TEXTAREA') {
      if (id) {
        return `text area with ID "${id}"`;
      } else if (name) {
        return `text area with name "${name}"`;
      }
      return 'text area';
    } else if (tagName.toUpperCase() === 'LABEL') {
      if (text) {
        if (text.length > 20) {
          return `label "${text.substring(0, 17)}..."`;
        }
        return `label "${text}"`;
      }
      return 'label';
    } else if (tagName.toUpperCase() === 'DIV' || tagName.toUpperCase() === 'SPAN') {
      if (text) {
        if (text.length > 20) {
          return `${tagName.toLowerCase()} "${text.substring(0, 17)}..."`;
        }
        return `${tagName.toLowerCase()} "${text}"`;
      } else if (id) {
        return `${tagName.toLowerCase()} with ID "${id}"`;
      } else if (element.className) {
        return `${tagName.toLowerCase()} with class "${element.className}"`;
      }
      return `${tagName.toLowerCase()} element`;
    }
    
    // Generic case
    if (text) {
      if (text.length > 20) {
        return `${tagName.toLowerCase()} "${text.substring(0, 17)}..."`;
      }
      return `${tagName.toLowerCase()} "${text}"`;
    } else if (id) {
      return `${tagName.toLowerCase()} with ID "${id}"`;
    }
    
    return `${tagName.toLowerCase()} element`;
  };
  
  return (
    <RecorderContext.Provider value={{ 
      recordingSession, 
      isRecording, 
      isPaused,
      recordingSteps,
      currentHighlight,
      addCustomAction: handleAddCustomAction,
      deleteStep: handleDeleteStep,
      editStep: handleEditStep
    }}>
      <div className="cs-recorder-container">
        <div className="cs-recorder-sidebar">
          <div className="cs-recorder-controls">
            <RecorderControls 
              url={url}
              isRecording={isRecording}
              isPaused={isPaused}
              isGeneratingTest={isGeneratingTest}
              onUrlChange={handleUrlChange}
              onStartRecording={handleStartRecording}
              onStopRecording={handleStopRecording}
              onTogglePause={handleTogglePause}
              onGenerateTest={handleGenerateTest}
              onOpenSettings={() => setShowSettings(true)}
            />
          </div>
          <div className="cs-recorder-steps">
            <StepsList steps={recordingSteps} />
          </div>
          <div className="cs-recorder-toolbar">
            <ActionToolbar />
          </div>
        </div>
        <div className="cs-recorder-browser">
          {!isRecording && showSettings ? (
            <RecorderSettings 
              settings={settings} 
              onSettingsChange={handleSettingsChange}
              onClose={() => setShowSettings(false)}
              onStartRecording={handleStartRecording}
            />
          ) : (
            <BrowserFrame
              ref={browserFrameRef}
              url={recordingSession?.url || url}
              width={settings.viewportWidth}
              height={settings.viewportHeight}
              isRecording={isRecording}
              recordingId={recordingSession?.id}
            />
          )}
        </div>
        {isRecording && currentHighlight && (
          <ElementHighlighter element={currentHighlight} />
        )}
      </div>
    </RecorderContext.Provider>
  );
};

export default RecorderComponent;


import React, { useRef, useEffect, forwardRef, useImperativeHandle } from 'react';
import { Api } from '../../services/api';
import './BrowserFrame.css';

interface BrowserFrameProps {
  url: string;
  width: number;
  height: number;
  isRecording: boolean;
  recordingId?: string;
}

export interface BrowserFrameRef {
  getIframe: () => HTMLIFrameElement | null;
  reload: () => void;
}

const BrowserFrame = forwardRef<BrowserFrameRef, BrowserFrameProps>((props, ref) => {
  const { url, width, height, isRecording, recordingId } = props;
  const iframeRef = useRef<HTMLIFrameElement>(null);
  const injectedScriptRef = useRef<boolean>(false);

  // Expose methods to parent component
  useImperativeHandle(ref, () => ({
    getIframe: () => iframeRef.current,
    reload: () => {
      if (iframeRef.current) {
        iframeRef.current.src = iframeRef.current.src;
      }
    }
  }));

  // Handle iframe load event
  const handleIframeLoad = () => {
    if (isRecording && recordingId && !injectedScriptRef.current) {
      injectRecorderScript();
    }
  };

  // Inject recorder script into iframe
  const injectRecorderScript = async () => {
    if (!recordingId || !iframeRef.current) {
      return;
    }

    try {
      // Get the iframe document
      const iframe = iframeRef.current;
      const iframeDoc = iframe.contentDocument || iframe.contentWindow?.document;

      if (!iframeDoc) {
        console.error('Cannot access iframe document');
        return;
      }

      // Create script element
      const script = iframeDoc.createElement('script');
      script.async = true;

      // Set script content or src
      if (process.env.NODE_ENV === 'development') {
        // In development, fetch the script content and inject it directly
        const response = await Api.get(`/api/recorder/inject?recordingId=${recordingId}&timestamp=${Date.now()}`);
        script.textContent = response.data;
      } else {
        // In production, use a script src
        script.src = `/api/recorder/inject?recordingId=${recordingId}&timestamp=${Date.now()}`;
      }

      // Append script to iframe
      iframeDoc.head.appendChild(script);
      
      // Mark as injected
      injectedScriptRef.current = true;
      
      console.log('Recorder script injected');
    } catch (error) {
      console.error('Failed to inject recorder script:', error);
    }
  };

  // When recording state changes
  useEffect(() => {
    if (isRecording && recordingId) {
      // Reset injection flag when recording starts
      injectedScriptRef.current = false;
      
      // Try to inject script if iframe is already loaded
      if (iframeRef.current) {
        injectRecorderScript();
      }
    }
  }, [isRecording, recordingId]);

  // Reset injection flag when URL changes
  useEffect(() => {
    injectedScriptRef.current = false;
  }, [url]);

  return (
    <div className="cs-browser-frame">
      <div className="cs-browser-header">
        <div className="cs-browser-controls">
          <span className="cs-browser-control cs-browser-control-close"></span>
          <span className="cs-browser-control cs-browser-control-minimize"></span>
          <span className="cs-browser-control cs-browser-control-maximize"></span>
        </div>
        <div className="cs-browser-address-bar">
          <span className="cs-browser-address">{url}</span>
        </div>
        <div className="cs-browser-actions">
          <button 
            className="cs-browser-refresh" 
            onClick={() => {
              if (iframeRef.current) {
                iframeRef.current.src = iframeRef.current.src;
              }
            }}
          >
            ↻
          </button>
        </div>
      </div>
      <div className="cs-browser-viewport" style={{ width, height }}>
        {url ? (
          <iframe
            ref={iframeRef}
            src={url}
            width={width}
            height={height}
            onLoad={handleIframeLoad}
            sandbox="allow-same-origin allow-scripts allow-popups allow-forms allow-modals"
            allow="camera; microphone; geolocation"
            className="cs-browser-iframe"
          ></iframe>
        ) : (
          <div className="cs-browser-placeholder">
            <p>Enter a URL to start recording</p>
          </div>
        )}
      </div>
      {isRecording && (
        <div className="cs-recording-indicator">
          <span className="cs-recording-dot"></span>
          Recording
        </div>
      )}
    </div>
  );
});

export default BrowserFrame;



import React, { useState } from 'react';
import './RecorderControls.css';

interface RecorderControlsProps {
  url: string;
  isRecording: boolean;
  isPaused: boolean;
  isGeneratingTest: boolean;
  onUrlChange: (url: string) => void;
  onStartRecording: () => void;
  onStopRecording: () => void;
  onTogglePause: () => void;
  onGenerateTest: () => void;
  onOpenSettings: () => void;
}

const RecorderControls: React.FC<RecorderControlsProps> = ({
  url,
  isRecording,
  isPaused,
  isGeneratingTest,
  onUrlChange,
  onStartRecording,
  onStopRecording,
  onTogglePause,
  onGenerateTest,
  onOpenSettings
}) => {
  const [localUrl, setLocalUrl] = useState<string>(url);

  // Handle URL input change
  const handleUrlChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    setLocalUrl(e.target.value);
  };

  // Handle URL form submission
  const handleUrlSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    
    // Add https:// prefix if not present
    let finalUrl = localUrl.trim();
    if (finalUrl && !finalUrl.startsWith('http://') && !finalUrl.startsWith('https://')) {
      finalUrl = 'https://' + finalUrl;
      setLocalUrl(finalUrl);
    }
    
    onUrlChange(finalUrl);
    
    if (isRecording) {
      // If already recording, this is a navigation within the recording
      // This will be handled by the BrowserFrame component
    } else {
      // If not recording, start a new recording
      onStartRecording();
    }
  };

  return (
    <div className="cs-recorder-controls">
      <div className="cs-url-form">
        <form onSubmit={handleUrlSubmit}>
          <div className="cs-url-input-container">
            <input
              type="text"
              className="cs-url-input"
              value={localUrl}
              onChange={handleUrlChange}
              placeholder="Enter URL to record (e.g., https://example.com)"
              disabled={isRecording && !isPaused}
            />
            {!isRecording && (
              <button
                type="submit"
                className="cs-go-button"
                disabled={!localUrl.trim()}
              >
                Go
              </button>
            )}
          </div>
        </form>
      </div>

      <div className="cs-control-buttons">
        {!isRecording ? (
          <>
            <button
              className="cs-control-button cs-record-button"
              onClick={onStartRecording}
              disabled={!url.trim()}
              title="Start Recording"
            >
              <span className="cs-button-icon">⬤</span>
              Record
            </button>
            <button
              className="cs-control-button cs-settings-button"
              onClick={onOpenSettings}
              title="Recorder Settings"
            >
              <span className="cs-button-icon">⚙️</span>
              Settings
            </button>
          </>
        ) : (
          <>
            <button
              className="cs-control-button cs-stop-button"
              onClick={onStopRecording}
              title="Stop Recording"
            >
              <span className="cs-button-icon">◼</span>
              Stop
            </button>
            <button
              className="cs-control-button cs-pause-button"
              onClick={onTogglePause}
              title={isPaused ? "Resume Recording" : "Pause Recording"}
            >
              <span className="cs-button-icon">
                {isPaused ? "▶" : "⏸"}
              </span>
              {isPaused ? "Resume" : "Pause"}
            </button>
            <button
              className="cs-control-button cs-generate-button"
              onClick={onGenerateTest}
              disabled={isGeneratingTest}
              title="Generate Test Code"
            >

import React, { useState, useContext } from 'react';
import { RecorderContext } from '../../contexts/RecorderContext';
import { RecordingStep } from '../../types/recorder';
import './StepsList.css';

interface StepsListProps {
  steps: RecordingStep[];
}

const StepsList: React.FC<StepsListProps> = ({ steps }) => {
  const { deleteStep, editStep } = useContext(RecorderContext);
  const [editingStepIndex, setEditingStepIndex] = useState<number | null>(null);
  const [editingStepText, setEditingStepText] = useState<string>('');
  const [selectedStep, setSelectedStep] = useState<number | null>(null);

  // Handle editing step
  const handleStartEditing = (index: number) => {
    setEditingStepIndex(index);
    setEditingStepText(steps[index].description);
  };

  // Handle saving edited step
  const handleSaveEdit = () => {
    if (editingStepIndex !== null) {
      const updatedStep = { ...steps[editingStepIndex], description: editingStepText };
      editStep(editingStepIndex, updatedStep);
      setEditingStepIndex(null);
      setEditingStepText('');
    }
  };

  // Handle canceling edit
  const handleCancelEdit = () => {
    setEditingStepIndex(null);
    setEditingStepText('');
  };

  // Handle deleting step
  const handleDeleteStep = (index: number) => {
    if (window.confirm('Are you sure you want to delete this step?')) {
      deleteStep(index);
      setSelectedStep(null);
    }
  };

  // Handle selecting step
  const handleSelectStep = (index: number) => {
    setSelectedStep(selectedStep === index ? null : index);
  };

  // Handle moving step up
  const handleMoveUp = (index: number) => {
    if (index > 0) {
      const reorderedSteps = [...steps];
      const temp = reorderedSteps[index];
      reorderedSteps[index] = reorderedSteps[index - 1];
      reorderedSteps[index - 1] = temp;
      
      // Update steps
      reorderedSteps.forEach((step, i) => {
        editStep(i, step);
      });
      
      // Update selected step
      setSelectedStep(index - 1);
    }
  };

  // Handle moving step down
  const handleMoveDown = (index: number) => {
    if (index < steps.length - 1) {
      const reorderedSteps = [...steps];
      const temp = reorderedSteps[index];
      reorderedSteps[index] = reorderedSteps[index + 1];
      reorderedSteps[index + 1] = temp;
      
      // Update steps
      reorderedSteps.forEach((step, i) => {
        editStep(i, step);
      });
      
      // Update selected step
      setSelectedStep(index + 1);
    }
  };

  // Format timestamp
  const formatTimestamp = (timestamp: number): string => {
    const date = new Date(timestamp);
    return date.toLocaleTimeString();
  };

  // Get icon for step type
  const getStepIcon = (type: string): string => {
    switch (type) {
      case 'navigation':
        return '🌐';
      case 'click':
        return '👆';
      case 'input':
      case 'change':
        return '✏️';
      case 'select':
        return '📋';
      case 'assertion':
        return '✅';
      case 'wait':
        return '⏱️';
      case 'custom_code':
        return '💻';
      case 'contextmenu':
        return '📌';
      case 'dragstart':
      case 'drop':
        return '🔀';
      case 'keydown':
        return '⌨️';
      default:
        return '🔹';
    }
  };

  return (
    <div className="cs-steps-list">
      <div className="cs-steps-header">
        <h3>Recorded Steps ({steps.length})</h3>
      </div>
      
      {steps.length === 0 ? (
        <div className="cs-steps-empty">
          <p>No steps recorded yet</p>
          <p>Start interacting with the page to record steps</p>
        </div>
      ) : (
        <ul className="cs-steps">
          {steps.map((step, index) => (
            <li 
              key={step.id} 
              className={`cs-step ${selectedStep === index ? 'selected' : ''}`}
              onClick={() => handleSelectStep(index)}
            >
              <div className="cs-step-header">
                <span className="cs-step-icon">{getStepIcon(step.type)}</span>
                <span className="cs-step-number">{index + 1}</span>
                <span className="cs-step-time">{formatTimestamp(step.timestamp)}</span>
                <span className="cs-step-type">{step.type}</span>
              </div>
              
              <div className="cs-step-content">
                {editingStepIndex === index ? (
                  <div className="cs-step-edit">
                    <input
                      type="text"
                      value={editingStepText}
                      onChange={(e) => setEditingStepText(e.target.value)}
                      className="cs-step-edit-input"
                      autoFocus
                    />
                    <div className="cs-step-edit-actions">
                      <button onClick={handleSaveEdit} className="cs-step-edit-save">Save</button>
                      <button onClick={handleCancelEdit} className="cs-step-edit-cancel">Cancel</button>
                    </div>
                  </div>
                ) : (
                  <div className="cs-step-description">{step.description}</div>
                )}
              </div>
              
              {selectedStep === index && (
                <div className="cs-step-actions">
                  <button
                    onClick={() => handleStartEditing(index)}
                    className="cs-step-action cs-step-edit-btn"
                    title="Edit Step"
                  >
                    ✎
                  </button>
                  <button
                    onClick={() => handleDeleteStep(index)}
                    className="cs-step-action cs-step-delete-btn"
                    title="Delete Step"
                  >
                    🗑
                  </button>
                  <button
                    onClick={() => handleMoveUp(index)}
                    className="cs-step-action cs-step-move-up"
                    disabled={index === 0}
                    title="Move Step Up"
                  >
                    ↑
                  </button>
                  <button
                    onClick={() => handleMoveDown(index)}
                    className="cs-step-action cs-step-move-down"
                    disabled={index === steps.length - 1}
                    title="Move Step Down"
                  >
                    ↓
                  </button>
                </div>
              )}
              
              {step.elementInfo && selectedStep === index && (
                <div className="cs-step-details">
                  <div className="cs-element-info">
                    <strong>Element:</strong> {step.elementInfo.tagName}
                    {step.elementInfo.id && <span> #<code>{step.elementInfo.id}</code></span>}
                    {step.elementInfo.className && <span> .<code>{step.elementInfo.className}</code></span>}
                  </div>
                  
                  {step.locators && step.locators.length > 0 && (
                    <div className="cs-locators">
                      <strong>Best Locator:</strong>
                      <code>{step.locators[0].value}</code>
                    </div>
                  )}
                  
                  {step.additionalData && Object.keys(step.additionalData).length > 0 && (
                    <div className="cs-additional-data">
                      <strong>Data:</strong>
                      <pre>{JSON.stringify(step.additionalData, null, 2)}</pre>
                    </div>
                  )}
                </div>
              )}
            </li>
          ))}
        </ul>
      )}
    </div>
  );
};

import React, { useEffect, useRef, useState } from 'react';
import './ElementHighlighter.css';

interface ElementHighlighterProps {
  element: any;
}

const ElementHighlighter: React.FC<ElementHighlighterProps> = ({ element }) => {
  const highlighterRef = useRef<HTMLDivElement>(null);
  const [showDetails, setShowDetails] = useState<boolean>(false);
  const [position, setPosition] = useState<{
    top: number;
    left: number;
    width: number;
    height: number;
  }>({
    top: 0,
    left: 0,
    width: 0,
    height: 0,
  });

  // Update highlighter position based on element bounds
  useEffect(() => {
    if (!element || !element.bounds) {
      return;
    }

    const bounds = element.bounds;
    
    setPosition({
      top: bounds.top || 0,
      left: bounds.left || 0,
      width: bounds.width || 0,
      height: bounds.height || 0,
    });
  }, [element]);

  // Toggle element details
  const toggleDetails = () => {
    setShowDetails(!showDetails);
  };

  // Get best locator from element
  const getBestLocator = () => {
    if (!element || !element.locators || element.locators.length === 0) {
      return null;
    }

    // Find the locator with the highest priority
    return element.locators.reduce((best: any, current: any) => {
      if (!best) return current;
      return current.priority > best.priority ? current : best;
    }, null);
  };

  // Get readable element type
  const getElementType = () => {
    if (!element) return 'Unknown';

    if (element.tagName) {
      const tag = element.tagName.toLowerCase();
      
      // Determine element type based on tag and attributes
      if (tag === 'input') {
        const type = element.type?.toLowerCase() || '';
        
        if (type === 'button' || type === 'submit' || type === 'reset') {
          return 'Button';
        } else if (type === 'checkbox') {
          return 'Checkbox';
        } else if (type === 'radio') {
          return 'Radio Button';
        } else if (type === 'text' || type === 'email' || type === 'password' || type === 'search') {
          return 'Text Input';
        } else {
          return 'Input';
        }
      } else if (tag === 'button') {
        return 'Button';
      } else if (tag === 'a') {
        return 'Link';
      } else if (tag === 'select') {
        return 'Dropdown';
      } else if (tag === 'textarea') {
        return 'Text Area';
      } else if (tag === 'img') {
        return 'Image';
      } else if (tag.match(/^h[1-6]$/)) {
        return 'Heading';
      } else if (tag === 'div' || tag === 'span') {
        return 'Container';
      } else {
        return tag.charAt(0).toUpperCase() + tag.slice(1);
      }
    }

    return 'Element';
  };

  if (!element) {
    return null;
  }

  // Calculate styles for the highlighter
  const style = {
    top: `${position.top}px`,
    left: `${position.left}px`,
    width: `${position.width}px`,
    height: `${position.height}px`,
  };

  // Determine where to place the details popup
  const detailsStyle = {
    top: position.top + position.height + 5,
    left: position.left,
  };

  // If too close to the bottom of the viewport, place details above
  if (position.top + position.height + 200 > window.innerHeight) {
    detailsStyle.top = position.top - 200 - 5;
  }

  // If too close to the right edge, align right edge of details with right edge of element
  if (position.left + 300 > window.innerWidth) {
    detailsStyle.left = position.left + position.width - 300;
  }

  const bestLocator = getBestLocator();

  return (
    <>
      <div 
        ref={highlighterRef} 
        className="cs-element-highlighter" 
        style={style}
        onClick={toggleDetails}
      >
        <div className="cs-highlighter-info">
          {element.tagName && <span className="cs-highlighter-tag">{element.tagName}</span>}
          {element.id && <span className="cs-highlighter-id">#{element.id}</span>}

import { useState, useCallback } from 'react';
import { Api } from '../services/api';
import { RecordingSession, RecordingResult, CustomActionRequest } from '../types/recorder';

interface StartRecordingParams {
  url: string;
  browser: string;
  viewportWidth: number;
  viewportHeight: number;
  tool: string;
  language: string;
  testType: string;
}

/**
 * Hook for managing test recording functionality
 */
export const useRecording = () => {
  const [loading, setLoading] = useState<boolean>(false);
  const [error, setError] = useState<string | null>(null);
  
  /**
   * Start a new recording session
   */
  const startRecording = useCallback(async (params: StartRecordingParams): Promise<RecordingSession> => {
    try {
      setLoading(true);
      setError(null);
      
      const response = await Api.post('/api/recorder/start', params);
      
      if (!response.data.success) {
        throw new Error(response.data.message || 'Failed to start recording');
      }
      
      return response.data.data as RecordingSession;
    } catch (err: any) {
      const errorMessage = err.response?.data?.message || err.message || 'Failed to start recording';
      setError(errorMessage);
      throw new Error(errorMessage);
    } finally {
      setLoading(false);
    }
  }, []);
  
  /**
   * Stop an active recording session
   */
  const stopRecording = useCallback(async (recordingId: string): Promise<RecordingResult> => {
    try {
      setLoading(true);
      setError(null);
      
      const response = await Api.post('/api/recorder/stop', { recordingId });
      
      if (!response.data.success) {
        throw new Error(response.data.message || 'Failed to stop recording');
      }
      
      return response.data.data as RecordingResult;
    } catch (err: any) {
      const errorMessage = err.response?.data?.message || err.message || 'Failed to stop recording';
      setError(errorMessage);
      throw new Error(errorMessage);
    } finally {
      setLoading(false);
    }
  }, []);
  
  /**
   * Toggle pause/resume for a recording session
   */
  const togglePause = useCallback(async (recordingId: string, isPaused: boolean): Promise<boolean> => {
    try {
      setLoading(true);
      setError(null);
      
      const response = await Api.post('/api/recorder/pause', { recordingId, isPaused });
      
      if (!response.data.success) {
        throw new Error(response.data.message || 'Failed to toggle pause state');
      }
      
      return true;
    } catch (err: any) {
      const errorMessage = err.response?.data?.message || err.message || 'Failed to toggle pause state';
      setError(errorMessage);
      throw new Error(errorMessage);
    } finally {
      setLoading(false);
    }
  }, []);
  
  /**
   * Add a custom action to the recording
   */
  const addCustomAction = useCallback(async (
    recordingId: string, 
    request: CustomActionRequest
  ): Promise<boolean> => {
    try {
      setLoading(true);
      setError(null);
      
      const payload = {
        recordingId,
        actionType: request.actionType,
        actionTarget: request.actionTarget,
        actionValue: request.actionValue,
        additionalParams: request.additionalParams || {}
      };
      
      const response = await Api.post('/api/recorder/action', payload);
      
      if (!response.data.success) {
        throw new Error(response.data.message || 'Failed to add custom action');
      }
      
      return true;
    } catch (err: any) {
      const errorMessage = err.response?.data?.message || err.message || 'Failed to add custom action';
      setError(errorMessage);
      throw new Error(errorMessage);
    } finally {
      setLoading(false);
    }
  }, []);
  
  /**
   * Get recording status
   */
  const getRecordingStatus = useCallback(async (recordingId: string): Promise<any> => {
    try {
      const response = await Api.get(`/api/recorder/status?recordingId=${recordingId}`);
      
      if (!response.data.success) {
        throw new Error(response.data.message || 'Failed to get recording status');
      }
      
      return response.data.data;
    } catch (err: any) {
      const errorMessage = err.response?.data?.message || err.message || 'Failed to get recording status';
      console.error(errorMessage);
      return {};
    }
  }, []);
  
  /**
   * Get recording events
   */
  const getRecordingEvents = useCallback(async (recordingId: string, offset: number = 0): Promise<any[]> => {
    try {
      const response = await Api.get(`/api/recorder/events?recordingId=${recordingId}&offset=${offset}`);
      
      if (!response.data.success) {
        throw new Error(response.data.message || 'Failed to get recording events');
      }
      
      return response.data.data || [];
    } catch (err: any) {
      const errorMessage = err.response?.data?.message || err.message || 'Failed to get recording events';
      console.error(errorMessage);
      return [];
    }
  }, []);
  
  /**
   * Generate test code from a recording
   */
  const generateTestCode = useCallback(async (
    recordingId: string,
    options: {
      testName?: string;
      packageName?: string;
      className?: string;
      description?: string;
      generatePageObjects?: boolean;
      customizations?: Record<string, any>;
    } = {}
  ) => {
    try {
      setLoading(true);
      setError(null);
      
      const payload = {
        recordingId,
        testName: options.testName || 'Recorded Test',
        packageName: options.packageName || 'com.cstestforge.tests',
        className: options.className || 'RecordedTest',
        description: options.description || 'Test generated from recording session',
        generatePageObjects: options.generatePageObjects !== false,
        customizations: options.customizations || {}
      };
      
      const response = await Api.post('/api/recorder/generate', payload);
      
      if (!response.data.success) {
        throw new Error(response.data.message || 'Failed to generate test code');
      }
      
      return response.data.data;
    } catch (err: any) {
      const errorMessage = err.response?.data?.message || err.message || 'Failed to generate test code';
      setError(errorMessage);
      throw new Error(errorMessage);
    } finally {
      setLoading(false);
    }
  }, []);
  
  /**
   * Get active recording sessions
   */
  const getActiveSessions = useCallback(async (): Promise<RecordingSession[]> => {
    try {
      const response = await Api.get('/api/recorder/sessions');
      
      if (!response.data.success) {
        throw new Error(response.data.message || 'Failed to get active sessions');
      }
      
      return response.data.data || [];
    } catch (err: any) {
      const errorMessage = err.response?.data?.message || err.message || 'Failed to get active sessions';
      console.error(errorMessage);
      return [];
    }
  }, []);
  
  return {
    loading,
    error,
    startRecording,
    stopRecording,
    togglePause,
    addCustomAction,
    getRecordingStatus,
    getRecordingEvents,
    generateTestCode,
    getActiveSessions
  };
};


import axios, { AxiosInstance, AxiosRequestConfig, AxiosResponse } from 'axios';

// API response interface
interface ApiResponse<T = any> {
  success: boolean;
  message: string;
  data?: T;
  errors?: string[];
  timestamp?: number;
}

/**
 * API Service for handling HTTP requests to the backend
 */
export class ApiService {
  private axiosInstance: AxiosInstance;
  private baseURL: string;
  
  constructor() {
    this.baseURL = process.env.REACT_APP_API_URL || '/api';
    
    // Create axios instance with default configuration
    this.axiosInstance = axios.create({
      baseURL: this.baseURL,
      timeout: 30000, // 30 seconds timeout
      headers: {
        'Content-Type': 'application/json',
        'Accept': 'application/json'
      }
    });
    
    // Add request interceptor
    this.axiosInstance.interceptors.request.use(
      (config) => {
        // Get the auth token from local storage if available
        const token = localStorage.getItem('auth_token');
        
        // If token exists, add it to the request headers
        if (token && config.headers) {
          config.headers['Authorization'] = `Bearer ${token}`;
        }
        
        return config;
      },
      (error) => {
        return Promise.reject(error);
      }
    );
    
    // Add response interceptor
    this.axiosInstance.interceptors.response.use(
      (response) => {
        return response;
      },
      (error) => {
        // Handle authentication errors
        if (error.response && error.response.status === 401) {
          // Clear local storage and redirect to login page if needed
          localStorage.removeItem('auth_token');
          
          // If we're not already on the login page, redirect
          if (!window.location.pathname.includes('/login')) {
            window.location.href = '/login';
          }
        }
        
        return Promise.reject(error);
      }
    );
  }
  
  /**
   * Make a GET request
   */
  public get<T = any>(url: string, config?: AxiosRequestConfig): Promise<AxiosResponse<ApiResponse<T>>> {
    return this.axiosInstance.get<ApiResponse<T>>(url, config);
  }
  
  /**
   * Make a POST request
   */
  public post<T = any>(url: string, data?: any, config?: AxiosRequestConfig): Promise<AxiosResponse<ApiResponse<T>>> {
    return this.axiosInstance.post<ApiResponse<T>>(url, data, config);
  }
  
  /**
   * Make a PUT request
   */
  public put<T = any>(url: string, data?: any, config?: AxiosRequestConfig): Promise<AxiosResponse<ApiResponse<T>>> {
    return this.axiosInstance.put<ApiResponse<T>>(url, data, config);
  }
  
  /**
   * Make a DELETE request
   */
  public delete<T = any>(url: string, config?: AxiosRequestConfig): Promise<AxiosResponse<ApiResponse<T>>> {
    return this.axiosInstance.delete<ApiResponse<T>>(url, config);
  }
  
  /**
   * Make a PATCH request
   */
  public patch<T = any>(url: string, data?: any, config?: AxiosRequestConfig): Promise<AxiosResponse<ApiResponse<T>>> {
    return this.axiosInstance.patch<ApiResponse<T>>(url, data, config);
  }
  
  /**
   * Upload files
   */
  public upload<T = any>(url: string, formData: FormData, config?: AxiosRequestConfig): Promise<AxiosResponse<ApiResponse<T>>> {
    const uploadConfig: AxiosRequestConfig = {
      ...config,
      headers: {
        ...(config?.headers || {}),
        'Content-Type': 'multipart/form-data'
      }
    };
    
    return this.axiosInstance.post<ApiResponse<T>>(url, formData, uploadConfig);
  }
  
  /**
   * Download a file
   */
  public download(url: string, filename?: string, config?: AxiosRequestConfig): Promise<void> {
    return new Promise((resolve, reject) => {
      this.axiosInstance
        .get(url, {
          ...config,
          responseType: 'blob'
        })
        .then((response) => {
          // Create a temporary anchor element to trigger download
          const downloadUrl = window.URL.createObjectURL(new Blob([response.data]));
          const link = document.createElement('a');
          link.href = downloadUrl;
          link.setAttribute('download', filename || this.getFilenameFromResponse(response) || 'download');
          document.body.appendChild(link);
          link.click();
          
          // Clean up
          window.URL.revokeObjectURL(downloadUrl);
          document.body.removeChild(link);
          
          resolve();
        })
        .catch((error) => {
          reject(error);
        });
    });
  }
  
  /**
   * Extract filename from Content-Disposition header
   */
  private getFilenameFromResponse(response: AxiosResponse): string | null {
    const contentDisposition = response.headers['content-disposition'];
    
    if (!contentDisposition) {
      return null;
    }
    
    const filenameMatch = contentDisposition.match(/filename="?([^"]+)"?/);
    return filenameMatch ? filenameMatch[1] : null;
  }
  
  /**
   * Set authentication token
   */
  public setAuthToken(token: string): void {
    localStorage.setItem('auth_token', token);
  }
  
  /**
   * Get authentication token
   */
  public getAuthToken(): string | null {
    return localStorage.getItem('auth_token');
  }
  
  /**
   * Clear authentication token
   */
  public clearAuthToken(): void {
    localStorage.removeItem('auth_token');
  }
  
  /**
   * Check if user is authenticated
   */
  public isAuthenticated(): boolean {
    return !!this.getAuthToken();
  }
}

// Context interface
interface RecorderContextType {
  recordingSession: RecordingSession | null;
  isRecording: boolean;
  isPaused: boolean;
  recordingSteps: RecordingStep[];
  currentHighlight: any;
  addCustomAction: (actionType: string, actionTarget: string, actionValue: string, additionalParams?: any) => Promise<void>;
  deleteStep: (index: number) => void;
  editStep: (index: number, updatedStep: RecordingStep) => void;
}

// Default context values
const defaultContext: RecorderContextType = {
  recordingSession: null,
  isRecording: false,
  isPaused: false,
  recordingSteps: [],
  currentHighlight: null,
  addCustomAction: async () => {},
  deleteStep: () => {},
  editStep: () => {}
};

// Create context
export const RecorderContext = createContext<RecorderContextType>(defaultContext);

// Context provider component
interface RecorderProviderProps {
  children: React.ReactNode;
  value: RecorderContextType;
}

export const RecorderProvider: React.FC<RecorderProviderProps> = ({ children, value }) => {
  return (
    <RecorderContext.Provider value={value}>
      {children}
    </RecorderContext.Provider>
  );
};

// Custom hook to use the RecorderContext
export const useRecorderContext = () => {
  const context = React.useContext(RecorderContext);
  
  if (context === undefined) {
    throw new Error('useRecorderContext must be used within a RecorderProvider');
  }
  
  return context;
};
